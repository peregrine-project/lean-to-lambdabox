% Macros for the print version only: the leanblueprint annotations, which the web version gets from
% the leanblueprint plasTeX package.
% \lean: the names as typed (\detokenize), with the characters the monospaced font lacks set
% otherwise: Unicode subscript digits (as in forall₂) as subscripts, sigma and alpha in math.
\ExplSyntaxOn
\tl_new:N \l__bp_lean_tl
\tl_const:Nx \c__bp_sigma_tl { \tl_to_str:n { σ } }
\tl_const:Nx \c__bp_alpha_tl { \tl_to_str:n { α } }
\tl_const:Nx \c__bp_dot_tl { \tl_to_str:n { . } }
\tl_const:Nx \c__bp_us_tl { \tl_to_str:n { _ } }
\cs_new_protected:Npn \__bp_lean_set:n #1
 {
  \tl_set:Nx \l__bp_lean_tl { \tl_to_str:n {#1} }
  \tl_replace_all:Nnn \l__bp_lean_tl { ₀ } { \textsubscript{0} }
  \tl_replace_all:Nnn \l__bp_lean_tl { ₁ } { \textsubscript{1} }
  \tl_replace_all:Nnn \l__bp_lean_tl { ₂ } { \textsubscript{2} }
  \tl_replace_all:Nnn \l__bp_lean_tl { ₃ } { \textsubscript{3} }
  \tl_replace_all:NVn \l__bp_lean_tl \c__bp_sigma_tl { \ensuremath{\sigma} }
  \tl_replace_all:NVn \l__bp_lean_tl \c__bp_alpha_tl { \ensuremath{\alpha} }
 }
\cs_new_protected:Npn \__bp_lean:n #1
 { \__bp_lean_set:n {#1} \tl_use:N \l__bp_lean_tl }
% A name that may break after its dots and underscores (in tables and prose).
\cs_new_protected:Npn \__bp_leanbreak:n #1
 {
  \__bp_lean_set:n {#1}
  \tl_replace_all:NVn \l__bp_lean_tl \c__bp_dot_tl { \c__bp_dot_tl \allowbreak{} }
  \tl_replace_all:NVn \l__bp_lean_tl \c__bp_us_tl { \c__bp_us_tl \allowbreak{} }
  \tl_use:N \l__bp_lean_tl
 }
% Links to the Lean sources (README, "Links to the Lean sources"). build.sh writes the URLs of
% every citation, as the web version links them, to ../.build/print-links.tex (scripts/leanlinks.py
% print-links), which print.tex reads: \bpdefurl{kind}{key}{URL} for the kinds name (a Lean name),
% file (a path) and lines (path:lines, one line or range of a citation). A citation then links to
% its URL, and one without a URL is an error, except a \lean name (of a planned node). Without that
% file (`leanblueprint pdf`), the citations print without links.
\bool_new:N \g__bp_links_bool
\tl_new:N \l__bp_key_tl
\tl_new:N \l__bp_dir_tl
\seq_new:N \l__bp_items_seq
\definecolor{bpsrclink}{HTML}{0B5394}
\cs_new_protected:Npn \bpdefurl #1#2#3
 {
  \bool_gset_true:N \g__bp_links_bool
  \tl_gset:cx { bp@url@ #1 @ \tl_to_str:n {#2} } { \tl_to_str:n {#3} }
 }
\cs_new_protected:Npn \__bp_href:nn #1#2
 { \group_begin: \hypersetup{urlcolor=bpsrclink} \href{#1}{#2} \group_end: }
\msg_new:nnn { bp } { nolink }
 { No~URL~for~the~citation~(#1)~#2~in~../.build/print-links.tex:~run~blueprint/build.sh~pdf. }
% \__bp_link:nnn{kind}{key}{text}: the text, linked to the URL of the citation; an error when the
% URLs are loaded and the citation has none. \__bp_maylink:nnn: the same, without the error.
\cs_new_protected:Npn \__bp_link:nnn #1#2#3
 {
  \bool_if:NTF \g__bp_links_bool
   {
    \cs_if_exist:cTF { bp@url@ #1 @ #2 }
     { \exp_args:Nv \__bp_href:nn { bp@url@ #1 @ #2 } {#3} }
     { \msg_error:nnnn { bp } { nolink } {#1} {#2} #3 }
   }
   {#3}
 }
\cs_new_protected:Npn \__bp_maylink:nnn #1#2#3
 {
  \cs_if_exist:cTF { bp@url@ #1 @ #2 }
   { \exp_args:Nv \__bp_href:nn { bp@url@ #1 @ #2 } {#3} }
   {#3}
 }
\cs_generate_variant:Nn \__bp_link:nnn { nV , nx }
\cs_generate_variant:Nn \__bp_maylink:nnn { nx }
% The key of a path as written in TeX: break hints removed and \_ as _ (leanlinks.tex_arg).
\tl_const:Nx \c__bp_break_tl { \tl_to_str:n { \allowbreak } }
\tl_const:Nx \c__bp_breakgroup_tl { \tl_to_str:n { \allowbreak {} } }
\tl_const:Nx \c__bp_escus_tl { \c_backslash_str \c__bp_us_tl }
\cs_generate_variant:Nn \tl_replace_all:Nnn { NVV }
\cs_new_protected:Npn \__bp_path_key:n #1
 {
  \tl_set:Nx \l__bp_key_tl { \tl_to_str:n {#1} }
  \tl_replace_all:NVn \l__bp_key_tl \c__bp_breakgroup_tl { }
  \tl_replace_all:NVn \l__bp_key_tl \c__bp_break_tl { }
  \tl_replace_all:NVV \l__bp_key_tl \c__bp_escus_tl \c__bp_us_tl
  \tl_trim_spaces:N \l__bp_key_tl
 }
% The lines of a citation of the file whose key is \l__bp_key_tl, each line or range linked.
\cs_new_protected:Npn \__bp_lines:n #1
 {
  \seq_set_from_clist:Nn \l__bp_items_seq {#1}
  \seq_map_indexed_inline:Nn \l__bp_items_seq
   {
    \int_compare:nNnT {##1} > { 1 } { , }
    \__bp_link:nxn { lines } { \l__bp_key_tl : \tl_to_str:n {##2} } {##2}
   }
 }

     }

 }
