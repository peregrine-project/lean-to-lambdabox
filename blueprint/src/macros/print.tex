% Macros for the print version only: the leanblueprint annotations, which the web version gets from
% the leanblueprint plasTeX package.
% \lean: the names as typed (\detokenize), with the characters the monospaced font lacks set
% otherwise: Unicode subscript digits (as in forall₂) as subscripts, sigma and alpha in math.
\ExplSyntaxOn
\tl_new:N \l__bp_lean_tl
\tl_const:Nx \c__bp_sigma_tl { \tl_to_str:n { σ } }
\tl_const:Nx \c__bp_alpha_tl { \tl_to_str:n { α } }
\tl_const:Nx \c__bp_dot_tl { \tl_to_str:n { . } }
\tl_const:Nx \c__bp_us_tl { \tl_to_str:n { _ } }
\cs_new_protected:Npn \__bp_lean_set:n #1
 {
  \tl_set:Nx \l__bp_lean_tl { \tl_to_str:n {#1} }
  \tl_replace_all:Nnn \l__bp_lean_tl { ₀ } { \textsubscript{0} }
  \tl_replace_all:Nnn \l__bp_lean_tl { ₁ } { \textsubscript{1} }
  \tl_replace_all:Nnn \l__bp_lean_tl { ₂ } { \textsubscript{2} }
  \tl_replace_all:Nnn \l__bp_lean_tl { ₃ } { \textsubscript{3} }
  \tl_replace_all:NVn \l__bp_lean_tl \c__bp_sigma_tl { \ensuremath{\sigma} }
  \tl_replace_all:NVn \l__bp_lean_tl \c__bp_alpha_tl { \ensuremath{\alpha} }
 }
\cs_new_protected:Npn \__bp_lean:n #1
 { \__bp_lean_set:n {#1} \tl_use:N \l__bp_lean_tl }
% A name that may break after its dots and underscores (in tables and prose).
\cs_new_protected:Npn \__bp_leanbreak:n #1
 {
  \__bp_lean_set:n {#1}
  \tl_replace_all:NVn \l__bp_lean_tl \c__bp_dot_tl { \c__bp_dot_tl \allowbreak{} }
  \tl_replace_all:NVn \l__bp_lean_tl \c__bp_us_tl { \c__bp_us_tl \allowbreak{} }
  \tl_use:N \l__bp_lean_tl
 }
\NewDocumentCommand{\lean}{m}

% Kind badges. \bpkind{K} prints the badge of kind K. \bpnodes{chapter} prints the badges of the
% nodes of a chapter with their numbers, which \bpdefinenodes (generated/kinds.tex) records.
% \bpsetkind (generated/kinds.tex) records the kind, status and inherited sorries of each label;
% inside a node environment, \label then prints the badge of the node's kind and, for the statuses
% inherits (with the labels of the sorries) and planned, a status badge. The web version prints the
% same badges (scripts/bpkinds.py).
\makeatletter
\newcommand{\bpdefinenodes}[2]{\expandafter\def\csname bp@nodes@#1\endcsname{#2}}
\newcommand{\bpnodes}[1]{\ifcsname bp@nodes@#1\endcsname\csname bp@nodes@#1\endcsname\fi}
\newcommand{\bpkind}[1]{\fcolorbox{bpline#1}{bpfill#1}{\normalfont\textsf{\scriptsize\csname bpname#1\endcsname}}}
\newcommand{\bp@statusbadge}[2]{\ \fcolorbox{bpstatus#1}{white}{\normalfont\textsf{\scriptsize\textcolor{bpstatus#1}{#2}}}}
\newcommand{\bpsetkind}[4]{\expandafter\gdef\csname bp@kind@#1\endcsname{#2}%
  \expandafter\gdef\csname bp@status@#1\endcsname{#3}%
  \expandafter\gdef\csname bp@sorries@#1\endcsname{#4}}
\def\bp@inherits{inherits}
\def\bp@planned{planned}
\newcommand{\bp@tag}[1]{\ifcsname bp@kind@#1\endcsname
  \edef\bp@k{\csname bp@kind@#1\endcsname}\expandafter\bpkind\expandafter{\bp@k}%
  \edef\bp@s{\csname bp@status@#1\endcsname}%
  \ifx\bp@s\bp@inherits\bp@statusbadge{inherits}{INHERITS \csname bp@sorries@#1\endcsname}\fi
  \ifx\bp@s\bp@planned\bp@statusbadge{planned}{PLANNED}\fi
  \ \fi}
