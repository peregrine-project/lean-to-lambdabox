% Macros shared by the web and the print version. Chapters define no macro of their own.

% Theorem-like environments: the nodes of the dependency graph.

\newtheorem{proposition}[theorem]{Proposition}
\newtheorem{lemma}[theorem]{Lemma}
\newtheorem{corollary}[theorem]{Corollary}
\theoremstyle{definition}
\newtheorem{definition}[theorem]{Definition}

% Inline code (escape underscores: \code{prepare\_erasure}).
\newcommand{\code}[1]{\texttt{#1}}

% Schematic style: a bold run-in label, and at most one highlight per node.
\newcommand{\lead}[1]{\textbf{#1.}\ }
\definecolor{hlcolor}{HTML}{8B1A1A}
\newcommand{\hl}[1]{\textbf{\textcolor{hlcolor}{#1}}}

\newcommand{\stInherited}{\textbf{\textcolor{badgecolor}{[INHERITED]}}}

\newcommand{\stPlanned}{\textbf{\textcolor{badgecolor}{[PLANNED]}}}

% Node furniture. \srcloc{path}{line}: where the node's first \lean declaration is (checked by the
% audit). \inherited{labels}: the labels (L1, ..., TrProj; Section sec:trust-inherited) of the
% lean4lean sorry sources that the node's declarations depend on, exactly (checked by the audit).
% \planned: the node states a declaration that does not exist yet; its \lean names are the names
% the declarations will have (checked by the audit: none of them exists).
\definecolor{furniture}{HTML}{6A6A6A}
\newcommand{\srcloc}[2]{\par{\small\textcolor{furniture}{\texttt{#1:#2}}}\par}
\newcommand{\inherited}[1]{\par{\small\textcolor{furniture}{Inherited from lean4lean
  (Section~\ref{sec:trust-inherited}): #1}}\par}
\newcommand{\planned}{\par{\small\textcolor{furniture}{\stPlanned{} Not formalized: the declaration
  does not exist.}}\par}
