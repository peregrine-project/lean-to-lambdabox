\section{Erasability: proofs and type formers}
\label{sec:erases:erasability}

\begin{definition}[Arities]
\label{def:IsArity}
\lean{LeanToLambdaBox.IsArity, LeanToLambdaBox.IsArityUpTo}
\leanok
\uses{def:lean4lean-grounding}
\lead{In short} A model type is an \hl{arity} when it is a $\Pi$-telescope ending in a sort; \emph{up to conversion}, when some convertible type is one.
\begin{itemize}
\item \code{IsArity}: one rule for \code{sort}, one congruence for \code{forallE};
\item $\code{IsArityUpTo}\ \mathit{env}\ U\ \Gamma\ A$: some $A'$ with $A \equiv A'$ in $\Gamma$ under $U$ and \code{IsArity}\ $A'$.
\end{itemize}
\lead{Lean} Only the refined form is invariant under conversion, which is why \code{Erasable} uses it. [S, p.\ 8:61] leaves \code{isArity} implicit; [L, Def.\ 1] calls it a type scheme.
\end{definition}

\begin{definition}[Erasability]
\label{def:Erasable}
\lean{LeanToLambdaBox.Erasable}
\leanok
\uses{def:IsArity, def:lean4lean-grounding}
\lead{In short} $\Erasable\ \mathit{env}\ U\ \Gamma\ e$ holds when $e$ has \hl{some type that is a proof or a type former}, the source-side counterpart of the shipping eraser's test (Chapter~\ref{chap:eraser}).
\begin{itemize}
\item \lead{Given} some type $A$ of $e$;
\item \lead{Then} $A : \mathrm{Sort}\ 0$ (so $e$ is a \textbf{proof}), or $A$ is an arity up to conversion (so $e$ is a \textbf{type former}).
\end{itemize}
\lead{Lean} [S]'s $\code{isErasable}\ \Sigma\ \Gamma\ t$ (p.\ 8:61) transposed to lean4lean's model terms; the existential matters since unique typing fails in the source calculus ([L, p.\ 6]).
\end{definition}

\begin{lemma}[Syntactic arities are stable]
\label{lem:IsArity-inst}
\lean{LeanToLambdaBox.IsArity.inst, LeanToLambdaBox.IsArity.liftN,
LeanToLambdaBox.IsArity.instL, LeanToLambdaBox.IsArity.of_piBody}
\leanok
\uses{def:IsArity}
\lead{In short} Being a syntactic arity survives substitution, lifting and universe instantiation, and a telescope whose $\Pi$-body is a sort is one.
\end{lemma}
\begin{proof}
\leanok
\uses{def:IsArity}
\lead{Idea} Induction on the telescope: each operation fixes \code{.sort} and maps \code{.forallE} structurally; \code{piBody} is the telescope's tail.
\end{proof}

\begin{lemma}[Arity-up-to-conversion is stable]
\label{lem:IsArityUpTo-inst}
\lean{LeanToLambdaBox.IsArityUpTo.inst, LeanToLambdaBox.IsArityUpTo.weakN,
LeanToLambdaBox.IsArityUpTo.defeqDFC, LeanToLambdaBox.IsArityUpTo.instL}
\leanok
\uses{def:IsArity, def:lean4lean-grounding}
\lead{In short} \code{IsArityUpTo} survives instantiating a context entry, weakening the context, passing to a definitionally equal context, and universe instantiation.
\lead{Lean} Each carries the matching lean4lean context-surgery witness; the substitution form also carries an ordered environment.
\end{lemma}
\begin{proof}
\leanok
\uses{lem:IsArity-inst, def:IsArity}
\lead{Idea} The conversion witness moves by the matching upstream lemma, the syntactic arity by Lemma~\ref{lem:IsArity-inst}.
\end{proof}

\begin{lemma}[Conversion of the type]
\label{lem:IsArityUpTo-defeq}
\lean{LeanToLambdaBox.IsArityUpTo.defeq}
\leanok
\uses{def:IsArity, def:lean4lean-grounding}
\lead{In short} If $A''$ is convertible to $A$ and $A$ is an arity up to conversion, so is $A''$ -- the property plain \code{IsArity} lacks, and the reason the refined predicate exists.
\lead{Lean} Asks for a well-formed environment and context.
\end{lemma}
\begin{proof}
\leanok
\uses{def:IsArity, asm:lean4lean-trust}
\lead{Idea} Compose the conversions by transitivity, keeping the syntactic witness; transitivity is lean4lean's, so the footprint carries \code{sorryAx}.
\end{proof}

\begin{lemma}[Irrelevance survives reduction]
\label{lem:Erasable-defeq}
\lean{LeanToLambdaBox.Erasable.defeq}
\leanok
\uses{def:Erasable, def:lean4lean-grounding}
\lead{In short} \hl{The box-soundness core}: if $e$ is erasable and $e \equiv e'$, then $e'$ is erasable, so an irrelevant term stays irrelevant as it reduces -- what makes erasing it to $\Box$ sound (Chapter~\ref{chap:correctness}).
\lead{Lean} Asks for a well-formed environment and context.
\end{lemma}
\begin{proof}
\leanok
\uses{def:Erasable, asm:lean4lean-trust}
\lead{Idea} Keep the type $A$, move the typing witness along the conversion; the proof/arity disjunct carries over. Inherits \code{sorryAx}.
\end{proof}

\begin{lemma}[Stability of irrelevance]
\label{lem:Erasable-weakN}
\lean{LeanToLambdaBox.Erasable.weakN, LeanToLambdaBox.Erasable.inst,
LeanToLambdaBox.Erasable.defeqDFC, LeanToLambdaBox.Erasable.instL,
LeanToLambdaBox.Erasable.mono}
\leanok
\uses{def:Erasable, def:lean4lean-grounding}
\lead{In short} Irrelevance survives, one lemma per move:
\begin{itemize}
\item weakening the context (\code{weakN});
\item substituting a well-typed term (\code{inst});
\item passing to a definitionally equal context (\code{defeqDFC});
\item universe instantiation (\code{instL}), which also asks the substituted levels be well formed;
\item extending the environment (\code{mono}).
\end{itemize}
\end{lemma}
\begin{proof}
\leanok
\uses{lem:IsArityUpTo-inst, def:Erasable}
\lead{Idea} Move the typing witness by the matching upstream lemma, the arity disjunct by Lemma~\ref{lem:IsArityUpTo-inst}; $\mathrm{Sort}\ 0$ is fixed throughout, so the proof disjunct carries over.
\end{proof}

\begin{theorem}[Box propagation through application]
\label{thm:Erasable-app}
\lean{LeanToLambdaBox.Erasable.app}
\leanok
\uses{def:Erasable, def:lean4lean-grounding}
\lead{In short} If $f$ is irrelevant and $f\,a$ is well typed, $f\,a$ is irrelevant -- the licence for \lbox{}'s $\Box\ u \to \Box$ rule ([S, Sect.\ 7.1], [L, Def.\ 5]).
\begin{itemize}
\item \lead{Given} $f$ irrelevant, $f\,a$ well typed (function type and argument typing separate);
\item \lead{Then} proof case: $\mathrm{imax}\ u\ v \approx 0$ forces $B[a] : \mathrm{Sort}\ 0$; type-former case: the arity of $B$ pushes through substitution.
\end{itemize}
\lead{Status} \stProved{}, \stInherited{} through lean4lean (uniqueness of typing, sort inversion).
\end{theorem}
\begin{proof}
\leanok
\uses{def:Erasable, def:IsArity, lem:IsArityUpTo-defeq, lem:IsArityUpTo-inst,
asm:lean4lean-trust}
\lead{Idea} Retype $f$ by uniqueness of typing and split on \code{Erasable}: the proof case
inverts $\mathrm{imax}\ u\ v \approx 0$ to get $B : \mathrm{Sort}\ 0$; the type-former case
pushes the arity through substitution and conversion (Lemmas~\ref{lem:IsArityUpTo-inst},
\ref{lem:IsArityUpTo-defeq}).
\end{proof}

\begin{definition}[Relevance of an inductive type former]
\label{def:InformativeInd}
\lean{LeanToLambdaBox.InformativeInd, LeanToLambdaBox.vResultSort,
LeanToLambdaBox.neverZeroB}
\leanok
\uses{def:lean4lean-grounding}
\lead{In short} $\code{InformativeInd}\ \mathit{env}\ I$: \emph{env} declares $I$, and $I$'s result sort \hl{never evaluates to \code{Prop}} under any instantiation of its universe parameters.
\begin{itemize}
\item \code{vResultSort}: reads the result level off a $\Pi$-telescope;
\item \code{neverZeroB}: decides ``never zero'' (successor: never; parameter: maybe; \code{max}: one side suffices; \code{imax}: iff its right side is).
\end{itemize}
\lead{Lean} \textbf{Semantic} -- lean4lean's own never-zero predicate, the one its kernel theory reads for large elimination -- not the syntactic ``successor'' test; the premise [S, Fig.\ 18] leaves unspelled. Gates \code{proj} and the source $\iota$ rule.
\end{definition}

\begin{lemma}[The relevance test is sound and complete]
\label{lem:neverZeroB_sound}
\lean{LeanToLambdaBox.neverZeroB_sound, LeanToLambdaBox.neverZeroB_complete}
\leanok
\uses{def:InformativeInd}
\lead{In short} $\code{neverZeroB}\ l = \mathit{true}$ implies $l$ never evaluates to zero; $\mathit{false}$ implies the all-zero instantiation witnesses that it does.
\lead{Status} \stProved{}; the committed fixture \code{test/ledger.expected} records a footprint of \code{propext}, \code{Classical.choice} and \code{Quot.sound} only -- no \code{sorryAx}, hence no trust edge on this node.
\end{lemma}
\begin{proof}
\leanok
\uses{def:InformativeInd}
\lead{Idea} Induction on the level, unfolding level evaluation; the \code{imax} case uses $\mathrm{imax}\ u\ 0 = 0$.
\end{proof}

\begin{lemma}[Relevance is monotone]
\label{lem:InformativeInd-mono}
\lean{LeanToLambdaBox.InformativeInd.mono, LeanToLambdaBox.informativeInd_of_succ}
\leanok
\uses{def:InformativeInd}
\lead{In short} Relevance survives environment extension (needed by the \code{proj} arm of Theorem~\ref{thm:Erases-mono}); every syntactic successor witness also proves the semantic criterion.
\end{lemma}
\begin{proof}
\leanok
\uses{def:InformativeInd}
\lead{Idea} Transport the lookup along the extension; a successor level stays a successor, hence never zero.
\end{proof}

\begin{lemma}[The two criteria are different]
\label{lem:informativeInd_prod}
\lean{LeanToLambdaBox.informativeInd_prod, LeanToLambdaBox.not_informativeInd_and,
LeanToLambdaBox.prodVTy, LeanToLambdaBox.andVTy}
\leanok
\uses{def:InformativeInd}
\lead{In short} The semantic criterion accepts \code{Prod} (result sort $\max(u{+}1,v{+}1)$), which the syntactic successor test rejects; it rejects \code{And} (result sort \code{Prop}). These are the separating witnesses for the semantic criterion.
\lead{Lean} Each statement is conditional on the environment declaring the stated literal type.
\end{lemma}
\begin{proof}
\leanok
\uses{lem:neverZeroB_sound, def:InformativeInd}
\lead{Idea} Compute each result sort; discharge \code{Prod} via Lemma~\ref{lem:neverZeroB_sound}; \code{And}'s level is zero.
\end{proof}

\section{Reading the kernel environment: blocks, constructors, constants}
\label{sec:erases:readings}

\begin{definition}[Block data in \texorpdfstring{\lbox{}}{lambda-box} coordinates]
\label{def:IndInfo}
\lean{LeanToLambdaBox.IndInfo, LeanToLambdaBox.IndInfo.block,
LeanToLambdaBox.indBlockKername, LeanToLambdaBox.ctorFieldCounts}
\leanok
\uses{def:InductiveId, def:Kername, def:lean4lean-grounding}
\lead{In short} $\code{IndInfo}\ \mathit{env}\ I\ \mathit{iid}\ \mathit{np}\ \mathit{nfs}$: \emph{env}'s data for inductive $I$ \hl{in \lbox{} coordinates}, read off a well-formed declaration list below \emph{env}.
\begin{itemize}
\item some declaration list declares a mutual block with $\mathit{np}$ parameters whose $\mathit{iid}$-indexed type former is $I$;
\item $\mathit{iid}$'s block kername is \code{indBlockKername} of that block's names;
\item $\mathit{nfs}$ lists $I$'s constructors' field counts (the $\Pi$-binders beyond the parameters).
\end{itemize}
\lead{Lean} A one-field structure; the list is required well formed \emph{below} \emph{env}, which makes the predicate monotone. \code{indBlockKername} is the only place the frontend's naming convention enters the specification.
\end{definition}

\begin{remark}
\lead{Caveat} \code{indBlockKername} joins a block's member names with no separator, so \code{[A, BC]} and \code{[AB, C]} mint one key. Latent: every block installed at the eight rungs is a singleton. \lead{Since the snapshot} \stFixed{} \code{611f49e} makes a run refuse a colliding block key; the encoding is unchanged, so the clause-level risk in \ErasesEnv{}'s \code{blocks} stands.
\end{remark}

\leanok

\lean{LeanToLambdaBox.IndDeclOf}

\label{def:CtorOf}
\lean{LeanToLambdaBox.CtorOf}
\leanok
\uses{def:lean4lean-grounding}
\lead{In short} $\code{CtorOf}\ \mathit{env}\ c\ I\ k$: $c$ is the $k$-th constructor of $I$, read off a well-formed list below \emph{env}. Turns a \code{Lean.Expr.const} node into Rocq's \code{tConstruct}.
\end{definition}

\begin{definition}[The plain-constant reading]
\label{def:ConstOrigin}
\lean{LeanToLambdaBox.ConstOrigin, LeanToLambdaBox.VDeclDefines}
\leanok
\uses{def:lean4lean-grounding}
\lead{In short} $\code{ConstOrigin}\ \mathit{env}\ c$: Rocq's \code{tConst} reading -- $c$ is a definition, opaque constant, example, mutual-block member or axiom, exhibited off a list below \emph{env}.
\begin{itemize}
\item \code{VDeclDefines}: the per-declaration test; a block's type names and constructors, and the quotient declaration, count as neither;
\item \hl{positive}: exhibits the declaration rather than excluding the other two readings.
\end{itemize}
\lead{Lean} That this reading excludes the other two is a kernel-theory theorem, entering as an upstream ask (Chapter~\ref{chap:trust}), never a premise here.
\end{definition}

\begin{lemma}[The readings are monotone]
\label{lem:IndInfo-mono}
\lean{LeanToLambdaBox.IndInfo.mono, LeanToLambdaBox.IndArity.mono,
LeanToLambdaBox.CtorOf.mono, LeanToLambdaBox.ConstOrigin.mono,
LeanToLambdaBox.IndInfo.of_wf', LeanToLambdaBox.ConstOrigin.of_axiom}
\leanok
\uses{def:IndInfo, def:IndArity, def:CtorOf, def:ConstOrigin}
\lead{In short} Each of the four readings survives an extension $\mathit{env} \le \mathit{env}'$.
\begin{itemize}
\item \code{IndInfo.of\_wf'}: builds block data from an explicit declaration list;
\item \code{ConstOrigin.of\_axiom}: a constant declared at a closed axiom's type has that axiom as its origin -- the cheapest introduction, the one a hand-built environment uses.
\end{itemize}
\end{lemma}
\begin{proof}
\leanok
\uses{def:IndInfo, def:CtorOf, def:ConstOrigin}
\lead{Idea} Compose the witness's bound $\mathit{env}_0 \le \mathit{env}$ with $\mathit{env} \le \mathit{env}'$ by transitivity.
\end{proof}

\begin{remark}
This bound is the sole reason $\mathit{env}_0 \le \mathit{env}$ appears in all four predicates, and the reason \code{IndDeclOf} (Definition~\ref{def:IndDeclOf}) is on no rule.
\end{remark}

\begin{lemma}[Exchanging block data between coordinates]
\label{lem:IndInfo-arity}
\lean{LeanToLambdaBox.IndInfo.arity, LeanToLambdaBox.IndArity.indInfo,
LeanToLambdaBox.CtorOf.indInfo, LeanToLambdaBox.CtorOf.indArity}
\leanok
\uses{def:IndInfo, def:IndArity, def:CtorOf}
\lead{In short} The \lbox{} identifier is the only thing \code{IndInfo} adds beyond \code{IndArity} -- source-coordinate data exhibits it, since it names a position in its own block. A constructor's own declaration is its type's block, so \code{CtorOf} alone derives all the block data the \code{ctor} rule needs.
\end{lemma}
\begin{proof}
\leanok
\uses{def:IndInfo, def:CtorOf}
\lead{Idea} Forget or recover the index by list membership/indexing; the identifier is \code{indBlockKername} of the block's names.
\end{proof}

\section{The erasure relation and its eleven rules}
\label{sec:erases:relation}

\Erases{} is lean4lean's reading relation \TrExprS{} (Chapter~\ref{chap:source}), target
replaced by an \lbox{} term, \code{sort}/\code{forallE} dropped, \code{box} added -- so it
inherits lean4lean's trust boundary. It is non-deterministic, so ``the erasure of $e$'' is
never a function here; determinism returns only downstream (Chapter~\ref{chap:correctness}).

\begin{definition}[The erasure relation]
\label{def:Erases}
\lean{LeanToLambdaBox.Erases, LeanToLambdaBox.Erases.box, LeanToLambdaBox.Erases.bvar,
LeanToLambdaBox.Erases.fvar, LeanToLambdaBox.Erases.ctor, LeanToLambdaBox.Erases.const,
LeanToLambdaBox.Erases.app, LeanToLambdaBox.Erases.lam, LeanToLambdaBox.Erases.letE,
LeanToLambdaBox.Erases.proj, LeanToLambdaBox.Erases.lit, LeanToLambdaBox.Erases.mdata}
\leanok
\uses{def:Erasable, def:CtorOf, def:IndInfo, def:ConstOrigin, def:InformativeInd,
def:LBTerm, def:Kername, def:lean4lean-grounding}
\lead{In short} $\Erases\ \mathit{env}\ \mathit{Us}\ \Delta\ e\ t$ relates a source term $e$,
read in context $\Delta$ as \TrExprS{} reads it, to a \lbox{} image $t$. \hl{Eleven rules},
[S, Fig.\ 18]'s counterpart where one exists:

\medskip
\noindent\begin{tabular}{llp{0.16\linewidth}p{0.34\linewidth}p{0.14\linewidth}}
\toprule
Rule & Source & Image & Side condition & [S] \\
\midrule
\code{box} & any $e$ & $\Box$ & $e$ translates to an erasable $\mathit{ve}$ & \code{erases\_box} \\
\code{bvar} & $\code{bvar}\ i$ & $\code{bvar}\ i$ & $\Delta$ binds $i$ & \code{erases\_tRel} \\
\code{fvar} & $\code{fvar}\ x$ & $\code{fvar}\ x$ & $\Delta$ binds $x$ & \code{erases\_tRel} \\
\code{ctor} & $\code{const}\ c\ \mathit{us}$ & $\code{construct}\ \mathit{iid}\ k\ []$ & $\code{CtorOf}\ \mathit{env}\ c\ I\ k$, \code{IndInfo} for $I$ & \code{erases\_tConstruct} \\
\code{const} & $\code{const}\ c\ \mathit{us}$ & $\code{const}\ (\code{toKername}\ c)$ & \emph{env} declares $c$, $\code{ConstOrigin}\ \mathit{env}\ c$ & \code{erases\_tConst} \\
\code{app} & $f\ a$ & $f'\ a'$ & both erase & \code{erases\_tApp} \\
\code{lam} & $\lambda x{:}A.\,b$ & $\lambda x.\,b'$ & $A$ translates, $b$ erases under $\lambda$ & \code{erases\_tLambda} \\
\code{letE} & $\code{let}\ x{:}A := v;\ b$ & $\code{let}\ x := v';\ b'$ & $A,v$ translate, $v$ erases, $b$ erases under \code{let} & \code{erases\_tLetIn} \\
\code{proj} & $\code{proj}\ S\ i\ e$ & $\code{proj}\ \langle \mathit{iid},\mathit{np},i\rangle\ t$ & $\code{IndInfo}\ \mathit{env}\ S\ \mathit{iid}\ \mathit{np}\ [\mathit{nf}]$, $i<\mathit{nf}$, $\code{InformativeInd}\ \mathit{env}\ S$ & \code{erases\_tProj} \\
\code{lit} & $\code{lit}\ l$ & $t$ & $l$'s unfolding erases to $t$ & --- \\
\code{mdata} & $\code{mdata}\ d\ e$ & $t$ & $e$ erases to $t$ & --- \\
\bottomrule
\end{tabular}
\medskip

\begin{itemize}
\item \code{ctor}'s argument list is empty; arguments arrive through \code{app}; no rule
produces a case or fixpoint node, and an inductive type \emph{name} gets none either -- it is
erasable, so $\Box$ is its only image (Lemma~\ref{lem:Erases-indInfo_erasable});
\item both binder rules keep the \textbf{source} name at the snapshot, and \code{letE}'s
non-dependence flag is ignored; \code{const} alone carries a constant-lookup premise, since
\code{CtorOf}/\code{IndInfo} already exhibit \code{ctor}'s block.
\end{itemize}
\lead{Lean} [S]'s \code{erases\_tCase} and its subsingleton-elimination rule, and
\code{erases\_tFix}/\code{erases\_tCoFix}, have no counterpart (\Lower{}'s business; Lean has
no coinduction); its sort/$\forall$ premises are absorbed into \code{box}. [L]'s counterpart
is not the erasure function but the invariant of [L, Def.\ 10].
\lead{Since the snapshot} \stLanded{} \code{e7fb418}: \code{lam} and \code{letE} bind the target
name independently, matching \Lower{}, which already quantified it. Two consequences, both intended:
\Erases{} is demonstrably non-deterministic on a $\lambda$ source, and no statement in the chain
relates an emitted binder name to a source one. \code{512e190} adds \code{Erases.alpha}: source
bodies equal up to binder names have the same images.
\end{definition}

\begin{definition}[The box alternative]
\label{def:ErasesBox}
\lean{LeanToLambdaBox.ErasesBox}
\leanok
\uses{def:Erasable, def:LBTerm, def:lean4lean-grounding}
\lead{In short} $\code{ErasesBox}\ \mathit{env}\ \mathit{Us}\ \Delta\ e\ t$ abbreviates the alternative every inversion lemma shares: $e$'s model translation is \code{Erasable} and $t = \Box$.
\end{definition}

\begin{lemma}[The inversion kit]
\label{lem:Erases-sort_inv}
\lean{LeanToLambdaBox.Erases.sort_inv, LeanToLambdaBox.Erases.forallE_inv,
LeanToLambdaBox.Erases.bvar_inv, LeanToLambdaBox.Erases.fvar_inv,
LeanToLambdaBox.Erases.const_inv, LeanToLambdaBox.Erases.ctor_inv,
LeanToLambdaBox.Erases.app_inv, LeanToLambdaBox.Erases.lam_inv,
LeanToLambdaBox.Erases.letE_inv, LeanToLambdaBox.Erases.proj_inv,
LeanToLambdaBox.Erases.lit_inv, LeanToLambdaBox.Erases.mdata_inv}
\leanok
\uses{def:Erases, def:ErasesBox, def:CtorOf, def:IndInfo, def:ConstOrigin,
def:InformativeInd}
\lead{In short} One lemma per source head: a disjunction whose first alternative is always \hl{\code{ErasesBox}}.
\begin{itemize}
\item sort, $\Pi$-type: only \code{ErasesBox};
\item bound/free variable: box, or the lookup premise;
\item constant: box; or \code{ctor} (\code{CtorOf}, \code{IndInfo}); or \code{const} (lookup, \code{ConstOrigin}) -- nothing excludes two of the three, the kernel theory's business;
\item application, $\lambda$, \code{let}, projection, literal, \code{mdata}: box, or the matching structural premises.
\end{itemize}
\lead{Lean} One lemma is keyed on the \textbf{target}: any source erasing to $\code{construct}\ \mathit{iid}\ k\ \mathit{args}$ has $\mathit{args}=[]$ and exhibited block data (source left open, since \code{lit}/\code{mdata} pass images through).
\end{lemma}
\begin{proof}
\leanok
\uses{def:Erases, def:ErasesBox}
\lead{Idea} Case analysis on the relation, except the target-keyed fact, proved by generalising the image and inducting so \code{lit}/\code{mdata} pass it through.
\end{proof}

\begin{theorem}[Both constant readings fire]
\label{thm:erases_ctor_fires}

\begin{theorem}[Erasure commutes with substitution]

\begin{theorem}[Open, erase, close]
\label{thm:Erases-abstract}

\item \lead{Given} $e$ erases to $t$ at depth $\mathit{dk}$'s free variable $v_0$, no loose bound variable at or above $\mathit{dk}$;

\begin{itemize}
\item \code{Erasable} ships weakening but no inverse, nor does lean4lean's typing judgement, so the obligation is \textbf{named as a premise}, never assumed as an axiom;
\item its hard half is the arity disjunct, needing a $\Pi$ inversion through a lift; a zero lift satisfies it as the identity, without vacuity.
\end{itemize}
\end{definition}

\begin{remark}
\lead{Caveat} Not a field of any capstone bundle: a named premise, never an axiom of Chapter~\ref{chap:trust}. Its non-vacuity witness is weak -- only at a zero lift, where the statement degenerates.
\end{remark}

\begin{definition}[Two scope conditions for strengthening]
\label{def:NoProj}
\lean{LeanToLambdaBox.NoProj, LeanToLambdaBox.NoProjBinders,
LeanToLambdaBox.NoProj.toIsUnique, LeanToLambdaBox.NoProj.toNoProjBinders,
LeanToLambdaBox.NoProj.toConstructor, LeanToLambdaBox.NoProjBinders.toConstructor,
LeanToLambdaBox.NoProj.natLitToConstructor, LeanToLambdaBox.NoProj.strLitToConstructor}
\leanok
\uses{def:lean4lean-grounding}
\lead{In short} Two projection-freeness conditions the strengthening theorem needs, since lean4lean's own uniqueness at a projection holds only \hl{up to definitional equality}.
\begin{itemize}
\item \code{NoProj}: projection-freeness everywhere, \textbf{including} a \code{let} binder's type (stronger than lean4lean's own condition, since \code{letE} records that type in the entry);
\item \code{NoProjBinders}: the weaker condition, only at $\lambda$/$\forall$ binder types and a \code{let}'s type/value -- where strengthening spends uniqueness;
\item projections stay permitted computationally; the binder clauses cannot be dropped -- equational uniqueness at a projection is \textbf{false}, not merely unproved.
\end{itemize}
\lead{Lean} Every literal's one-step unfolding satisfies both conditions.
\end{definition}

\begin{lemma}[The two conditions are different]
\label{lem:noProjBinders_ofNatBody}
\lean{LeanToLambdaBox.ofNatBody, LeanToLambdaBox.noProjBinders_ofNatBody,
LeanToLambdaBox.noProj_ofNatBody_refuted}
\leanok
\uses{def:NoProj}
\lead{In short} The prepared body of a class method, $\mathrm{fun}\ (\alpha : \mathrm{Type})\ (x : \mathrm{Nat})\ (\mathit{self} : \code{OfNat}\ \alpha\ x) \Rightarrow \mathit{self}.1$, satisfies \code{NoProjBinders} (its three binder types are projection-free) and provably \textbf{fails} \code{NoProj} (its projection sits in the body). That is what the relaxation buys: the corpus contains class-method projections.
\end{lemma}
\begin{proof}
\leanok
\uses{def:NoProj}
\lead{Idea} Unfold both predicates at the literal term.
\end{proof}

\begin{theorem}[Strengthening at depth]
\label{thm:Erases-strengthen_fvlift}
\lean{LeanToLambdaBox.Erases.strengthen_fvlift,
LeanToLambdaBox.Erases.strengthen_fvlift_binders}
\leanok
\uses{def:Erases, def:ErasableStrengthen, def:NoProj, def:lean4lean-grounding}
\lead{In short} A derivation at a free-variable extension of a context also holds at the smaller context, \hl{modulo the commissioned premise}.
\begin{itemize}
\item \lead{Given} the source satisfies the scope condition, and is \emph{already translatable} at the small context;
\item \lead{Given} the commissioned premise of Definition~\ref{def:ErasableStrengthen};
\item \lead{Then} the derivation strengthens to the small context.
\end{itemize}
\lead{Lean} Two versions, one per scope condition; the weaker pays in \code{box} (conversion
uniqueness -- instead of equational uniqueness -- plus Lemma~\ref{lem:Erasable-defeq}), moving
two premises to full/well-formed from free-variable-only/ordered.
\lead{Status} \stProved{}, \stInherited{} through lean4lean on the weak-scope version only;
neither version is in the capstone's import closure.
\end{theorem}
\begin{proof}
\leanok
\uses{def:Erases, lem:erases_weakFV, lem:Erasable-defeq, def:ErasableStrengthen,
asm:lean4lean-trust}
\lead{Idea} Push the small-context translation up by re-proved weakening, identify it with the
derivation's own witness by uniqueness, and every translation-bearing arm becomes a rewrite;
the commissioned premise is spent once, in \code{box}. Lean4lean's inverse-weakening lemma is
unusable here: it recovers a small-context translation only up to definitional equality,
survivable in \code{box} but fatal in \code{lam}, where the induction hypothesis needs an
equation between binder types.
\end{proof}

\begin{theorem}[Context uniformity]
\label{thm:erases_uniform_closed}
\lean{LeanToLambdaBox.erases_strengthen_closed, LeanToLambdaBox.erases_uniform_closed,
LeanToLambdaBox.erases_uniform_of_nil}
\leanok
\uses{def:Erases, def:ErasableStrengthen, def:NoProj, def:LBClosed,
def:lean4lean-grounding}
\lead{In short} For a closed, free-variable-free source with projection-free binders and closed image, erasure does \hl{not depend on the context} it was produced at.
\begin{itemize}
\item \lead{Given} such a source and image (closedness/freeness follow from the assumed translation);
\item \lead{Then} a derivation at one context holds at every other, via the empty context.
\end{itemize}
\lead{Lean} The one-sided corollary needs \textbf{no} commissioned premise; only the round trip does.
\lead{Status} \stProved{}, \stInherited{} through lean4lean.
\end{theorem}
\begin{proof}
\leanok
\uses{thm:Erases-strengthen_fvlift, thm:erases_weak_any, asm:lean4lean-trust}
\lead{Idea} Compose Theorem~\ref{thm:Erases-strengthen_fvlift} (weak-scope) with Theorem~\ref{thm:erases_weak_any}; the second leg cannot be Lemma~\ref{lem:erases_weakFV}'s weakening, since the target context is arbitrary.
\end{proof}

\section{Environment erasure}
\label{sec:erases:env}

The remaining definitions lift the term-level relation to whole environments: \ErasesEnv{}
for a compiled program, \LowerEnv{} for the pass layer's output (Chapter~\ref{chap:lowering}).

\begin{definition}[Environment erasure]
\label{def:ErasesEnv}
\lean{LeanToLambdaBox.ErasesEnv, LeanToLambdaBox.ErasesEnv.mk}
\leanok
\uses{def:Erases, def:ConstOrigin, def:IndInfo, def:IndDeclOf, def:InformativeInd,
def:ReachableFrom, def:ElimDecl, def:CasesOnShape, def:isCasesOnName, def:Kername,
def:LBTerm-envLookup, def:GlobalDeclarations}
\lead{In short} $\ErasesEnv\ \mathit{env}\ \mathit{bo}\ \Gamma_{\mathrm{spec}}\ t$: $\Gamma_{\mathrm{spec}}$ is a correct, \hl{dependency-selective} erasure of the source environment seen from program $t$, with $\mathit{bo}$ the \textbf{compiler body table}. Seven clauses, each triggered by $t$'s reachability of a kername -- [S]'s \code{erases\_deps}, not the pointwise erasure of [S, Table 2]:
\begin{itemize}
\item \code{keys}: the environment's keys are distinct (the \code{fresh\_global} half of [S]'s well-formedness, so a shadowing entry cannot retarget a pass);
\item \code{deps}: every kername the program reaches is declared;
\item \code{tabled}: a constant the compiler table defines is a plain constant of the model; unlike the others it has no reachability trigger, refuting a reading only at constants the erasure emits no key for;
\item \code{defns}: every reached, tabled constant's entry has a body erasing the compiler body at \textbf{every level scope and instantiation} --- \stRefuted{} (status line below);
\item \code{axioms}: a reached, body-less, non-eliminator constant is declared body-less;
\item \code{blocks}: a reached inductive block is in the environment's own list, with the target's arity/propositionality data;
\item \code{elims}: a reached \code{casesOn} of a \emph{relevant} inductive has the canonical eliminator body.
\end{itemize}
\lead{Lean} No constructor clause: \code{blocks} already supplies its arity.
\lead{At the snapshot} \stRefuted{} The \code{defns} clause is unsatisfiable. Read at the empty level scope the instantiation is the identity, and no arm of \Erases{} produces an image for a $\lambda$ whose domain is a level-parameter sort there (\code{no\_erases\_lam\_sort\_param}), so one universe-polymorphic tabled body refutes the clause. Mechanised twice, axioms \code{propext}/\code{Classical.choice}/\code{Quot.sound}, no \code{sorry}.
\lead{Since the snapshot} \stLanded{} \code{92105c6}: the clause reads the tabled body at the declaration's own level parameters, through a new column \code{SourceTable.levels?} carried by \code{ErasesEnv}, \code{SpecContent}, \code{SpecEnv}, \code{RegInvShape'} and \code{ErasureBridge} --- MetaRocq's own shape. The clause is not unconditionally satisfiable: nothing in the type ties the column to the body table, and at the rungs only the table rules the bad instance out.
\end{definition}

\begin{definition}[The emitted environment as a lowered, pruned image]
\label{def:LowerEnv}
\lean{LeanToLambdaBox.LowerEnv}

\uses{def:Lower, def:LowerFix, def:ElimDecl, def:LBTerm-envLookup, def:GlobalDeclarations}
\lead{In short} $\LowerEnv\ \Gamma_{\mathrm{spec}}\ \Gamma$: the emitted \lbox{} environment $\Gamma$ is the \hl{lowered, pruned image} of $\Gamma_{\mathrm{spec}}$. Eight fields:
\begin{itemize}
\item \code{keys}: the emitted keys are distinct;
\item \code{defs}: a shared-key body is pass-related, or one member of a lowered fixpoint block;
\item \code{defsTotal}: every non-runtime-key definition survives pruning as a definition -- without it, a pruned definition would satisfy \code{defs} vacuously while the target is stuck at its constant node;
\item \code{axioms}: a body-less constant stays body-less or is pruned;
\item \code{inds}: inductive blocks carry over unchanged, since the target's constructor-arity and propositionality queries read the specification's numbers;
\item \code{sub}: pruning only removes;
\item \code{closed}, \code{specClosed}: both environments' bodies are closed.
\end{itemize}
\end{definition}

\section*{Deviations and caveats}
\begin{itemize}
\item Constructors are \emph{applied form}: \code{ctor} fires at the bare head, arguments via \code{app} (cf.\ F-ETA2); no case or fixpoint rule, \code{Lean.Expr} having neither node (\Lower{}'s business).
\item \code{mdata}/\code{lit} are added; \code{proj}'s relevance premise, free in Rocq's typing, is shown necessary by Theorem~\ref{thm:erases_proj_needs_informative} (design tag N18).
\item The de Bruijn variable rule splits into \code{bvar}/\code{fvar} for locally-nameless opening (Theorem~\ref{thm:Erases-abstract}); \ErasesEnv{} is dependency-selective, not pointwise.
\item \stRefuted{} \lead{At the snapshot} \code{defns} over every level scope is unsatisfiable at a polymorphic tabled body; \stOpen{} \ErasesEnv{} and \LowerEnv{} remain the capstone's two bridge fields, produced by no theorem (Chapter~\ref{chap:trust}).
\item \lead{Deviation} Quantifying \code{defns} over every scope was a chosen divergence: [S] states the clause once at the constant's own universe context and derives the instantiated form under typing side conditions. Reverted at \code{92105c6}.
\item \code{ErasesUniform.lean} -- \code{ErasableStrengthen}, the \code{NoProj} conditions, and Theorem~\ref{thm:erases_uniform_closed} -- lies outside the capstone's import closure, with no consumer.
\end{itemize}
