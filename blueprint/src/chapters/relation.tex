\section{The erasure relation}\label{sec:relation-erases}

\begin{definition}[Side conditions on inductive types]
  \label{def:EraseProof-Subsingleton}
  \lean{EraseProof.Subsingleton, EraseProof.PropInductive, EraseProof.IsProof,
    EraseProof.FieldsErasable, EraseProof.CasesReduced}
  \leanok
  \uses{def:EraseProof-MBinding, def:EraseProof-IsErasable, def:EraseProof-SrcEval}
  \srcloc{proof/EraseProof/Spec.lean}{1033}
  \lead{In short} MetaRocq's conditions on eliminations, \hl{stated on Metalean's constructor
  nodes}.
  \begin{itemize}
  \item \code{PropInductive E B T}: \code{T}'s Metalean block has level \code{0};
  \item \code{Subsingleton E B T}: whenever a constructor node of \code{T} is a proof
        (\code{IsProof}), \code{T} has at most one constructor and its fields are proofs;
  \item \code{FieldsErasable E B T}: every field of a typed constructor node is erasable;
  \item \code{CasesReduced iv us}: one constructor, and the discriminant's sort at \code{us}
        (\code{casesSortLevel}) is not never zero (\code{Level.isNeverZero}); the eraser's
        reduced-case test.
  \end{itemize}
  \lead{Reference} \code{isPropositional} (\code{PCUICFirstorder.v}); \code{Is\_proof},
  \code{Subsingleton} (\code{PCUICElimination.v}).
\end{definition}

\begin{definition}[Erasure relation]
  \label{def:EraseProof-Erases}
  \lean{EraseProof.Erases, EraseProof.ErasesAlt, EraseProof.IndIdOf, EraseProof.SpecialSpine,
    EraseProof.ctorFields, EraseProof.blockKername}
  \leanok
  \uses{def:EraseProof-MBinding, def:EraseProof-TrM, def:Erasure-declExclusion,
    def:EraseProof-LBEval, def:EraseProof-Subsingleton, def:EraseProof-IsErasable,
    def:EraseProof-EvalEnv, def:EraseProof-EvalEnv-isAtom, def:EraseProof-EvalEnv-etaSpine,
    def:EraseProof-ConstsIn, def:LBTerm, def:toKername, def:Erasure-TravCtx}
  \srcloc{proof/EraseProof/Spec.lean}{1064}
  \lead{In short} \code{Erases E B lps $\sigma$ $\Gamma$ $\Delta$ e t}: the $\lambda_\Box$ term
  \code{t} \hl{is an erasure of the Lean term \code{e}}.
  \begin{tabular}{p{0.2\linewidth}p{0.66\linewidth}}
  \toprule
  Rule & Erasure, with its side condition \\
  \midrule
  \code{box} & an erasable term (\code{ErasableS}) erases to $\Box$ \\
  \code{bvar}, \code{fvar}, \code{app}, \code{const}, \code{mdata} & one rule each;
    \code{const} for non-atom, non-special constants \\
  \code{lam}, \code{letE} & under the binder; \hl{the binder's translation is a type}
    (\code{lam}), the value's translation is typed (\code{letE}) \\
  \code{ctor} & a saturated constructor spine to \code{construct} applied to the erased
    arguments; \hl{the inductive is not propositional} \\
  \code{cases} & a \code{casesOn} with at least its arity to \code{case} on the erased major, one
    \code{ErasesAlt} branch per constructor; \hl{\code{Subsingleton}} and not
    \code{CasesReduced} \\
  \code{casesReduced} & one constructor without index field, \code{FieldsErasable}: the erased
    alternative applied to one $\Box$ per field, the discriminant dropped (DV-25) \\
  \code{eta} & an under-applied spine erases as its $\eta$-expansion (DV-23) \\
  \code{proj} & \code{proj} on the erased structure; \code{Subsingleton}, and the block in
    \code{Prop} or its declared sort never zero \\
  \code{litZero}, \code{litSucc} & a \code{Nat} literal to its constructor chain
    (\code{nat := .peano}) \\
  \bottomrule
  \end{tabular}
  \begin{itemize}
  \item \code{ErasesAlt}: an alternative under its field binders, $\lambda$s peeled (each binder's
        translation a type) or $\eta$-introduced (\code{intro}, on any alternative, $\lambda$s
        included), as \code{lambdaOrIntroToArity};
        \code{blockKername}: the shipping \code{mutualBlockKername}.
  \end{itemize}
  \lead{Why typed binders} substitution moves the body's typings into the context of the
  substituted binder, only $\equiv$ to the original one; that conversion needs the binder typed,
  which an untyped translation does not give (lean4lean's \code{TrExprS.lam} carries it too).
  \lead{Why \code{intro} on $\lambda$s} substituting a $\lambda$ for an alternative that is a
  variable yields a $\lambda$ alternative related to the $\beta$-redex of its erased branch, which
  only $\eta$-introduction derives (\code{ErasesAlt.inst}).
  \lead{Reference} MetaRocq \code{erases} (\code{Extract.v}): \code{erases\_tConstruct},
  \code{erases\_tCase}, \code{erases\_tProj}, \code{erases\_box}.
\end{definition}

  \label{def:EraseProof-ErasesDeps}

    def:EraseProof-ProgEnv, def:EraseProof-LBEval, def:EraseProof-Erases,
    lem:EraseProof-Erases-abstract, def:EraseProof-IsErasable, def:EraseProof-EvalEnv,
    def:EraseProof-EvalEnv-isAtom, def:EraseProof-EvalEnv-etaSpine, def:EraseProof-SrcEval,
    def:EraseProof-SrcValue, def:EraseProof-ConstsIn, def:LBTerm}
  \srcloc{proof/EraseProof/Relation/Inst.lean}{1327}
  \lead{In short} Substituting \hl{a typed source value} into a typed term commutes with
  erasure.
  \begin{itemize}
  \item \lead{Given} \code{ProgEnv}, \code{EnvWF E}, \code{E[$\Gamma$] $\vdash$ ok},
        \code{SubEnv $\sigma$.decls P}; \code{SrcValue $\sigma$ a}; typed translations of
        \code{a} and \code{b}; erasures \code{a'} (closed) and \code{b'};
  \item \lead{Then} \code{Erases \ldots\ (b.instantiate1' a) (csubst a' 0 b')}
        (\code{inst}, and \code{inst\_let} under a \code{let});
  \item \code{ErasesAlt.inst}: the same for an alternative under its field binders, at the
        index of the last binder, the field step of $\lambda_\Box$'s \code{iotaRed}.
  \end{itemize}
  \lead{Why the premises} \code{E[$\Gamma$] $\vdash$ ok} rebuilds the binders' translations after
  substitution; \code{SubEnv} with \code{ProgEnv} makes declared types closed, so the
  $\eta$-rule commutes with substitution.
  \lead{Why values} a substituted partial spine would be erased as a whole, while its erasure
  is the $\eta$-expansion applied (DV-23).
  \lead{Reference} \code{erases\_subst0} (\code{ESubstitution.v}).
\end{lemma}
\begin{proof}
  \uses{def:EraseProof-TrLevel, def:EraseProof-natLitM, lem:EraseProof-TrM-weakBV,
    lem:EraseProof-SatTr-subst, lem:EraseProof-DCong-defeq, lem:EraseProof-ProgEnv-lookup,
    lem:EraseProof-TypeEq-sort_ne_forallE, def:Erasure-declExclusion, def:Erasure-Pure-instLevels,
    def:EraseProof-Subsingleton, lem:EraseProof-Erases-extend, def:EraseProof-DepClosed,
    def:toKername, def:Erasure-TravCtx}
  \leanok
  \lead{Idea} \code{instG}: induction on both derivations, generalized over a substitution at any
  depth, with a head invariant (\code{HeadInv}) for spines; erasability under substitution for the
  \code{box} rule.
\end{proof}

\section*{Caveats}
\begin{itemize}
\item \lead{Caveat} Constructors and \code{casesOn} are $\eta$-expanded inside the relation, as
  the eraser emits them; MetaRocq $\eta$-expands in a separate pass.
\item \lead{Caveat} The relation has no rule for recursors or quotients in Phase I.
\item \lead{Caveat} \code{casesReduced} does not model the substitution of index fields; the
  fragment test excludes constructors with index fields (Section~\ref{sec:scope-fragment}).
\end{itemize}
