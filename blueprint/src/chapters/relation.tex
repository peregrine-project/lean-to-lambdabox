  \label{def:EraseProof-Erases}
  \lean{EraseProof.Erases, EraseProof.BoxCert, EraseProof.noCert, EraseProof.ErasesAlt, EraseProof.ErasesInd, EraseProof.ErasesDecl, EraseProof.ErasesDeps}
  \leanok

  \srcloc{proof/EraseProof/Spec.lean}{1180}
  \lead{In short} \hl{\code{Erases cert $\sigma$ $\Gamma$ e t}: the $\lambda_\Box$ term \code{t} erases the marked term \code{e}}, with no typing premise.
  \begin{itemize}

  \item \code{ErasesAlt}, \code{ErasesInd}, \code{ErasesDecl}, \code{ErasesDeps}: alternatives, inductives, constants and the environment.
  \end{itemize}
  \lead{Reference} \code{MR E/Extract.v:erases}, \code{erases\_deps}; DV-16, DV-25, DV-32.

\begin{lemma}[Inversion and closedness]
  \label{lem:EraseProof-Erases-app-inv}
  \lean{EraseProof.Erases.app_inv, EraseProof.AbsFV.LBV, EraseProof.AbsFV.LBV_foldl, EraseProof.AbsFV.LBV_mkAppN, EraseProof.AbsFV.LBV_of_toCL?, EraseProof.Erases.closed, EraseProof.Erases.ctx, EraseProof.Erases.appS, EraseProof.Erases.ctor_inv, EraseProof.Erases.cases_inv}

  \srcloc{proof/EraseProof/Relation/Basic.lean}{453}
  \lead{In short} \hl{\code{Erases} inverts on applications, constructors and \code{casesOn}}, is closed, and ignores the context unless a certificate reads it.
  \begin{itemize}
  \item \code{Erases.app\_inv}, \code{Erases.ctor\_inv}, \code{Erases.cases\_inv}: an application, a constructor spine, a \code{casesOn} spine erases only by its rules;
  \item \code{Erases.appS}: an application erases to an application of erasures, for every head;
  \item \code{Erases.closed}, \code{AbsFV.LBV}, \code{AbsFV.LBV\_of\_toCL?}: a term with bounded loose variables erases to a closed one;

\begin{proof}
  \leanok
  \lead{Idea} Inversion of the rule that fits the head; special spines carry their arguments in one rule.
\end{proof}

\section{Instantiation}\label{sec:relation-inst}

Substituting an evaluated, erased argument keeps the erasure.

\begin{lemma}[Erasure is stable under instantiation]
  \label{lem:EraseProof-Erases-inst}
  \lean{EraseProof.Erases.inst, EraseProof.Erases.instG, EraseProof.Erases.inst_let, EraseProof.ErasesAlt.inst, EraseProof.Erases.instLevels}
  \leanok
  \uses{def:EraseProof-Erases, lem:EraseProof-Erases-inst-aux}
  \srcloc{proof/EraseProof/Relation/Inst.lean}{209}
  \lead{In short} \hl{If $b$ erases to $b'$ and $a$ to $a'$, then $b[a]$ erases to $b'[a']$}; the same for \code{let}, alternatives and universe instantiation.
  \begin{itemize}
  \item \code{Erases.inst}, \code{Erases.inst\_let}, \code{ErasesAlt.inst}: at every depth, for \code{noCert}, with $a$ having bounded loose variables;
  \item \code{Erases.instG}: the general form at depth \code{d} and any context;
  \item \code{Erases.instLevels}: instantiating universe parameters leaves the erasure unchanged.
  \end{itemize}
  \lead{Reference} \code{MR E/ErasureFunctionProperties.v:erases\_subst}.
\end{lemma}
\begin{proof}
  \uses{lem:EraseProof-Erases-inst-aux}
  \leanok
  \lead{Idea} Induction on the derivation, with \code{csubst} commuting with the rules; no certificate is transported.
\end{proof}

\section*{Caveats}
\begin{itemize}
\item Instantiation holds for \code{noCert}, the relation of the simulation; \code{Erases (CLCert env)} is not instantiated.
\item \code{LBTerm} has no \code{tCoFix}, \code{tLazy} or \code{tForce} (DV-16).
\item The lemmas of \code{TargetLift} are copies; they are cited by the lemmas that use them.
\end{itemize}
