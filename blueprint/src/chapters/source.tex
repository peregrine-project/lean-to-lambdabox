\section{Erasability}\label{sec:source-erasable}

\begin{definition}[Erasable terms]
  \label{def:EraseProof-IsErasable}
  \lean{EraseProof.IsErasable, EraseProof.IsArity, EraseProof.ErasableS}
  \leanok
  \uses{def:EraseProof-MBinding, def:EraseProof-TrM}
  \srcloc{proof/EraseProof/Spec.lean}{864}
  \lead{In short} A Metalean term is erasable when \hl{its type is an arity or a proposition}.
  \begin{itemize}
  \item \code{IsArity E $\Gamma$ T}: \code{T} equals, as a type, a $\Pi$-telescope ending in a
        sort;
  \item \code{IsErasable E $\Gamma$ t}: some \code{T} types \code{t}, and \code{T} is an arity or
        has type \code{Prop};
  \item \code{ErasableS E B lps $\Gamma$ $\Delta$ e}: some translation of the Lean term \code{e} is
        erasable.
  \end{itemize}
  \lead{Reference} MetaRocq \code{isErasable}; Letouzey's Definition 3.
\end{definition}

  \label{lem:EraseProof-IsErasable-inst}

  \lead{Reference} \code{erasure/theories/ESubstitution.v}, \code{EArities.v}.
\end{lemma}
\begin{proof}
  \leanok
  \lead{Idea} Metalean substitution, level instantiation and inversion of typing.
\end{proof}

\section{Source semantics}\label{sec:source-eval}

The source semantics is big-step, weak call-by-value on Lean terms, with rules for inductive
types. It is not Metalean's weak-head reduction, which is call-by-name; \code{SrcEval.defeq}
justifies each rule by a Metalean definitional equality.

\begin{definition}[Evaluation environments]
  \label{def:EraseProof-EvalEnv}
  \lean{EraseProof.EvalEnv, EraseProof.evalEnvOf, EraseProof.EvalEnv.casesOn?,
    EraseProof.EvalEnv.unfold?, EraseProof.casesMajor, EraseProof.casesArity,
    EraseProof.specialArity, EraseProof.quotEvalArity}
  \leanok
  \srcloc{proof/EraseProof/Spec.lean}{572}
  \uses{def:Erasure-EnvView, def:Erasure-casesInfo, def:Erasure-isRecursiveDecl,
    def:EraseProof-ConstsIn, def:Erasure-ErasureConfig}
  \lead{In short} What evaluation reads of the program: its declarations, the configuration's
  \code{@[extern]} remapping, and \hl{the shipping recognizer of canonical \code{casesOn}}.
  \begin{itemize}
  \item \code{evalEnvOf view cfg P}: the environment of \code{P} under the eraser's view;
  \item \code{unfold?}: $\delta$ of definitions only; not theorems, opaques or recognized
        \code{casesOn} (DV-24, DV-25);
  \item \code{casesOn?}: the shipping \code{casesInfo?} on the declarations;
        \code{casesMajor}, \code{casesArity}: positions in a \code{casesOn} spine;
  \item \code{specialArity}: the arity of a constructor, a recognized \code{casesOn}, or
        \code{Quot.mk} (3) and \code{Quot.lift} (6) by \code{quotEvalArity}.
  \end{itemize}
\end{definition}

\begin{definition}[Atoms]
  \label{def:EraseProof-EvalEnv-isAtom}
  \lean{EraseProof.EvalEnv.isAtom, EraseProof.EvidentZero,
    EraseProof.DeltaFree, EraseProof.EvidentProp, EraseProof.OwnRules, EraseProof.IsThm,
    EraseProof.SrcAtom, EraseProof.AtomSpine}
  \leanok
  \uses{def:Erasure-declExclusion, def:Erasure-Pure-instLevels, def:EraseProof-EvalEnv,
    def:EraseProof-ConstsIn}
  \srcloc{proof/EraseProof/Spec.lean}{693}
  \lead{In short} Constants that evaluate to themselves: \hl{without $\delta$, arity- or
  proof-typed, or a theorem}.
  \begin{itemize}
  \item \code{isAtom c}: no \code{unfold?}, not a recognized \code{casesOn}, no rules of its own
        (\code{OwnRules}), and its type an arity (the shipping \code{arityShape}), evidently a proposition
        (\code{EvidentProp}, \code{EvidentZero}), or a theorem (\code{IsThm});
  \item \code{SrcAtom}: $\lambda$, sorts, $\Pi$; \code{AtomSpine}: an atom constant applied.
  \end{itemize}
  \lead{Reference} \code{atom} of \code{PCUICWcbvEval.v}; inductive type formers are atoms.
\end{definition}

\begin{definition}[\texorpdfstring{$\eta$}{Eta}-expansion of spines]
  \label{def:EraseProof-EvalEnv-etaSpine}
  \lean{EraseProof.EvalEnv.etaSpine?, EraseProof.piInst, EraseProof.lamsOf,
    EraseProof.ShortSpine}
  \leanok
  \uses{def:Erasure-Pure-instLevels, def:EraseProof-EvalEnv, def:EraseProof-ConstsIn}
  \srcloc{proof/EraseProof/Spec.lean}{624}
  \lead{In short} \code{etaSpine? c us as}: a constructor, recognized \code{casesOn} or quotient
  spine with fewer arguments than its arity, \hl{as \code{fun xs => c as xs}} over the remaining
  $\Pi$-binders of its type, \code{as} unevaluated (DV-23).
  \begin{itemize}
  \item \code{piInst}: a type instantiated at arguments; \code{lamsOf}: the $\lambda$s;
  \item \code{ShortSpine}: such a spine, which \code{beta} and \code{appCong} skip.
  \end{itemize}
  \lead{Reference} the eraser's \code{withAppEtaToMinArity}; MetaRocq's \code{EEtaExpanded.v}.
\end{definition}

\begin{definition}[Source evaluation]
  \label{def:EraseProof-SrcEval}
  \lean{EraseProof.SrcEval, EraseProof.CtorValue, EraseProof.BlocksCong,
    EraseProof.PropProj, EraseProof.EvalEnv.propInd, EraseProof.EvalEnv.spineType?,
    EraseProof.casesSortLevel}
  \leanok
  \uses{def:Erasure-declExclusion, def:Erasure-Pure-instLevels, def:EraseProof-EvalEnv,
    def:EraseProof-EvalEnv-isAtom, def:EraseProof-EvalEnv-etaSpine, def:EraseProof-ConstsIn}
  \srcloc{proof/EraseProof/Spec.lean}{767}
  \lead{In short} \code{SrcEval $\sigma$ e v}: the closed Lean term \code{e} \hl{evaluates to the
  value \code{v}}; constructor and \code{casesOn} spines evaluate as a whole.
  \begin{tabular}{p{0.2\linewidth}p{0.66\linewidth}}
  \toprule
  Rule & Step \\
  \midrule
  \code{beta}, \code{zeta}, \code{delta} & PCUIC's, on spines that are not short \\
  \code{constAtom}, \code{atom}, \code{lit} & atoms, $\lambda$, sorts, $\Pi$, \code{Nat} literals
    are values \\
  \code{eta} & an under-applied spine evaluates to its $\eta$-expansion (DV-23) \\
  \code{ctor} & a saturated constructor evaluates its arguments (\code{eval\_construct}) \\
  \code{casesIota} & the major evaluates to a \code{CtorValue}; the chosen alternative applied to
    the fields (\code{eval\_iota}); \code{casesOn} is primitive (DV-25) \\
  \code{casesLitZero}, \code{casesLitSucc} & the same on a \code{Nat} literal major \\
  \code{casesProof} & sort evidently zero, one field-less constructor, the major an atom whose
    statement is that constructor's type: the alternative (DV-24) \\
  \code{proj}, \code{projProp} & a field of a constructor value; a \code{Prop} projection of a
    neutral proof is neutral (\code{PropProj}, DV-24) \\
  \code{quotMk} & \code{Quot.mk $\alpha$ r a} evaluates its element \code{a} only (DV-29) \\
  \code{quotLift} & \code{f} evaluates to \code{fun x => b}, \code{q} to \code{Quot.mk $\alpha$'
    r' a}: \hl{the quotient $\iota$}, \code{b} at \code{a} (DV-29) \\
  \code{appCong}, \code{mdata} & a stuck head (not \code{BlocksCong}: no constructor,
    \code{casesOn} or quotient head) applied; metadata dropped \\
  \bottomrule
  \end{tabular}
  \lead{Reference} PCUIC \code{eval}; $\lambda_\Box$'s \code{eval\_iota\_sing} and
  \code{eval\_proj\_prop} for the proof rules.
\end{definition}

\section{Restriction to the closure}\label{sec:source-restrict}

\begin{definition}[Closures and sub-environments]
  \label{def:EraseProof-ConstsIn}
  \lean{EraseProof.ConstsIn, EraseProof.KernameInj, EraseProof.SubEnv, EraseProof.findDecl}
  \leanok
  \uses{def:toKername}
  \srcloc{proof/EraseProof/Spec.lean}{499}
  \lead{In short} Predicates on declaration lists: \hl{a sublist of the program}, injective
  kernames.
  \begin{itemize}
  \item \code{ConstsIn p e}: every constant of \code{e}, and \hl{the structure of every
        projection}, satisfies \code{p};
  \item \code{findDecl decls c}: the first declaration named \code{c} (MetaRocq's
        \code{lookup\_env});
  \item \code{SubEnv decls P}: every declaration of \code{decls} is that of \code{P};
        \code{KernameInj}: \code{toKername} injective on \code{decls}.
  \end{itemize}
  \lead{Why the structure} \code{projProp} reads whether \code{S} is a \code{Prop} inductive:
  with \code{S} uncounted, \code{.proj S 0 h} evaluates in the program but not in its closure.
  The shipping \code{exprConsts} counts it too (S-30).
  \lead{Reference} MetaRocq \code{term\_global\_deps} on \code{tProj}.
\end{definition}

\begin{definition}[Dependency-closed closures]
  \label{def:EraseProof-DepClosed}
  \lean{EraseProof.DepClosed, EraseProof.FVarsIn}
  \leanok
  \uses{def:EraseProof-PhaseIDecl, def:EraseProof-ConstsIn}
  \srcloc{proof/EraseProof/Spec.lean}{895}
  \lead{In short} \code{DepClosed decls}: \hl{the closure holds what the traversal and the
  recognizer read} of it.
  \begin{itemize}
  \item the constants of every read expression (\code{ReadExpr});
  \item the members of every block, the constructors of an inductive, the inductive of a
        constructor, and the block of a recursor;
  \item \code{FVarsIn p e}: every free variable of \code{e} satisfies \code{p}.
  \end{itemize}
\end{definition}

\begin{lemma}[Evaluation in the closure]

  \lead{In short} Given \code{SubEnv decls P}, \code{DepClosed decls} and the constants of
  \code{e} in \code{decls}, an evaluation under \code{evalEnvOf view cfg P} is \hl{one under

\section*{Caveats}
\begin{itemize}
\item \lead{Caveat} Recursors have no evaluation rule: the fragment test keeps them off the
  evaluated path. Quotients evaluate, and the fragment test admits them.
\item \lead{Caveat} A computational axiom or an opaque is stuck: no value, no claim.
\item \lead{Caveat} A \code{casesOn} on a lemma of an inductive with proof fields has no rule:
  such programs are vacuous (DV-24).
\end{itemize}
