\section{Erasability}\label{sec:source-erasable}

\begin{definition}[Erasable terms]
  \label{def:EraseProof-IsErasable}
  \lean{EraseProof.IsErasable, EraseProof.IsArity, EraseProof.ErasableS}
  \leanok
  \uses{def:EraseProof-TrS}
  \srcloc{proof/EraseProof/Erasability.lean}{22}
  \lead{In short} A term is erasable when \hl{some type of it is an arity or a proposition}.
  \begin{itemize}
  \item \code{IsArity}: \code{Pi x1 ... xn, Sort u} in the model;
  \item \code{IsErasable venv U Gamma e}: some type \code{T} of \code{e} is an arity, or has a sort
        equivalent to level zero (no SProp, DV-4);
  \item \code{ErasableS venv Us Delta e}: the source term \code{e} has an erasable \code{TrS}
        image (DV-1).
  \end{itemize}
  \lead{Reference} MetaRocq \code{isErasable} (\code{erasure/theories/Extract.v}); MetaCoq paper
  Section 7.3; Letouzey Definition 1.
\end{definition}

  \item \code{EvidentProp}: \code{Pi xs, h.\{vs\} as} with \code{h} $\delta$-less, a syntactic arity
        of exactly as many binders as arguments, and its sort at the occurrence
        (\code{Pure.instLevel}) evidently zero.
  \end{itemize}
  \lead{Reference} \code{tInd} and \code{tConstruct} in PCUIC's \code{atom}; MetaRocq
  \code{isPropositionalArity} (\code{pcuic/theories/PCUICFirstorder.v}).
\end{definition}

\section{Source evaluation}\label{sec:source-eval}

\begin{definition}[Weak call-by-value evaluation]
  \label{def:EraseProof-SrcEval}
  \lean{EraseProof.SrcEval, EraseProof.SrcAtom, EraseProof.BlocksCong}
  \leanok
  \uses{def:EraseProof-EvalEnv, def:EraseProof-EvalEnv-isAtom, def:EraseProof-RecursiveDecl,
    def:Erasure-Pure-instLevels}
  \srcloc{proof/EraseProof/Source/Eval.lean}{45}
  \lead{In short} \code{SrcEval sigma e v}: the closed Lean term \code{e} evaluates to \code{v}, big
  step, call by value, \hl{with \code{let} kept} (DV-1, DV-22).

  \begin{tabular}{lp{0.64\linewidth}}
  \toprule
  Rule & Meaning \\
  \midrule
  \code{beta}, \code{zeta} & $\beta$ with a value argument; \code{let} with a value \\
  \code{delta} & a non-recursive unfoldable constant evaluates as its body, levels instantiated \\
  \code{fixAtom}, \code{fixApp} & a recursive constant is a value; applied, its body is unfolded
    (PCUIC \code{tFix} with recursive argument 0, DV-7) \\
  \code{constAtom} & an atom constant is a value (DV-11) \\
  \code{appCong} & an application whose function value is not a $\lambda$, sort, $\Pi$ or
    recursive constant (\code{BlocksCong}) \\
  \code{mdata}, \code{atom} & metadata is transparent; $\lambda$, sorts and $\Pi$
    (\code{SrcAtom}) are values \\
  \bottomrule
  \end{tabular}

  Axioms, non-atom opaques and remapped constants have no rule: they are stuck.
  \lead{Reference} PCUIC \code{eval} (\code{pcuic/theories/PCUICWcbvEval.v}); MetaCoq paper
  Section 5.6; Letouzey Definition 9; DV-1, DV-7, DV-11, DV-22 (Chapter~\ref{chap:divergences}).
\end{definition}

\begin{lemma}[Subject reduction for evaluation]
  \label{lem:EraseProof-SrcEval-defeq}
  \lean{EraseProof.SrcEval.defeq}
  \leanok
  \uses{def:EraseProof-SrcEval, def:EraseProof-ProgEnv, def:EraseProof-SubEnv}
  \srcloc{proof/EraseProof/Source/Defeq.lean}{20}
  \inherited{L1, L2, L3, L4, L5}
  \lead{In short} Evaluation of a translated, typed closed term is \hl{a typed definitional
  equality} of the model.
  \begin{itemize}
  \item \lead{Given} \code{henv : ProgEnv P venv}, \code{hsub : SubEnv sigma.decls P}: typing in
        \code{P}'s model, evaluation over a sub-environment;
  \item \code{he : TrS venv Us [] e e'}, \code{hT : e' : T} in the model;
        \code{hev : SrcEval sigma e v};
  \item \lead{Then} \code{v} translates to some \code{v'} with \code{e'} $\equiv$ \code{v'} at
        \code{T}.
  \end{itemize}
  \lead{Reference} \code{subject\_reduction\_eval}, \code{wcbveval\_red}
  (\code{pcuic/theories/PCUICClassification.v}); MetaCoq paper Section 5.4; Letouzey Lemma 2.
\end{lemma}
\begin{proof}
  \uses{lem:EraseProof-SrcEval-beta_step}
  \leanok
  \lead{Idea} Induction on \code{SrcEval}: the typed steps for \code{beta}, \code{zeta},
  \code{delta}, \code{fixApp}, \code{appCong}; reflexivity for \code{fixAtom}, \code{constAtom},
  \code{atom}; \code{mdata} by inversion of \code{TrS}.
\end{proof}

\begin{lemma}[Atoms are erasable]

  \srcloc{proof/EraseProof/Erasability/Eval.lean}{204}
  \inherited{L1, L2, L3, L4, L5}
  \lead{In short} A translated occurrence of an atom constant is erasable \hl{at every level
  instantiation}, in every context.
  \begin{itemize}
  \item \code{VArityShape T n w}: \code{T} is the model arity \code{Pi x1 ... xn, Sort w};
        \code{TrS.arityShape} translates a syntactic arity to one;
  \item \code{EvidentZero.sound}: an evidently zero level translates to a level equivalent to zero;
  \item \code{TrS.spine\_type}, \code{TrS.evidentProp\_tail}, \code{TrS.evidentProp\_sort}: the
        image of an evident proposition has a sort equivalent to zero.
  \end{itemize}
  \lead{Reference} none directly: \code{tInd} has no \code{erases} rule, and propositional

  \uses{lem:EraseProof-ProgEnv-lookup, lem:EraseProof-SubEnv-findDecl_of_some,

    lem:EraseProof-ProgEnv-ordered}
  \leanok
  \lead{Idea} \code{ProgEnv.lookup} gives the declared type at the constant's own level
  parameters: an arity, or of sort equivalent to zero; lean4lean's \code{HasType.instL} moves it to
  the use levels and weakens it; its \code{IsDefEq.uniq} types application spines.
\end{proof}

\section*{Caveats}
\begin{itemize}
\item \lead{Caveat} Opaques do not unfold (DV-10): an evaluation reaching a non-atom opaque is
  stuck, while the eraser emits the opaque's body.
\item \lead{Caveat} Evaluation is call by value (DV-22); evaluations in other orders are not
  covered.
\item \lead{Caveat} A recursive constant of any type is a value (DV-7): \code{unsafe def x : CN :=
  x} evaluates to itself, while Lean's compiled code loops on it.
\end{itemize}
