\begin{theorem}[Correctness of \texorpdfstring{\code{\#erase}}{\#erase}]
  \label{thm:EraseProof-erase_correct}
  \lean{EraseProof.erase_correct, EraseProof.erase_correct_of_pure}
  \leanok
  \uses{def:EraseProof-erase_correct_statement, def:EraseProof-erasePure_erases_statement}
  \srcloc{proof/EraseProof/Final.lean}{46}
  \lead{In short} If \code{\#erase} returns \code{p} for an in-fragment, typed \code{e} that
  evaluates to \code{v}, then \hl{\code{p}'s main term evaluates in $\lambda_\Box$ to an erasure
  of \code{v}}, and \code{v} means what \code{e} means.
  \begin{itemize}
  \item \lead{Given} \code{henv : ProgEnv P $\zeta$ E B}; \code{hwf : Metalean.EnvWF E};
        \code{hview : ViewAgrees view P};
  \item \code{he : TrM E B lps .nil [] e t}; \code{hty : $\exists$ T, E[.nil] $\vdash$ t : T};
  \item \code{hfrag : collectDeps view cfg e = .ok decls};
  \item \code{hrun : eraseEntry view cfg e = pure (p, inl)};
  \item \code{hev : SrcEval (evalEnvOf view cfg P) e v};
  \item \lead{Then} $\exists$ \code{lenv tb vb}, \code{p = .untyped lenv (some tb)} $\wedge$
        \code{Erases E B lps (evalEnvOf view cfg P) .nil [] v vb} $\wedge$
        \code{LBEval defaultFlags lenv tb vb} $\wedge$ $\forall$ \code{T}, \code{E[.nil] $\vdash$
        t : T} $\to$ $\exists$ \code{t'}, \code{TrM E B lps .nil [] v t'} $\wedge$
        \code{E[.nil] $\vdash$ t $\equiv$ t' : T};
  \item \code{erase\_correct\_of\_pure}: the statement follows from M2's statement;
        \code{erase\_correct} applies it to \code{erasePure\_erases}.
  \end{itemize}
  \lead{Reference} MetaRocq \code{erase\_correct}, at \code{default\_wcbv\_flags}, with
  \code{subject\_reduction\_eval}.
\end{theorem}
\begin{proof}
  \uses{def:EraseProof-PhaseIDecl, lem:EraseProof-collectDeps_sub, lem:EraseProof-collectDeps_spec,
    lem:EraseProof-eraseEntry_pure, def:EraseProof-ViewAgrees,
    def:EraseProof-erases_correct_statement, thm:EraseProof-erases_correct,
    thm:EraseProof-erasePure_erases, def:EraseProof-MBinding, def:EraseProof-TrM,
    def:EraseProof-ProgEnv, def:Erasure-EraseError, def:Erasure-EnvView, def:Erasure-collectDeps,
    def:Erasure-eraseEntry, def:EraseProof-LBEval, def:EraseProof-Erases,
    def:EraseProof-ErasesDeps, lem:EraseProof-Erases-extend, def:EraseProof-EvalEnv,
    def:EraseProof-SrcEval, lem:EraseProof-SrcEval-defeq, def:EraseProof-ConstsIn,
    def:EraseProof-DepClosed, def:EraseProof-ClosedDecls, lem:EraseProof-SrcEval-restrict,
    def:LBTerm, def:toKername, def:ASTType, def:Erasure-ErasureConfig}
  \leanok
  \lead{Idea} Composition, with no further bridge lemma.
  \lead{Steps}
  \begin{enumerate}
  \item \code{hfrag} puts the run on the pure path (\code{eraseEntry\_pure}); M2 makes its output
        an erasure at the closure \code{decls}, in a closed $\lambda_\Box$ environment;
  \item \code{decls} is a dependency-closed sub-environment of \code{P}, with injective names and
        arities (\code{collectDeps\_sub}, \code{collectDeps\_spec}); \code{hev} restricts to it;
  \item M1 simulates the evaluation at \code{decls}; \code{Erases.extend} moves the value's
        erasure to \code{P};
  \item \code{SrcEval.defeq} gives the last conjunct.
  \end{enumerate}
\end{proof}

\section{Reading the statement}\label{sec:final-reading}

\begin{itemize}
\item \lead{\code{henv}} the Lean declarations \code{P} are the Lean form of the Metalean
  environment \code{E}; \code{B} maps each Lean name to its Metalean counterpart. \code{P} is
  \hl{the program's dependency closure}, not a whole Lean environment: over all of \code{Init} the
  premises are unsatisfiable, since \code{Init} holds declarations Metalean cannot express
  (\code{unsafe} definitions, nested inductives such as \code{Lean.Syntax}).
\item \lead{\code{hwf}} \code{E} is well formed in Metalean's sense (MetaRocq's \code{wf\_ext}).
\item \lead{\code{hview}} what \code{\#erase} reads from the Lean environment agrees with \code{P}
  on the names of \code{P}: the view of the closure alone (the non-vacuity instance NV-1), or the
  view of the whole environment that \code{\#erase} builds.
\item \lead{\code{he}, \code{hty}} the input \code{e} is the Lean form of a closed, well-typed
  Metalean term \code{t}.
\item \lead{\code{hfrag}} the shipping fragment test accepts \code{e} and returns its closure
  \code{decls}; \code{\#erase} runs the same test before it takes the pure path. The only premise
  without a MetaRocq counterpart (DV-26), bounded below by \code{collectDeps\_ok\_of}.
\item \lead{\code{hrun}} the entry point behind \code{\#erase} returns the program \code{p}.
\item \lead{\code{hev}} \code{e} evaluates to \code{v} under the source semantics. The theorem is
  \hl{partial correctness}: Metalean proves neither canonicity nor normalization, and \code{SrcEval}
  is tied to Metalean only through definitional equality, so nothing shows that a typed program
  evaluates (DV-17).
\item \lead{Conclusion} \code{p} is an untyped $\lambda_\Box$ program whose main term \code{tb}
  evaluates (MetaRocq's \code{default\_wcbv\_flags}) to a value \code{vb} that is an erasure of
  \code{v}. The last conjunct is subject reduction: \hl{the source semantics is Metalean's
  conversion}, not an arbitrary relation.
\item \lead{Well-founded recursion} excluded: the fragment test rejects such definitions (they
  have an \code{\_unsafe\_rec} companion), and their recursion reduces only through an accessibility
  proof, which is opaque (DV-24), so \code{hev} never holds for them.
\item \lead{No recursion} the fragment is non-recursive: the test rejects structural recursion
  too (an \code{\_unsafe\_rec} companion, \code{Nat.add} and \code{Nat.beq} included) and
  recursors other than the companions of \code{casesOn}, so \hl{no \code{Nat} arithmetic}. Nat
  literals are in only under \code{nat := .peano}.
\item \lead{Scope note} \hl{\code{erase\_correct} says nothing} about programs that evaluate
  through a constant without $\delta$ in the source, although the fragment admits them and the
  emitted program may compute:
  \begin{itemize}
  \item opaques with a value: the eraser emits the value, the source does not unfold it (DV-10);
  \item axioms of data type, and \code{@[extern]} definitions under the default
        \code{extern := .preferAxiom}, which the source treats as axioms (DV-12);
  \item a \code{casesOn} on a lemma of an inductive with proof fields: \code{casesProof} covers
        only field-less constructors (DV-24).
  \end{itemize}
  Example: \code{Ex.opaqueEx} of \leanfile{tests/corpus/Examples.lean}, the successor of
  \code{opaque opaqueTwo : CNat := two}, is in the fragment; the source is stuck at
  \code{opaqueTwo}.
\item \lead{The object} the theorem is about the shipping code itself (\code{collectDeps},
  \code{eraseEntry}), not a model of it. \code{proof/spec.pin} hashes every shipping definition
  the statements reach, \hl{except the three functions under verification}, \code{collectDeps},
  \code{erasePure} and \code{eraseEntry}, which it names: a fix to them leaves the statement
  unchanged.
\end{itemize}

\section{Hypotheses}\label{sec:final-hypotheses}

Every hypothesis has a counterpart among those of MetaRocq's \code{erase\_correct}, except
\code{hfrag}.

\begin{tabular}{p{0.14\linewidth}p{0.26\linewidth}p{0.5\linewidth}}
\toprule
Hypothesis & MetaRocq & Why it is there \\
\midrule
\code{henv} & \code{Sigma} as a PCUIC environment & Lean's declarations are Metalean's
  environment, as its loader builds it \\
\code{hwf} & \code{wf\_ext Sigma} & Metalean's own well-formedness \\
\code{hview} & \code{Sigma} $\sim$\code{\_ext X} & the environment \code{\#erase} reads is \code{P}
  on \code{P}'s names \\
\code{he}, \code{hty} & \code{welltyped Sigma [] t} & typing lives on Metalean terms; \code{TrM}
  names the Lean term \code{\#erase} sees \\
\code{hfrag} & none & \hl{the router's own decidable test}; it names syntactic classes
  (Section~\ref{sec:scope-fragment}, DV-26) \\
\code{hrun} & \code{erase Sigma t = t'} & names the output of \code{\#erase} \\
\code{hev} & \code{Sigma |- t} $\Downarrow$ \code{v} & the proof is by induction on source
  evaluation \\
\bottomrule
\end{tabular}

\begin{itemize}
\item \lead{Against lean4lean's} the same shape, plus the subject-reduction conjunct and a
  fragment with inductive types, constructors, \code{casesOn}, projections and literals; the
  premise \code{hfrag} is explicit where lean4lean's in-scope condition followed from
  \code{henv}, \code{hview} and \code{he}.
\end{itemize}

\section*{Caveats}
\begin{itemize}
\item \lead{Caveat} Partial correctness: nothing is claimed when \code{\#erase} fails or the source
  does not evaluate.
\item \lead{Caveat} No first-order corollary: \code{vb} is an erasure of \code{v}, not pinned to
  its structural erasure; no claim about the constants to inline.
\item \lead{Caveat} No well-formedness of the emitted program beyond closedness; only
  \code{peregrine validate} checks it (C11).
\item \lead{Caveat} Programs defined by well-founded recursion are excluded; the theorem would be
  vacuous on them.
\item \lead{Caveat} Opaques with values, data axioms, \code{@[extern]} definitions and matches on
  lemmas with proof fields do not evaluate in the source: the theorem says nothing about them.
\item \lead{Caveat} \code{P} is a dependency closure: the theorem does not apply with \code{P} the
  whole environment of a Lean file.
\end{itemize}
