\section{Terms}\label{sec:model-terms}

Metalean's translator is \code{partial} and asks an unverified type inference for projections;
the bridge restates it rule by rule as a relation.

\begin{definition}[What a Lean name denotes]
  \label{def:EraseProof-MBinding}
  \lean{EraseProof.MBinding, EraseProof.Binds, EraseProof.MBinding.isSpecial,
    EraseProof.MBinding.wk, EraseProof.Binds.wk, EraseProof.Binds.set}
  \leanok
  \srcloc{proof/EraseProof/Spec.lean}{38}
  \lead{In short} \code{Binds $\zeta$ = Name $\to$ Option (MBinding $\zeta$)} maps a Lean name to
  \hl{the Metalean head it denotes}: Metalean's binding table with heads in place of positions.
  \begin{itemize}
  \item \code{const}: a constant (axiom, definition, opaque);
  \item \code{ind}, \code{ctor}, \code{recr}: a sort of an inductive block, one of its
        constructors (with the order of its fields in Metalean), its recursor;
  \item \code{quot}: one of the four quotient constants;
  \item \code{isSpecial}: the heads the translator saturates (all but \code{const}).
  \end{itemize}
  \lead{Reference} \leanfile{Metalean/Frontend/Table.lean}.
\end{definition}

\begin{definition}[Universe levels]
  \label{def:EraseProof-TrLevel}
  \lean{EraseProof.TrLevel, EraseProof.TrLevels}
  \leanok
  \srcloc{proof/EraseProof/Spec.lean}{77}
  \lead{In short} \code{TrLevel lps u}: the Lean level \code{u} over the parameters \code{lps}
  as \hl{a Metalean level}; level metavariables do not translate.
  \lead{Reference} \code{translateLevel}, \code{translateLevels}
  (\leanfile{Metalean/Frontend/Translate.lean}).
\end{definition}

\begin{definition}[Saturated heads]
  \label{def:EraseProof-SatTr}
  \lean{EraseProof.SatTr, EraseProof.SpecialHead, EraseProof.MetaOrderOK,
    EraseProof.RecrLevels, EraseProof.quotArity}
  \leanok
  \uses{def:EraseProof-MBinding, def:EraseProof-TrLevel}
  \srcloc{proof/EraseProof/Spec.lean}{219}
  \lead{In short} \code{SatTr E lps b us args t}: the head \code{b} at levels \code{us}, applied to
  the translated arguments \code{args}, is \hl{the saturated Metalean node or the $\eta$-term}
  \code{t}.
  \begin{itemize}
  \item saturated: \code{ind}, \code{ctor} (fields permuted to Metalean's order,
        \code{MetaOrderOK}: ordinary fields, then recursive ones), \code{recr}
        (\code{RecrLevels}: the motive's level split off), quotient constants (\code{quotArity});
  \item under-applied: the term of Metalean's builders \code{indTerm}, \code{ctorTerm},
        \code{recrTerm}, \code{quotTerm}, applied (DV-1; trusted,
        Section~\ref{sec:trust-trusted});
  \item \code{SpecialHead B f}: \code{f}'s head is a saturating name.
  \end{itemize}
  \lead{Reference} \code{saturate} and \code{translateSpecial} of
  \leanfile{Metalean/Frontend/Translate.lean}.
\end{definition}

\begin{definition}[Literals and proof projections]
  \label{def:EraseProof-natLitM}
  \lean{EraseProof.natLitM, EraseProof.natSig, EraseProof.natZeroSig, EraseProof.natSuccSig,
    EraseProof.projProofTerm}
  \leanok
  \srcloc{proof/EraseProof/Spec.lean}{155}
  \lead{In short} Two terms of Metalean's checker, copied as total definitions.
  \begin{itemize}
  \item \code{natLitM}: a \code{Nat} literal as \hl{its unary constructor chain}, as Metalean's
        fast checker denotes it;
  \item \code{projProofTerm}: Metalean's \code{proofProjection} written totally; the minor
        returns the field, the motive is the field's type with the earlier fields projected.
  \end{itemize}
  \lead{Reference} \leanfile{Metalean/Checker/Acceleration/Literal.lean},
  \leanfile{Metalean/Checker/Slow/Projection.lean}.
\end{definition}

\begin{definition}[Typing in an extension of the context]
  \label{def:EraseProof-TypedExt}
  \lean{EraseProof.TypedExt}
  \leanok
  \srcloc{proof/EraseProof/Spec.lean}{303}
  \lead{In short} \code{TypedExt E $\Gamma$ e T}: \code{e} has type \code{T} in \hl{a context
  \code{$\Gamma$'} into which $\Gamma$ embeds} by an injective renaming that preserves the types
  of the variables.
  \begin{itemize}
  \item \lead{Use} the typing premise of the projection rules of \code{TrM}: the major is typed
        where the Metalean term places it, possibly under the case telescope of a minor;
  \item \lead{Why} with the major typed in $\Gamma$ itself, re-typing it after a substitution
        inside a minor with permuted fields needs strengthening, which Metalean lacks;
  \item \lead{Weaker than typing in $\Gamma$} \code{$\Gamma$' = $\Gamma$} with the identity
        renaming gives it (DV-1).
  \end{itemize}
\end{definition}

\begin{definition}[The translation of terms]
  \label{def:EraseProof-TrM}
  \lean{EraseProof.TrM, EraseProof.MLDecl,
    EraseProof.MLDecl.denote, EraseProof.MLDecl.wk, EraseProof.MLCtx, EraseProof.MLCtx.pushLam,
    EraseProof.MLCtx.findB, EraseProof.MLCtx.findF}
  \leanok
  \uses{def:EraseProof-MBinding, def:EraseProof-TrLevel, def:EraseProof-SatTr,
    def:EraseProof-natLitM, def:EraseProof-TypedExt}
  \srcloc{proof/EraseProof/Spec.lean}{313}
  \lead{In short} \code{TrM E B lps $\Gamma$ $\Delta$ e t}: the Lean term \code{e} is \hl{the Lean
  form of the Metalean term \code{t}} in the context $\Gamma$; $\Delta$ (\code{MLCtx}) maps the
  Lean locals to Metalean variables or let values.
  \begin{tabular}{p{0.2\linewidth}p{0.66\linewidth}}
  \toprule
  Rule & Translation \\
  \midrule
  \code{bvar}, \code{fvar} & the local's entry in $\Delta$ \\
  \code{sort}, \code{const} & levels by \code{TrLevel}, \code{TrLevels} \\
  \code{special} & a spine headed by a saturating name: arguments one by one, then \code{SatTr} \\
  \code{app}, \code{lam}, \code{forallE} & one rule each; \code{app} only for other heads \\
  \code{letE} & the body with the local denoting the value, as the translator inlines it \\
  \code{mdata} & dropped \\
  \code{lit} & \code{natLitM}, when \code{B} binds \code{Nat.zero} and \code{Nat.succ} to the
    constructors \code{0} and \code{1} of \code{Nat} \\
  \code{projStruct} & Metalean's \code{IsStructure.projTerm}, parameters chosen by typing the
    major (\code{TypedExt}) \\
  \code{projProof} & \code{projProofTerm}, for a proof of a one-constructor, index-free,
    non-recursive inductive that is not a structure; the major typed by \code{TypedExt} \\
  \bottomrule
  \end{tabular}
  \lead{Why the literal's constructor bindings} Lean's kernel reads a literal through the names
  \code{Nat.zero} and \code{Nat.succ}, and so does \code{visitLiteral}. Without them, a \code{Nat}
  with constructors \code{Nat.z}, \code{Nat.s} beside another inductive's constructor named
  \code{Nat.zero} erases \code{0} to that other constructor.
  \lead{Reference} \code{translateWith} of \leanfile{Metalean/Frontend/Translate.lean};
  lean4lean's \code{TrExprS}.
\end{definition}

\begin{lemma}[Substitution under a telescope]
  \label{lem:EraseProof-Defeq-inst_tele}

  \srcloc{proof/EraseProof/Metatheory/SubstRen.lean}{162}
  \lead{In short} Given \code{E[$\Gamma$] $\vdash$ a : A}, a judgement in \code{$\Gamma$.snoc A ++
  $\Theta$} holds in \code{$\Gamma$ ++ $\Theta$[a]}, \hl{the binder types of $\Theta$ untyped}.
  \lead{Why} Metalean's public \code{Defeq.substitution} asks every variable to go to a typed
  term; its private \code{Defeq.substRen}, copied with attribution (Apache-2.0), also renames.
  \lead{Reference} \code{Defeq.substRen} (\leanfile{Metalean/Typing/Substitution.lean}); MetaRocq
  \code{substitution} (\code{PCUICSubstitution.v}).
\end{lemma}

\begin{definition}[Program environments]
  \label{def:EraseProof-ProgEnv}
  \lean{EraseProof.ProgEnv, EraseProof.ConstTyped, EraseProof.BindsExt,
    EraseProof.IndBlock, EraseProof.IndBlock.decls, EraseProof.quotKindOf, EraseProof.peelForalls}
  \leanok
  \uses{def:EraseProof-MBinding, def:EraseProof-SatTr, def:EraseProof-TrM, def:Erasure-casesInfo,
    def:EraseProof-ConstsIn}
  \srcloc{proof/EraseProof/Spec.lean}{497}
  \lead{In short} \code{ProgEnv P $\zeta$ E B}: the Lean declarations \code{P} (newest first) are
  \hl{those of the Metalean environment \code{E}}, named by \code{B}, as Metalean's loader builds
  it.
  \begin{itemize}
  \item every rule: the declaration's name is fresh and its universe parameters are
        distinct;
  \item \code{axiom}, \code{defn}: safe, their types (and values) translate; a definition is not
        recursive (\code{all = [name]});
  \item \code{opaque}: safe, its type and \hl{its value translate, the value typed at the
        type};
  \item \code{thm}: a Metalean opaque whose type is a proposition; \hl{its value does not mention
        the theorem's own name} (\code{ConstsIn});
  \item \code{induct}: a block (\code{IndBlock}): types, constructors with their field orders,
        recursors with the block's counts, \hl{each own minor premise with
        \code{nfields + 2 * nrecFields} binders} (\code{minors}); the type of every constant
        translates (\code{ConstTyped}); not nested, not unsafe;
  \item \code{IndBlock.ctorType}: under its \code{nparams + numFields} leading $\Pi$s
        (\code{peelForalls}) a constructor's Lean type is \code{I.\{lps\} params indices};
  \item \code{IndBlock.recrType}: under its \code{recrEnd s} leading $\Pi$s a recursor's Lean type
        is \code{motive\_s indices major};
  \item \code{IndBlock.indexVars}: \hl{a field that Metalean's block uses as an index variable is
        a Lean index field} (\code{indexFields});
  \item \code{quot}: the four constants, with \code{Eq} bound as Metalean requires.
  \end{itemize}
  \lead{Why \code{minors}} the recognizer reads a minor's arity from the syntax of the recursor's
  type, which \code{TrM} ties to Metalean only up to \code{mdata}: without the clause, a domain
  wrapped in \code{mdata} gets a bogus \code{casesOn} recognized.
  \lead{Why no self-reference} Metalean keeps no value for theorems: without the premise, a
  theorem \code{o := o} is loadable and takes the traversal's fixpoint branch; Lean's kernel never
  admits it.
  \lead{Why the opaque's value} Metalean keeps none, but the eraser emits it, and \code{ErasesDecl}
  relates it; a value mentioning a constant outside \code{P} is loadable otherwise. The kernel
  checks the value (\code{add\_opaque}), and the translation implies \code{ConstsIn}.
  \lead{Why distinct universe parameters} \code{TrLevel} maps a repeated name to its first
  position, so with \code{lps = [u, u]} the $\eta$-rule of a constructor would equate
  \code{Sort 1} and \code{Sort 0}; Lean's kernel rejects repeated universe parameters.
  \lead{Why the constructor type shape} \code{ConstTyped} ties a type to Metalean's only up to
  $\equiv$, which admits \code{I.mk : $\Pi$ (h : True), I True.intro} by proof irrelevance; the
  proof-irrelevant $\iota$ needs the parameters syntactically, as the kernel checks them
  (\code{is\_valid\_ind\_app}).
  \lead{Why \code{indexVars}} Lean's kernel decides large elimination by the field's syntactic
  occurrence among the result's arguments (\code{elim\_only\_at\_universe\_zero}), Metalean by
  \code{targetIndices}. Without it, \code{I : Nat $\to$ Prop} with \code{mk (n : Nat) : I (id n)}
  eliminates into every sort, and \code{visitCases} erases a match returning \code{n} to $\Box$.
  \lead{Why the recursor type shape} the oracle's type of a saturated recursor spine is then
  syntactically the motive's application, so its minors need no typing; the kernel generates
  exactly this type (\code{declare\_recursors}).
  \lead{Reference} \leanfile{Metalean/Checker/Slow/Loader.lean}; lean4lean's \code{TrEnv'}.
\end{definition}

\section{No-confusion of type formers}\label{sec:model-noconf}

Metalean states the injectivity of each type former but no disequality between two of them. The
lemmas follow from Metalean's domain model: a type equality equates the interpretations of both
sides, and each head contributes an atom that no other head's interpretation contains.

\begin{lemma}[Sort, \texorpdfstring{$\Pi$}{Pi} and inductive types differ]
  \label{lem:EraseProof-TypeEq-sort_ne_forallE}
  \lean{EraseProof.TypeEq.sort_ne_forallE, EraseProof.TypeEq.sort_ne_ind,
    EraseProof.TypeEq.forallE_ne_ind}
  \leanok
  \srcloc{proof/EraseProof/Metatheory/NoConf.lean}{25}
  \lead{In short} In a well-formed environment and context, \hl{a sort, a $\Pi$-type and an
  inductive type are pairwise unequal as types}.
  \begin{itemize}
  \item \lead{Given} \code{hE : EnvWF E} and \code{h$\Delta$ : E[$\Delta$] $\vdash$ ok};
  \item \lead{Then} \code{$\neg$ E[$\Delta$] $\vdash$ .sort l $\equiv$ .forallE t$_1$ t$_1$'},
        likewise for \code{.sort} against \code{.ind} and \code{.forallE} against \code{.ind}; the
        other orientations follow by symmetry.
  \end{itemize}
  \lead{Reference} MetaRocq \code{ws\_cumul\_pb\_Sort\_Prod\_inv},
  \code{invert\_cumul\_sort\_ind}, \code{invert\_cumul\_prod\_ind}
  (\code{pcuic/theories/PCUICConversion.v}).
\end{lemma}
\begin{proof}
  \leanok
  \lead{Idea} \leandecl{Metalean.TypeEq.rawInterpret_eq} equates the interpretations; one side
  holds its head's atom, which the other side's ideal does not contain.
  \lead{Steps}
  \begin{enumerate}
  \item invert the typing of each side (\code{Defeq.forallE\_inv}, \code{IndTyping.ofTyping});
  \item show the atom (\code{sortAtom}, \code{piAtom}) is a member of its own interpretation;
  \item rewrite with the equality and case on the order of shapes: only the bottom shape is below
        another head's atom.
  \end{enumerate}
\end{proof}

\begin{lemma}[Quotient types differ from the other type formers]
  \label{lem:EraseProof-TypeEq-sort_ne_quot}
  \lean{EraseProof.TypeEq.sort_ne_quot, EraseProof.TypeEq.forallE_ne_quot,
    EraseProof.TypeEq.ind_ne_quot}
  \leanok
  \srcloc{proof/EraseProof/Metatheory/NoConf.lean}{80}
  \lead{In short} In a well-formed environment and context, \hl{a quotient type equals no sort,
  $\Pi$-type or inductive type} as a type.
  \begin{itemize}
  \item \lead{Given} \code{hE : EnvWF E} and \code{h$\Delta$ : E[$\Delta$] $\vdash$ ok};
  \item \lead{Then} \code{$\neg$ E[$\Delta$] $\vdash$ X $\equiv$ .quot $\theta$ u $\alpha$ r} for
        \code{X} a sort, a $\Pi$-type or an inductive type.
  \end{itemize}
  \lead{Reference} none in MetaRocq: PCUIC has no primitive quotients.
\end{lemma}
\begin{proof}
  \leanok
  \lead{Idea} As for the other heads, with the atom \code{quotAtom} of the quotient's
  interpretation (\code{QuotTyping.ofTyping}).
\end{proof}

\section*{Caveats}
\begin{itemize}
\item \lead{Caveat} The bridge relates terms; it does not run Metalean's translator, whose
  \code{partial} functions and unverified type inference stay outside the proof.
\item \lead{Caveat} Recursor rules of Lean's declarations are not related: Metalean computes
  $\iota$ from the inductive block, and no lemma reads them.
\item \lead{Caveat} The quotient lemmas serve Phase II, where quotients are in scope.
\end{itemize}
