\section{The statement: the motive, the step interfaces, T5}
\label{sec:correctness:statement}

\code{Simulates} carries an environment accumulator for the induction; three \SEval{} rules are factored
into step interfaces, assembled into T5 below.

\begin{definition}[The induction motive]
\label{def:Simulates}
\lean{LeanToLambdaBox.Simulates}
\leanok
\uses{def:Erases, def:ErasesEnv, def:Lower, def:WcbvEval, def:eraseFlags, def:lean4lean-grounding}
\lead{In short} $\mathsf{Simulates}\ env\ bo\ Us\ \Gamma_{spec}\ \Gamma\ e\ v$: every composite \lbox{}
image of $e$ is matched by an image \hl{the emitted program reaches from the value}.
\begin{itemize}
\item \lead{Given} $e$ translates to $ve$, $\Erases{}\ e\ t_0$, $\Lower{}\ t_0\ t$ at $\Gamma_{spec}$,
$\ErasesEnv{}\ t_0$;
\item \lead{Then} $\exists v_0, v'$: $\Erases{}\ v\ v_0$, $\Lower{}\ v_0\ v'$,
$\WcbvEval{}\ t\ v'$ at \code{eraseFlags}, $\ErasesEnv{}\ v_0$.
\end{itemize}
\lead{Lean} \code{Simulates}, local context fixed at empty; the fourth conjunct is the accumulator the
beta, zeta, delta, iota arms need at a contractum.
\end{definition}

\begin{definition}[The three step interfaces]
\label{def:StepIota}
\lean{LeanToLambdaBox.StepIota, LeanToLambdaBox.StepProj, LeanToLambdaBox.StepDelta}
\leanok
\uses{def:Simulates, def:SEval, def:UpstreamAsks, def:LowerEnv, def:StepDefeq, def:CasesOnShape,
def:CtorOf, def:IndArity, def:InformativeInd, def:ConstOrigin}
\lead{In short} Each restates one \SEval{} rule as a hypothesis bundle proving \hl{\code{Simulates} at
the redex}, opening with \code{UpstreamAsks\ env}.
\begin{itemize}
\item \code{StepIota}: a \code{casesOn} spine at a constructor value, over-application $extra$ explicit;
\item \code{StepProj}: $\mathtt{proj}\ S\ i\ disc$;
\item \code{StepDelta}: a spine headed by a constant with a compiler body, unfolded at the call site's
universe levels, provided it is no \code{casesOn} name.
\end{itemize}
\lead{Lean} all three \code{abbrev}s, each inhabited by a theorem below (Theorems~\ref{thm:step_iota},
\ref{thm:step_proj}, \ref{thm:step_delta}).
\end{definition}

\begin{definition}[T5: the erasure-correctness statement]
\label{def:ErasesCorrectStmt}
\lean{LeanToLambdaBox.ErasesCorrectStmt}
\leanok
\uses{def:Simulates, def:Erases, def:ErasesEnv, def:Lower, def:LowerEnv, def:SEval, def:WcbvEval,
def:eraseFlags, def:UpstreamAsks, def:SEvalFlags, def:lean4lean-grounding}
\lead{In short} The proposition that \hl{source evaluation is simulated by the erased, lowered program}.
\begin{itemize}
\item \lead{Given} a well-formed kernel environment and \code{UpstreamAsks\ env};
\item a closed source term $e$ that translates ($\TrExprS$) and evaluates, $\SEval\ e\ v$, at source flags $fl$;
\item an erasure $\Erases\ e\ t_0$ and a lowering $\Lower\ \Gamma_{spec}\ t_0\ t$;
\item $\ErasesEnv$ at $(\Gamma_{spec}, t_0)$, and $\LowerEnv\ \Gamma_{spec}\ \Gamma$ for the emitted $\Gamma$;
\item \lead{Then} there are $v_0, v'$ with $\Erases\ v\ v_0$, $\Lower\ \Gamma_{spec}\ v_0\ v'$ and
      $\WcbvEval\ \Gamma\ t\ v'$ at \code{eraseFlags}.
\end{itemize}
\lead{Lean} \code{ErasesCorrectStmt}, an \code{abbrev}; the source flags stay a parameter, the target
flags are fixed. Eight binders carry seven premises: $\ErasesEnv$'s environment premise reads the tabled
body at the declaration's own level scope, not at the empty one.
\lead{Paper} [JACM]'s five premises plus $\LowerEnv$ and \code{UpstreamAsks}.
\end{definition}

\begin{definition}[T5, folded through the composite relation]
\label{def:ErasesCorrectLBStmt}
\lean{LeanToLambdaBox.ErasesCorrectLBStmt}
\leanok
\uses{def:ErasesCorrectStmt, def:ErasesLB, def:WcbvEval, def:LowerEnv, def:UpstreamAsks}
\lead{In short} The same simulation read through \hl{\code{ErasesLB}}, the composite relation from a
source term straight to its emitted \lbox{} term.
\begin{itemize}
\item \lead{Given} \code{ErasesLB} relates $v$ to $v'$ via some intermediate erasure, the environment
premise universally quantified over that middle term;
\item \lead{Then} $\exists v'.\ \code{ErasesLB}\ v\ v' \wedge \WcbvEval{}\ t\ v'$.
\end{itemize}
\lead{Lean} \code{ErasesCorrectLBStmt}, an abbreviation; this is the form Chapter~\ref{chap:refinement} consumes.
\end{definition}

\begin{remark}
\lead{Caveat} T5 drops $\ErasesEnv$ at the value $v_0$, although its inner induction (\code{Simulates})
proves it. A consumer that needs it must re-derive it.
\end{remark}

\section{Subject reduction for the source evaluator}
\label{sec:correctness:subject}

Locates the source evaluator's steps inside the kernel's definitional equality; closes with the theorem
every hard arm opens with.

\begin{lemma}[Spine parts translate]
\label{lem:trExprS_spine_head}
\lean{LeanToLambdaBox.trExprS_spine_head, LeanToLambdaBox.trExprS_spine_mem}
\leanok
\uses{def:lean4lean-grounding}
\lead{In short} If an application spine $\mkApps{}\ hd\ as$ translates, so does its head, and so does
\hl{each argument separately}.
\lead{Lean} \code{trExprS\_spine\_head} and \code{trExprS\_spine\_mem}; both leave the local context general.
\end{lemma}
\begin{proof}
\leanok
\uses{def:lean4lean-grounding}
\lead{Idea} Induction on the argument list, peeling one application node of the translation per step;
\code{trExprS\_spine\_mem} iterates it along each prefix.
\end{proof}

\begin{lemma}[Subject reduction along a spine, at an abstract step relation]
\label{lem:SEval-defeq_spine}
\lean{LeanToLambdaBox.SEval.defeq_spine, LeanToLambdaBox.SEval.defeq_spine_value}
\leanok
\uses{def:lean4lean-grounding, def:SEval}
\lead{In short} Any step relation keeping translations definitionally equal \hl{lifts to whole spines}.
\begin{itemize}
\item \lead{Given} $P$ a relation with $e$ translating to $ev$, $P\ e\ v$ $\Rightarrow$ $v$ translates to
some $vv$ definitionally equal to $ev$; heads definitionally equal; every argument $P$-related to its
value;
\item \lead{Then} the whole spine's translation is definitionally equal to the value spine's.
\end{itemize}
\lead{Lean} \code{SEval.defeq\_spine}; \code{SEval.defeq\_spine\_value} keeps a constant head fixed.
\end{lemma}
\begin{proof}
\leanok
\uses{lem:trExprS_spine_head, def:lean4lean-grounding, asm:lean4lean-trust}
\lead{Idea} Induction on the argument list via lean4lean's application congruence and uniqueness of
translations. Leaving $P$ abstract serves both value arms and the delta arm.
\end{proof}

\begin{theorem}[Subject reduction as definitional equality]
\label{thm:SEval-defeq}
\lean{LeanToLambdaBox.SEval.defeq}
\leanok
\uses{def:SEval, def:lean4lean-grounding, def:StepDefeq}
\lead{In short} If $e$ evaluates under \SEval{}, its value's translation is \hl{definitionally equal} to
$e$'s.
\begin{itemize}
\item \lead{Given} $e$ translates to $ve$ in the kernel environment;
\item $e$ evaluates to $v$ under \SEval{};
\item \lead{Then} $v$ translates to some $vv$ with $ve$ and $vv$ definitionally equal in $env$.
\end{itemize}
\lead{Lean} \code{SEval.defeq}; stated at a general local context, with no restriction on the source flags.
\lead{Status} \stProved{}, \stInherited{} through lean4lean.
\end{theorem}
\begin{proof}
\leanok
\uses{lem:SEval-defeq_spine, def:StepDefeq, lem:trExprS_spine_head, asm:lean4lean-trust}
\lead{Idea} Induction on the evaluation: beta by lean4lean's instantiation and beta lemmas plus
uniqueness of types; zeta likewise for \code{let}; sort, Pi reflexive; the two value arms by
Lemma~\ref{lem:SEval-defeq_spine}; delta, iota, projection are not kernel reductions of $env$, so each
rule instead carries its own \code{StepDefeq} premise, spent here.
\end{proof}

\section{Erasure along a spine, and the box arms}
\label{sec:correctness:box}

\begin{lemma}[Erasure is a spine congruence]
\label{lem:Erases-mkApps_forall2}
\lean{LeanToLambdaBox.Erases.mkApps_forall₂}
\leanok
\uses{def:Erases, def:LBTerm-mkApps}
\lead{In short} A spine erases \hl{pointwise}: head to head, arguments to arguments.
\begin{itemize}
\item \lead{Given} $hd$ erases to $th$; the arguments $as$ erase pointwise to $ts$;
\item \lead{Then} $\mkApps{}\ hd\ as$ erases to $\mathsf{mkApps}\ th\ ts$.
\end{itemize}
\lead{Lean} \code{Erases.mkApps\_forall}$_2$, whose pointwise premise is a \code{List.Forall}$_2$ witness.
\end{lemma}
\begin{proof}
\leanok
\uses{def:Erases}
\lead{Idea} Induction on the pointwise witness, applying \Erases{}'s application rule once per argument.
\end{proof}

\begin{lemma}[Inversion of erasure along a spine]
\label{lem:erases_mkApps_inv}
\lean{LeanToLambdaBox.erases_mkApps_inv}
\leanok
\uses{def:Erases, def:Erasable, def:lean4lean-grounding, def:LBTerm-mkApps}
\lead{In short} A spine's erasure is either \hl{structural} or has an irrelevant prefix.
\begin{itemize}
\item \lead{Given} $\mkApps{}\ hd\ as$ erases to $t$;
\item \lead{Then} (a) structurally: $hd\to th$, $as\to ts$ pointwise, $t = \mathsf{mkApps}\ th\ ts$; or
(b) a prefix $pre$ of $as$ (head included) is irrelevant: $\mkApps{}\ hd\ pre$ translates to an
\Erasable{} term, the rest erases pointwise to $ts$, $t = \mathsf{mkApps}\ \Box\ ts$.
\end{itemize}
\end{lemma}
\begin{proof}
\leanok
\uses{def:Erases, lem:Erases-sort_inv}
\lead{Idea} Induction on the argument list; \Erases{}'s inversion gives the box rule or the congruence at
each node.
\end{proof}

\begin{lemma}[An irrelevant head makes the spine irrelevant]
\label{lem:erasable_mkApps}
\lean{LeanToLambdaBox.erasable_mkApps}
\leanok
\uses{def:Erasable, def:lean4lean-grounding}
\lead{In short} An \Erasable{} head makes the \hl{whole applied spine} \Erasable{}.
\begin{itemize}
\item \lead{Given} a well-formed environment and local context; $\mkApps{}\ hd\ as$ translates to $ve$;
the head translates to $hve$, which is \Erasable{};
\item \lead{Then} $ve$ is \Erasable{}.
\end{itemize}
\end{lemma}
\begin{proof}
\leanok
\uses{lem:trExprS_spine_head, thm:Erasable-app, asm:lean4lean-trust}
\lead{Idea} \Erasable{}'s application rule iterated along the spine, uniqueness of translations
reconciling the model terms.
\end{proof}

\begin{lemma}[The box arm, at the erasure]
\label{lem:erases_correct_box}
\lean{LeanToLambdaBox.erases_correct_box, LeanToLambdaBox.erases_correct_boxSpine}
\leanok
\uses{def:Erases, def:Erasable, def:WcbvEval, def:eraseFlags, def:SEval}
\lead{In short} An \Erasable{} term always erases to $\Box$, and \hl{$\Box$ evaluates to itself}.
\begin{itemize}
\item \lead{Given} $e$ translates, some translation is \Erasable{}, and $e$ evaluates to $v$;
\item \lead{Then} $v$ erases to $\Box$, and $\Box$ evaluates to an erasure of $v$.
\end{itemize}
\lead{Lean} \code{erases\_correct\_boxSpine} is the spine form, folding $\Box\,t_1\dots t_n$ to $\Box$
by $\Box\,x \to \Box$ when each discarded argument evaluates.
\end{lemma}
\begin{proof}
\leanok
\uses{thm:SEval-defeq, lem:erasable_mkApps, lem:WcbvEval-mkApps_box, def:Erases, asm:lean4lean-trust}
\lead{Idea} Theorem~\ref{thm:SEval-defeq} moves irrelevance to the value, so \Erases{}'s box rule
applies at $v$; $\Box$ evaluates to itself, and the folding lemma collapses $\Box$ applied to evaluating
arguments.
\end{proof}

\begin{lemma}[The box arm, at the composite]
\label{lem:erases_correct_boxLow}
\lean{LeanToLambdaBox.erases_correct_boxLow, LeanToLambdaBox.erases_correct_boxSpineLow}
\leanok
\uses{def:Lower, def:ErasesEnv, def:ErasesBox, def:Erases, def:WcbvEval, def:eraseFlags}
\lead{In short} The form every arm needs: \hl{the box arm on the composite}, erasure then lowering.
\begin{itemize}
\item \lead{Given} $e$'s erasure is $\Box$ (\code{ErasesBox}), $\Box$ lowers to $t$, $\ErasesEnv{}\ \Box$;
\item \lead{Then} for $e$'s value $v$: $v_0 = \Box$, $v' = \Box$ satisfy \Erases{}, \Lower{},
$\WcbvEval{}\ t\ \Box$, $\ErasesEnv{}\ v_0$.
\end{itemize}
\lead{Lean} \code{erases\_correct\_boxSpineLow} takes middle term $\mathsf{mkApps}\ \Box\ ts$, each
lowered argument evaluating.
\end{lemma}
\begin{proof}
\leanok
\uses{lem:erases_correct_box, lem:ErasesEnv-ofReach, prop:Lower-source_const, thm:eval_deterministic,
lem:Lower-source_app, lem:WcbvEval-mkApps_box, asm:lean4lean-trust}
\lead{Idea} $\Box$ lowers only to $\Box$; Lemma~\ref{lem:erases_correct_box} supplies an erasure of the
value, forced to $\Box$ by determinism; \ErasesEnv{}'s box closure re-establishes the accumulator. The
spine form inverts the pass at the $\Box$-headed spine and folds by the boxed-application lemma.
\end{proof}

\section{Inverting the pass at an application}
\label{sec:correctness:lowerinv}

\begin{lemma}[The two readings of a lowered application]
\label{lem:Lower-source_app}
\lean{LeanToLambdaBox.Lower.source_app, LeanToLambdaBox.Lower.source_mkApps}
\leanok
\uses{def:Lower, def:ElimDecl, def:LBTerm-mkApps}
\lead{In short} A lowered application is either a \hl{congruence} or a saturated eliminator spine the
pass turned into a \code{case} node.
\begin{itemize}
\item \lead{Given} $\Lower{}\ (\mathtt{app}\ f\ a)\ t$;
\item \lead{Then} (a) $t = \mathtt{app}\ f'\ a'$, $f\to f'$, $a\to a'$; or (b) a saturated eliminator
spine: runtime key $kn$ with \code{ElimDecl} data $(iid, np, dp, nfs)$, parameters of length $dp$, a
discriminant and one minor premise per constructor, $\mathtt{app}\ f\ a =
\mathsf{mkApps}\ (\mathtt{const}\ kn)\ (pre \mathbin{+\!\!+} disc :: minors)$.
\end{itemize}
\lead{Lean} \code{Lower.source\_mkApps} covers a non-application, non-constant head: congruence only.
\end{lemma}
\begin{proof}
\leanok
\uses{lem:Lower-ne_fix_of_block, def:Lower}
\lead{Idea} Case analysis on \Lower{}: an over-applied eliminator node re-associates into the congruence
reading; the three block arms (\code{fixConst}, \code{fixBody}, \code{fixEta}) are excluded (an
application is neither lambda nor constant).
\end{proof}

\begin{definition}[Beta-readiness of a target head]

\lean{LeanToLambdaBox.BetaReady}
\leanok
\uses{def:WcbvEval}
\lead{In short} $\mathsf{BetaReady}\ \Gamma\ fl\ X\ c$: the target reads $X$ as a function that
\hl{beta-steps into $c$}.
\begin{itemize}
\item whenever a function position evaluates to $X$, an argument to $av$, and $c[av]$ to $r$, the
application evaluates to $r$.
\end{itemize}
\lead{Lean} \code{BetaReady}, stated at arbitrary target flags.
\end{definition}

\begin{lemma}[The beta transport across the pass]
\label{lem:Lower-appReady}
\lean{LeanToLambdaBox.Lower.appReady}
\leanok
\uses{def:BetaReady, def:Lower, def:ElimDecl}
\lead{In short} At a well-behaved specification environment, a lambda's image is \hl{beta-ready} at the

\item \lead{Given} closed bodies, guarded fix; $\Lower{}\ (\mathtt{lambda}\ nm\ b_0)\ t$;
\item \lead{Then} $\exists c$: $\Lower{}\ b_0\ c$, $\mathsf{BetaReady}\ \Gamma\ fl\ t\ c$.
\end{itemize}
\lead{Lean} \code{Lower.appReady} is one half of \code{Lower.appReady\_aux}'s joint induction; the other

\leanok
\uses{def:LowerBlock, thm:Lower-constToFix, prop:LowerBlock-substList_fixSubst,

body unfolds the guarded fixpoint at index $0$ to a lambda, itself beta-ready; its $\eta$-wrapper
(Definition~\ref{def:LBTerm-etaFix}) is beta-ready one beta step ahead of the same fact; every other arm
is refuted on shape.
\end{proof}

\section{The environment accumulator and the emitted constructor arity}
\label{sec:correctness:env}

\begin{lemma}[The environment relation is antitone in the program]
\label{lem:ErasesEnv-subterm}
\lean{LeanToLambdaBox.ErasesEnv.subterm}
\leanok
\uses{def:ErasesEnv, def:SubTerm}
\lead{In short} \ErasesEnv{} \hl{passes to subterms}.
\begin{itemize}
\item \lead{Given} \ErasesEnv{} holds of $t$, and $d$ is a subterm of $t$;
\item \lead{Then} \ErasesEnv{} holds of $d$.
\end{itemize}
\end{lemma}
\begin{proof}
\leanok
\uses{lem:ReachableFrom-through_body, def:ErasesEnv}
\lead{Idea} Each clause asks only about kernames reachable from the term; a subterm reaches no more. The
most-used lemma of the chapter.
\end{proof}

\begin{lemma}[The closure kit for the environment relation]
\label{lem:ErasesEnv-ofReach}
\lean{LeanToLambdaBox.ErasesEnv.ofReach, LeanToLambdaBox.ErasesEnv.box,
LeanToLambdaBox.ErasesEnv.substPair}
\leanok
\uses{def:ErasesEnv, def:ReachableFrom}
\lead{In short} \ErasesEnv{} transports along \hl{any reachability-shrinking map}.
\begin{itemize}
\item \code{ErasesEnv.ofReach}: transports along any map that does not increase reachable kernames;
\item \code{ErasesEnv.box}: $\Box$ reaches nothing, so it always satisfies the relation;
\item \code{ErasesEnv.substPair}: the contractum $b[s]$ of a beta or zeta step names no kername its two
parts do not.
\end{itemize}
\end{lemma}
\begin{proof}
\leanok
\uses{lem:ErasesEnv-subterm, lem:ReachableFrom-through_body}
\lead{Idea} All three reduce to the reachability calculus ($\Box$ has none; a substitution's reachable
set is the union of its two parts').
\end{proof}

\begin{lemma}[The environment relation at a spine]
\label{lem:ErasesEnv-mkApps}
\lean{LeanToLambdaBox.ErasesEnv.mkApps}
\leanok
\uses{def:ErasesEnv, def:LBTerm-mkApps, def:ReachableFrom}
\lead{In short} A spine is answered by \hl{whatever answers its head and its arguments}.
\begin{itemize}
\item \lead{Given} \ErasesEnv{} holds of $f$ and of every member of $l$;
\item \lead{Then} it holds of $\mathsf{mkApps}\ f\ l$.
\end{itemize}
\end{lemma}
\begin{proof}
\leanok
\uses{lem:ErasesEnv-ofReach, lem:ReachableFrom-through_body, def:ErasesEnv}
\lead{Idea} Re-establish each clause at the spine, splitting the reachability witness through the
application spine.
\end{proof}

\begin{lemma}[The emitted constructor arity of a reached block]
\label{lem:ErasesEnv-ctorArity}
\lean{LeanToLambdaBox.ErasesEnv.ctorArity}
\leanok
\uses{def:ErasesEnv, def:LowerEnv, def:IndInfo, def:constructorArity, def:ReachableFrom}
\lead{In short} A reached block's emitted constructor arity is \hl{$np + nfs_k$}.
\begin{itemize}
\item \lead{Given} \ErasesEnv{} holds of $t$; \LowerEnv{} prunes $\Gamma_{spec}$ to $\Gamma$; the source
block data is $\mathsf{IndInfo}\ env\ I\ iid\ np\ nfs$; the program reaches the block kername of $iid$;
$k < |nfs|$;
\item \lead{Then} the emitted environment reports constructor arity $np + nfs_k$ for constructor $k$ of

\end{itemize}
\end{lemma}
\begin{proof}
\leanok
\uses{def:ErasesEnv, def:LowerEnv, def:ElimDecl}
\lead{Idea} Reachability triggers the \code{blocks} clause, holding the block's emitted declaration;
\LowerEnv{} carries it over unchanged; unfolding the arity function gives $np + nfs_k$.
\end{proof}

\begin{lemma}[A constructor spine within its arity evaluates componentwise]
\label{lem:wcbvEval_mkApps_construct}
\lean{LeanToLambdaBox.wcbvEval_mkApps_construct}
\leanok
\uses{def:WcbvEval, def:eraseFlags, def:constructorArity, def:LBTerm-mkApps}
\lead{In short} An under-arity constructor spine evaluates \hl{argument by argument}.
\begin{itemize}
\item \lead{Given} constructor $k$ of $iid$ has emitted arity $ar$ in $\Gamma$; $ts$ is no longer than
$ar$; each $ts_i$ evaluates to $vs_i$;
\item \lead{Then} $\mathsf{mkApps}\ (\mathtt{construct}\ iid\ k\ [\,])\ ts$ evaluates to
$\mathsf{mkApps}\ (\mathtt{construct}\ iid\ k\ [\,])\ vs$.
\end{itemize}
\lead{Lean} \code{wcbvEval\_mkApps\_construct}, at \code{eraseFlags}.
\end{lemma}
\begin{proof}
\leanok
\uses{def:WcbvEval, def:eraseFlags}
\lead{Idea} Strong induction on the spine length, peeling the last argument: the atomic constructor rule
starts the run, the applied constructor rule accumulates one evaluated argument at a time.
\end{proof}

\begin{remark}
\lead{Deviation} The \emph{applied-form} constructor regime (\code{with\_constructor\_as\_block} false),
the Lean frontend's convention; differs from the block form CertiRocq and agda2lambox use
(Chapter~\ref{chap:lambdabox}).
\end{remark}

\begin{lemma}[A constructor's index is in range]
\label{lem:CtorOf-lt_nfs}
\lean{LeanToLambdaBox.CtorOf.lt_nfs}
\leanok
\uses{def:UpstreamAsks, def:CtorOf, def:IndInfo}
\lead{In short} Under \code{UpstreamAsks\ env}, a constructor's index is \hl{below its block's
field-count list length}.
\begin{itemize}
\item \lead{Given} $c$ is constructor $k$ of $I$; $I$'s block data is $(iid, np, nfs)$;
\item \lead{Then} $k < |nfs|$.
\end{itemize}
\end{lemma}
\begin{proof}
\leanok
\uses{def:UpstreamAsks, lem:IndInfo-inj}
\lead{Idea} The upstream declaration-level uniqueness identifies the field-count list with the member's

\end{proof}

\section{An under-applied eliminator spine erases nothing that has a value}

\section{The fix-closing convention}
\label{sec:correctness:closefix}

A short module pins the de Bruijn convention shared by the eraser's block-closing loop and \lbox{}'s
\code{fix} node, with no source-side notion.

\begin{definition}[The n-way abstraction of a block's fresh variables]
\label{def:closeFix}
\lean{LeanToLambdaBox.closeFix, LeanToLambdaBox.closeFixFold}
\leanok
\uses{def:toBvar, def:LBTerm}
\lead{In short} Folds single-variable abstraction over a block's fresh variables, \hl{last sibling
first}.
\begin{itemize}
\item \code{closeFixFold}: folds \code{toBvar} over (free variable, level) instructions, head first;
\item $\code{closeFix}\ \mathit{ids}\ \mathit{base}\ t$: runs it on $\mathit{ids}$ reversed, indexed from
$\mathit{base}$ --- for $\mathit{ids} = [x_0,\dots,x_{m-1}]$, $x_{m-1}$ becomes
$\mathtt{bvar}\ \mathit{base}$ and $x_0$ becomes $\mathtt{bvar}\ (\mathit{base}+m-1)$, matching the
\lbox{} \code{fix} node's convention (sibling $k$ as $\mathtt{bvar}\ (m-1-k)$).
\end{itemize}
\end{definition}

\begin{lemma}[The metatheory's recursion is the shipping loop]
\label{lem:closeFixFold_eq_foldl}
\lean{LeanToLambdaBox.closeFixFold_eq_foldl}
\leanok
\uses{def:closeFix, def:toBvar}
\lead{In short} \code{closeFixFold} \hl{equals a left fold} of \code{toBvar}.
\begin{itemize}
\item \lead{Then} $\code{closeFixFold}\ \mathit{pairs}\ t$ equals the left fold of \code{toBvar} over
$\mathit{pairs}$ starting from $t$.
\end{itemize}
\end{lemma}
\begin{proof}
\leanok
\uses{def:closeFix}
\lead{Idea} Induction on the instruction list, generalising the accumulator.
\end{proof}

\begin{remark}
\lead{Caveat} Bridges the shipping loop to the metatheory's recursion. Not proved here: the
\code{fixvars} lookup, that the variable recorded for member $k$ is the one the closing instruction
names (Chapters~\ref{chap:run}, \ref{chap:lowering}).
\end{remark}

\begin{lemma}[Per-sibling abstraction, occurrence reflection, and a worked two-member block]
\label{lem:closeFixFold_fvar_head}
\lean{LeanToLambdaBox.closeFixFold_fvar_head, LeanToLambdaBox.hasFVar_closeFix_of,
LeanToLambdaBox.closeFix_not_hasFVar, LeanToLambdaBox.closeFix_2block_last,
LeanToLambdaBox.closeFix_2block_first}
\leanok
\uses{def:closeFix, def:LBTerm}
\lead{In short} Closing \hl{removes exactly the block identifiers} and introduces none.
\begin{itemize}
\item closing $x$ first gives $\mathtt{bvar}\ \mathit{lvl}$ regardless of the rest (a \code{bvar} is inert
to later abstraction);
\item a post-closing occurrence pre-existed and is unabstracted; a pre-closing non-occurrence, or an
abstracted variable, does not occur after;
\item two-member block: $x_0\,x_1$ closes to $\mathtt{bvar}\ 1\ \mathtt{bvar}\ 0$, the \code{fix} node's
convention.
\end{itemize}
\end{lemma}
\begin{proof}
\leanok
\uses{lem:closeFixFold_eq_foldl, def:closeFix}
\lead{Idea} Induction, using that \code{toBvar} is the identity where its variable does not occur and
introduces no free variable; the two-member statements are instances computed by unfolding.
\end{proof}

\begin{definition}[The eta-wrapper of a block's node]
\label{def:LBTerm-etaFix}
\lean{LeanToLambdaBox.LBTerm.etaFix}
\leanok
\uses{def:LBTerm}
\lead{In short} $\code{etaFix}\ \mathit{defs}\ j$: a \hl{one-binder eta-expansion} of member $j$'s
\code{fix} node, applied to its own bound variable.
\begin{itemize}
\item $\code{etaFix}\ \mathit{defs}\ j = \mathtt{lambda}\ \mathtt{anon}\ ((\mathtt{fix}\ \mathit{defs}\
j)\ (\mathtt{bvar}\ 0))$;
\item the shipping eraser's \code{Erasure.etaExpandFix} registers this as a member's body rather than the
bare \code{fix} node (F-ETA); every block \Lower{} lowers has $\mathit{principalArgIdx} = 0$, so one
binder is the only shape reached.
\end{itemize}
\lead{Lean} MetaRocq's \code{eta\_fixpoint} at $1+\mathit{rarg}=1$; the general $\mathit{rarg}+1$ form is
not stated, since no reached block has another shape.
\end{definition}

\section{The first-order fragment and the observation}
\label{sec:correctness:firstorder}

T5's relation is non-deterministic; following [JACM, Sect.~7.3], it collapses on first-order values --- the
observation the capstone reports.

\begin{definition}[A first-order block]
\label{def:FirstOrderDecl}
\lean{LeanToLambdaBox.FirstOrderDecl, LeanToLambdaBox.FOType}
\leanok
\uses{def:lean4lean-grounding}
\lead{In short} A block whose fields are \hl{bare type formers only}.
\begin{itemize}
\item \code{FOType}: a field is a \textbf{bare} $\mathtt{const}\ I$, $I$ among the block's names or
accepted by closure $fo$ (never a bound variable);
\item \code{mono} (no universe parameters), \code{informative} (result sort a successor),
\code{noIndices} (arity = parameters), \code{fields} (every constructor binder first-order).
\end{itemize}
\end{definition}

\begin{remark}
\lead{Deviation} \code{informative}/\code{fields} descend from [L, Def.~6, 14]; \code{mono},
\code{noIndices} are this development's own restrictions. MetaRocq's \code{firstorder\_ind} is false on
\code{nat}, a defect in the cited artifact rather than a gap here, so this development cites it rather
than transcribing it; the extra \code{mono} and \code{noIndices} restrictions are disclosed but never
argued necessary.
\end{remark}

\begin{definition}[First-order inductive types]
\label{def:FirstOrderInd}
\lean{LeanToLambdaBox.FirstOrderInd, LeanToLambdaBox.FOClosed, LeanToLambdaBox.HasInduct}
\leanok
\uses{def:FirstOrderDecl, def:lean4lean-grounding}
\lead{In short} $I$ is first-order when some \hl{closed set of names} contains it.
\begin{itemize}
\item $fo$ is closed when every name it accepts is a declared block, all first-order w.r.t.\ $fo$ itself;
\item $I$ is first-order in $env$ when some closed $fo$ contains $I$.
\end{itemize}
\lead{Lean} \code{FirstOrderInd}; the closure sits inside the definition, else the constantly-true set
would admit any type former.
\end{definition}

\begin{lemma}[A first-order type former is declared and informative]
\label{lem:FirstOrderInd-indDeclOf}
\lean{LeanToLambdaBox.FirstOrderInd.indDeclOf, LeanToLambdaBox.FirstOrderInd.informativeInd}
\leanok
\uses{def:FirstOrderInd, def:IndDeclOf, def:InformativeInd}
\lead{In short} A first-order $I$ is \hl{declared and informative}.
\begin{itemize}
\item \lead{Given} $I$ is first-order in $env$;
\item \lead{Then} $I$ belongs to a declared block and is informative.
\end{itemize}
\end{lemma}
\begin{proof}
\leanok
\uses{lem:InformativeInd-mono, lem:IndInfo-constant_isArity, def:FirstOrderDecl}
\lead{Idea} Unfold the closure for declaredness; informativity follows from \code{informative}'s
successor shape.
\end{proof}

\begin{lemma}[A first-order spine is neither a sort nor a Pi]
\label{lem:FirstOrderInd-notSortNotPi}
\lean{LeanToLambdaBox.FirstOrderInd.notSortNotPi}
\leanok
\uses{def:FirstOrderInd, def:UpstreamAsks, def:lean4lean-grounding}
\lead{In short} A typeable first-order spine is \hl{neither a sort nor a Pi}, up to definitional
equality.
\begin{itemize}
\item \lead{Given} \code{UpstreamAsks\ env}; a well-formed context; $I$ first-order;
$\mathsf{mkApps}\ (\mathtt{const}\ I\ ius)\ iargs$ typeable;
\item \lead{Then} that spine is definitionally equal to neither.
\end{itemize}
\end{lemma}
\begin{proof}
\leanok

\begin{remark}
\lead{Caveat} Excludes lambda and type values from the induction; rests on \code{UpstreamAsks}, so is a

\end{remark}

\code{Type}-valued, parameter- and index-free, with one \code{Prop}-eliminating recursor.
\end{itemize}
\end{lemma}
\begin{proof}
\leanok
\uses{def:FirstOrderInd, def:FirstOrderDecl}
\lead{Idea} \code{decl\_wf} discharges each clause, vacuously (no constructors);
\code{firstOrderInd\_E} exhibits the closure accepting exactly $E$.
\end{proof}

\begin{remark}
\lead{Caveat} The \emph{only} proved instance of \code{FirstOrderInd} in the tree, at an inductive with
no constructors: the substantive branch of Theorem~\ref{thm:firstorder_erases_core} --- the constructor

\emph{assume} $\code{FirstOrderInd}\ env\ \code{Nat}$, discharged nowhere.
\end{remark}
