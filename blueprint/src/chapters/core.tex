  \hl{kername-injective, from the view, without duplicates, dependency-closed and closed}, and

\begin{lemma}[No recursive declaration]
  \label{lem:EraseProof-isRecursiveDecl_false}

    EraseProof.NonRec.TrM.consts, EraseProof.NonRec.nameOccurs_false,
    EraseProof.NonRec.ConstsIn_mono, EraseProof.NonRec.forall_mem_of_rel}

  \srcloc{proof/EraseProof/Core/NonRec.lean}{120}
  \lead{In short} For a declaration of \code{P} with a value, \code{isRecursiveDecl} is
  \code{false}: \hl{the fixpoint branch of \code{visitMutual} is dead on the pure path}.
  \begin{itemize}
  \item \lead{Given} \code{ProgEnv P $\zeta$ E B}, \code{findDecl P c = some ci}, a value of
        \code{ci};
  \item \lead{Then} \code{isRecursiveDecl ci = false}.
  \end{itemize}
  \lead{Reference} none (DV-7 does not arise).
\end{lemma}
\begin{proof}
  \uses{def:EraseProof-TrLevel, def:EraseProof-SatTr, def:EraseProof-natLitM,

  theorems and opaques.
\end{proof}
