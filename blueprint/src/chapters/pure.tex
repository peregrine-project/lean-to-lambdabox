\chapter{The pure path}\label{chap:pure}

\section*{At a glance}
\begin{itemize}
\item \lead{Goal} The shipping code the final theorem is about: the fragment test with the
  dependency closure, the \code{casesOn} recognizer, a pure erasability oracle, a pure backend for
  the shipping traversal, and the entry point that routes in-fragment inputs to it.
\item \lead{Main results} None: this chapter holds shipping definitions.
\item \lead{Status} \stShipping{} every node (shipping entries S-18 to S-21 and S-25 to S-37,
  Chapter~\ref{chap:shipping-changes}); \code{\#erase} runs \code{eraseEntry}, and the statements
  of the final theorem and its lemmas name these declarations.
\item \lead{Nodes} \bpnodes{chap:pure}
\item \lead{Lean files} \leanfiles{LeanToLambdaBox/Erasure/}{Collect, Pure, Entry, ExprOps}.
\item \lead{Reference} MetaRocq's \code{erase}, \code{is\_erasableb} and \code{erase\_global\_deps}
  (\code{erasure/theories/ErasureFunction.v}).
\end{itemize}

\begin{definition}[Exclusions of the fragment]
  \label{def:Erasure-declExclusion}
  \lean{Erasure.declExclusion, Erasure.recShapeInFragment, Erasure.exprConsts, Erasure.arityShape, Erasure.mayBeNonZero,
    Erasure.sortDecided}
  \leanok
  \uses{def:Erasure-EraseError, def:Erasure-EnvView, def:Erasure-ErasureConfig}
  \srcloc{LeanToLambdaBox/Erasure/Entry.lean}{100}
  \lead{In short} The exclusions that \hl{a term or a single declaration carries by itself}; each
  is \code{outOfFragment}.
  \begin{itemize}
  \item \code{exprConsts cfg e}: the constants of \code{e} (\code{Nat} for a literal), \hl{the
        structure of each projection} (S-30, as MetaRocq's \code{term\_global\_deps}), or a free
        variable, metavariable, level metavariable, string literal, and under
        \code{nat := .machine} a \code{Nat} literal or \code{Nat}/\code{Int} constructors and
        \code{casesOn};
  \item \code{declExclusion}: a recursor unless \code{recShapeInFragment} holds of it (a single
        inductive without indices, a universe parameter more than the inductive's, induction
        hypotheses without arguments and about the last fields; S-47), an \code{unsafe} or
        \code{partial} constant, one with an \code{\_unsafe\_rec} companion (except, under
        \code{structural := .kernel}, a definition for which \code{isStructuralRecursive} holds;
        S-44) or an \code{implemented\_by} target, a nested inductive, an \code{@[extern]} constructor under \code{preferAxiom};
        a definition, theorem or opaque of a mutual block, whose \code{all} is not its own name
        alone (S-51);
  \item \code{sortDecided}: an inductive's sort is written \code{Prop} or never zero
        (\code{true}), may be both (\code{false}, by \code{mayBeNonZero}), or is zero without
        being written \code{Prop} (excluded); an inductive whose type is not syntactically
        an arity (\code{arityShape} is \code{none}, as for an alias of a sort) is excluded;
  \end{itemize}
\end{definition}

\begin{definition}[The fragment test]
  \label{def:Erasure-collectDeps}
  \lean{Erasure.collectDeps, Erasure.closure, Erasure.closureStep, Erasure.dropKnown,
    Erasure.viewFind, Erasure.recognizeCasesOn, Erasure.eliminatedUndecided,
    Erasure.findBlockCollision, Erasure.findCollision, Erasure.collectFuel,
    instDecidableEqModPath, instDecidableEqKername}
  \leanok
  \uses{def:Erasure-declExclusion, def:Erasure-casesInfo, def:Erasure-findConst,
    def:toKername}
  \srcloc{LeanToLambdaBox/Erasure/Entry.lean}{271}
  \lead{In short} \code{collectDeps view cfg e}: \hl{the closure of \code{e} as the pure
  traversal reads it}, or the first exclusion of Section~\ref{sec:scope-fragment}.
  \begin{itemize}
  \item \lead{Reads} the term; the type of every constant; the structure of every projection; the values of definitions and opaques,
        not of theorems; block members, the constructors of an inductive, the inductive of a
        constructor; for a recognized \code{T.casesOn} (\code{recognizeCasesOn}), \code{T} and
        \code{T.rec}, not the value; for an inductive \code{T} with one constructor, \code{T.rec}
        (S-36), which the oracle reads to type a projection;
  \item \code{closure}: depth-first, at most \code{collectFuel} $= 2^{32}$ declarations; a
        recursor met in a read expression is checked by \code{declExclusion} (\code{dropKnown}); an unknown constant is
        \code{outOfFragment}; a view that answers with a declaration of another name is rejected;
        a recognized \code{casesOn} is recorded only when it is a definition;
  \item \code{eliminatedUndecided}: an inductive whose sort may be both \code{Prop} and not,
        cased on or projected, is excluded; so is any inductive under
        \code{remove\_irrel\_constr\_args};
  \item \lead{Collisions} after the scan, \code{nameCollision} when two closure constants share a
        kername (\code{findCollision}, R-3), or a block name is shared or is the kername of a
        closure constant (\code{findBlockCollision}).
  \end{itemize}
  \lead{Reference} the environment \code{erase\_global\_deps} keeps, computed on the source side
  (DV-3); the hypothesis \code{hfrag} of the final theorem (DV-26).
\end{definition}

\begin{definition}[The \texorpdfstring{\code{casesOn}}{casesOn} recognizer]
  \label{def:Erasure-casesInfo}
  \lean{Erasure.casesInfo?, Erasure.ctorArity?, Erasure.peelLams, Erasure.forallDom?,
    Erasure.forallArity, Erasure.minorArity, Erasure.spine, Erasure.isBVarAt, Erasure.isConstAt,
    Erasure.isMinorBody, Erasure.ctorAlts, Erasure.altFields, Erasure.blockCtorCount}
  \leanok
  \uses{def:Erasure-findConst}
  \srcloc{LeanToLambdaBox/Erasure/Pure.lean}{91}
  \lead{In short} \code{casesInfo? decls n}: the \code{CasesInfo} of \code{n} exactly when
  \code{n} is \hl{a canonical \code{T.casesOn}} of \code{decls}, as \code{getCasesInfo?} gives it
  on the \code{Meta} path.
  \begin{itemize}
  \item \lead{Canonical} \code{T} safe, not nested, its recursor in \code{decls} without auxiliary
        motives; the value is \code{fun ps motive is t ms => T.rec ps motives' minors' is t} at
        the recursor's level parameters, the own minors \code{fun fs ihs => m$_i$ fs};
  \item \lead{Type} \hl{at least the recognized arity of syntactic $\Pi$s} (\code{forallArity},
        S-37): a type behind \code{mdata} or an alias is not recognized, so the
        $\eta$-expansion's binders are the type's own;
  \item \lead{Not recognized} a \code{casesOn} built from projections (large-eliminating
        \code{Prop} structures), a sparse \code{casesOn}, \code{T.c.elim};
  \item \code{ctorArity?}: parameters plus fields of a constructor of \code{decls}.
  \end{itemize}
  \lead{Reference} \code{Lean.getCasesInfo?}; the shape of \code{Lean.mkCasesOn}.
\end{definition}

\begin{definition}[The pure erasability oracle]
  \label{def:Erasure-Pure-isErasable}
  \lean{Erasure.Pure.isErasable, Erasure.Pure.Ctx, Erasure.Pure.alwaysZero, Erasure.Pure.findLocal,
    Erasure.Pure.whnf, Erasure.Pure.inferType, Erasure.Pure.isArity,
    Erasure.Pure.instForalls, Erasure.Pure.instProjs, Erasure.Pure.isPropFormerType,
    Erasure.Pure.typeLevel, Erasure.oracleFuel, Erasure.Pure.iotaMinors?, Erasure.Pure.iotaRhs?,
    Erasure.Pure.iotaApp?}
  \leanok
  \uses{def:Erasure-EraseError, def:Erasure-Pure-instLevels, def:Erasure-TravCtx,
    def:Erasure-findConst}
  \srcloc{LeanToLambdaBox/Erasure/Pure.lean}{401}
  \lead{In short} \code{Pure.isErasable cx fuel ls e}: \code{e} is erasable if its head is a
  theorem, its type is an arity, or its type's sort is evidently zero; every reduction uses
  \hl{the kernel's $\delta$}.
  \begin{itemize}
  \item a spine headed by a theorem of the closure is erasable without inference (S-34): the
        kernel types a theorem's statement in \code{Prop}, which the oracle may not see through
        a recursor;
  \item \code{Ctx}: the program's declarations; \code{findLocal}: a traversal local by variable;
  \item \code{whnf}: head $\beta$, $\zeta$ (also of let-bound locals), $\delta$ of every
        definition at its number of levels, \code{mdata}, $\iota$ of a projection of a
        constructor, and $\iota$ of a recursor (S-43, S-48): \code{iotaMinors?} accepts the
        \code{T.rec} of one inductive without indices whose minors have induction hypotheses
        without binders on the last fields, \code{iotaApp?} a saturated application of it, and
        \code{iotaRhs?} builds the $\beta$-reduced right-hand side when the major's normal form is
        a constructor application of \code{T}: the minor applied to the fields and to
        \code{T.rec} on each recursive field; a \code{Nat} literal major is not reduced;
  \item \code{inferType}: infer-only retyping; binders the oracle enters are de Bruijn, free
        variables are the traversal's locals; a \code{Nat} literal has type \code{Nat}; a
        projection has its field's type, through \code{instForalls} and \code{instProjs}, as
        Lean's \code{inferProjType}, \hl{only for a large-eliminating inductive}: when
        \code{S.rec} has as many level parameters as \code{S}, it is an error (S-36);
  \item \code{isArity} after \code{whnf}; \code{alwaysZero}: \code{zero}, \code{max} of two,
        \code{imax} with a right one;
  \item \code{isPropFormerType}: \code{Prop} or a $\Pi$-type ending in \code{Prop} (S-25);
        \code{typeLevel}: the level of the sort of a term's type (S-27);
  \item fuel exhaustion and ill-typed input are errors; \code{oracleFuel} $= 2^{20}$ per call.
  \end{itemize}
  \lead{Reference} \code{is\_erasableb} (\code{erasure/theories/ErasureFunction.v}); MetaCoq paper
  Section 7.2, Fig.~17; the procedure is ours.
\end{definition}

\section{The pure backend and the entry point}\label{sec:pure-backend}

\begin{definition}[The pure backend]
  \label{def:Erasure-PureM}
  \lean{Erasure.PureM, Erasure.PureCtx, Erasure.PureState, Erasure.EraseT.runPure,
    Erasure.PureM.findConst?, Erasure.PureM.declInfo?, Erasure.PureM.unsafeRecBase?,
    Erasure.PureM.freshFVarId, Erasure.PureM.instantiate1, Erasure.PureM.isErasable,
    Erasure.PureM.isPropFormerType, Erasure.PureM.typeLevel, Erasure.PureM.whnf,
    Erasure.PureM.casesInfo?, Erasure.PureM.ctorArity?, Erasure.PureM.prepare,
    Erasure.instBackendPureM}
  \leanok
  \uses{def:Erasure-EraseT, def:Erasure-findConst, def:Erasure-Pure-isErasable,
    def:Erasure-EnvView, def:Erasure-casesInfo}
  \srcloc{LeanToLambdaBox/Erasure/Pure.lean}{435}
  \lead{In short} $\mathrm{PureM} = $ \code{ReaderT PureCtx (StateT PureState
  (Except EraseError))}: the traversal's backend \hl{without \code{Meta}}.
  \begin{itemize}
  \item \code{PureCtx}: the closure and the view; \code{PureState}: a counter for fresh
        variables;
  \item \code{findConst?} $=$ \code{declInfo?}: lookup in the closure, no \code{\_unsafe\_rec}
        redirect; \code{declInfo?} ignores the configuration, so \code{structural} has no effect
        on the pure path; \code{instantiate1}: the structural \code{Expr.instantiate1'};
  \item \code{isErasable}, \code{inferType}, \code{isPropFormerType}, \code{typeLevel},
        \code{whnf}: the pure oracle at \code{oracleFuel}; \code{whnf} reduces an alternative's
        type to a $\Pi$ (S-31); \code{casesInfo?}, \code{ctorArity?}: the recognizer
        over the closure;
  \item \code{prepare}: the identity (no \code{macro\_inline}, matcher inlining or
        \code{csimp});
  \item the instance: every constructor field kept, no instances, no logging; a constant missing
        from the closure is \code{outOfFragment}, fuel exhaustion the error \code{fuel};
        \code{error} and \code{fail} are the error \code{failed}, so \hl{an unexpected case of
        the traversal ends the pure path with an error} (S-33).
  \end{itemize}
\end{definition}

\begin{definition}[The entry point]
  \label{def:Erasure-eraseEntry}
  \lean{Erasure.eraseEntry, Erasure.route, Erasure.Route, Erasure.dupKername}
  \leanok
  \uses{def:Erasure-collectDeps, def:Erasure-erasePure, def:Erasure-erase}

  \lead{In short} What \code{\#erase} runs: \code{route} calls \code{collectDeps},

  \begin{itemize}
  \item \code{ok decls}: \code{erasePure}; its errors are errors of \code{\#erase}, with

  \item any other error: an error of \code{\#erase}.
  \end{itemize}

\section*{Caveats}
\begin{itemize}
\item \lead{Caveat} The pure path skips Lean's compiler passes: on some in-fragment corpus files
  its output differs from the \code{Meta} path's (register entry S-21).
\item \lead{Caveat} \code{EnvView.ofEnvironment} decides which constants are \code{@[extern]}, and
  so which constants the source semantics treats as stuck.
\item \lead{Caveat} The fragment test is syntactic: it does not check types, which the
  hypotheses of the final theorem state separately.
\item \lead{Caveat} \code{Pure.whnf} has $\iota$ only for the recursors of one inductive without
  indices whose induction hypotheses take no arguments and are about the last fields
  (\code{iotaApp?}), not for others such as \code{Eq.rec}, for quotients, or for a \code{Nat}
  literal major: an alternative whose type needs another $\iota$ to become a $\Pi$ is an error of
  the pure path (S-31), where the \code{Meta} path reduces it.
\item \lead{Caveat} The pure path fails on a kernel-valid projection of a proof of an inductive
  that is not large-eliminating (S-36); Lean's elaborator never produces one.
\end{itemize}
