\section{What a run is}
\label{sec:run:spelling}

\begin{definition}[A run of an \code{EraseM} computation]
\label{def:run-spelling}
\uses{def:Erasure-EraseM}
\lead{In short} \code{EraseM} unfolds to a function of five arguments; a successful run is a
\textbf{bare equation} on that unfolding, never a wrapper.
\begin{itemize}
\item \code{EraseM alpha} is \code{StateT ErasureState (ReaderT ErasureContext CoreM)}, and
\code{CoreM} unfolds to \code{ReaderT Core.Context (StateRefT' IO.RealWorld Core.State (EIO
Exception))};
\item an \code{x : EraseM alpha} takes five arguments: erasure state \code{s}, erasure reader
\code{ctx}, core reader \code{cctx : Core.Context}, reference cell \code{ref : ST.Ref IO.RealWorld
Core.State}, world token \code{w : Void IO.RealWorld};
\item \lead{Then} the run of \code{x} from \code{(s, ctx, cctx, ref, w)} \textbf{succeeds} with result
\code{r}, final state \code{s'} and world \code{w'} exactly when
\code{x s ctx cctx ref w = .ok (r, s') w'}.
\end{itemize}
\lead{Lean} this equation, not a \code{def} wrapper, is every lemma's hypothesis and every
induction's motive shape in this chapter through Chapter~\ref{chap:coldstart}, so goals agree on the
head symbol; the node has no declaration of its own. ``The run of the family'' names
\code{Erasure.visitExpr.mutual\_fixpoint\_induct},
\code{partial\_fixpoint}'s auto-generated (so uncitable) principle for the eighteen-member
\code{mutual} block.
\end{definition}

\begin{remark}
\lead{Why} \code{partial\_fixpoint} packs the eighteen members into one \code{PProd} chain in
\textbf{tuple order}, not source order (\code{visitLambda} is third in the source but ninth in the
tuple); every \code{motive\_1} \ldots \code{motive\_18} numbering follows the tuple order.
\end{remark}

\section{The run algebra}
\label{sec:run:algebra}

\begin{lemma}[Inversion for a successful bind]
\label{lem:Erasure-run_bind_ok}
\lean{Erasure.run_bind, Erasure.run_bind_ok, Erasure.run_bind_ne_ok}
\leanok
\uses{def:run-spelling}
\lead{In short} \hl{A bind's run succeeds exactly when there is a successful intermediate run}.
\begin{itemize}
\item \lead{Given} a bind \code{x >>= f} run from \code{(s, ctx, cctx, ref, w)};
\item \lead{Then} \code{(x >>= f) s ctx cctx ref w = .ok (b, s') w'} iff there are \code{a}, \code{s\_1},
\code{w\_1} with \code{x s ctx cctx ref w = .ok (a, s\_1) w\_1} and
\code{f a s\_1 ctx cctx ref w\_1 = .ok (b, s') w'}; on failure the error propagates.
\end{itemize}
\lead{Lean} a third corollary, from the contrapositive (\code{f} fails from every intermediate
result), discharges every error path.
\end{lemma}
\begin{proof}
\leanok
\uses{def:run-spelling}
\lead{Idea} Restate the goal at the \code{EST} layer, unfold \code{EST.bind}, and case on the first
action's outcome.
\end{proof}

\begin{lemma}[The primitive runs]
\label{lem:Erasure-run_pure}
\lean{Erasure.run_pure, Erasure.run_get, Erasure.run_set, Erasure.run_modify,
Erasure.run_modifyGet, Erasure.run_read, Erasure.run_read_bind, Erasure.run_withReader,
Erasure.run_throw, Erasure.run_throw_ne_ok, Erasure.run_throwError_ne_ok}
\leanok
\uses{def:run-spelling}
\lead{In short} Each monad primitive's run is exactly what it looks like.
\begin{center}
\begin{tabular}{ll}
\toprule
Primitive & Run \\
\midrule
\code{pure a} & succeeds immediately with \code{a}; state, world unchanged \\
\code{get} & returns the state \\
\code{set s0} & installs \code{s0} \\
\code{modify g} & replaces the state by \code{g s} \\
\code{modifyGet g} & does both \\
\code{read} & returns the erasure reader \\
\code{withReader f x} & runs \code{x} under \code{f ctx} \\
\code{throw e}, \code{throwError msg} & never return \code{.ok} \\
\bottomrule
\end{tabular}
\end{center}
\lead{Lean} \code{run\_read\_bind} chains \code{(read >>= f) s ctx cctx ref w} to
\code{f ctx s ctx cctx ref w}.
\end{lemma}
\begin{proof}
\leanok
\uses{lem:Erasure-run_bind_ok}
\lead{Idea} All but three are \code{rfl} once the run hits all five arguments; \code{run\_read\_bind}
composes the bind equation with \code{run\_read}, \code{run\_throw\_ne\_ok} is a \code{nomatch}, and
\code{run\_throwError\_ne\_ok} unfolds to opaque \code{getRef}/\code{addMessageContext} binds ending
in \code{run\_throw\_ne\_ok}.
\end{proof}

\begin{lemma}[Lifted actions are state-transparent]
\label{lem:Erasure-run_liftCoreM_ok}
\lean{Erasure.run_liftCoreM, Erasure.run_liftCoreM_ok, Erasure.run_liftCoreM_state,
Erasure.run_liftMetaM, Erasure.run_liftMetaM_ok, Erasure.run_liftMetaM_state}
\leanok
\uses{def:run-spelling}
\lead{In short} \hl{An action lifted from \code{CoreM} cannot touch the erasure state}.
\begin{itemize}
\item \lead{Given} \code{(liftM x : EraseM alpha)} for \code{x : CoreM alpha}, or
\code{Erasure.liftMetaM x} for \code{x : MetaM alpha} under the reader's local context, running it
carries the erasure state through unchanged; only the world token moves.
\end{itemize}
\end{lemma}
\begin{proof}
\leanok
\uses{lem:Erasure-run_bind_ok, lem:Erasure-run_pure}
\lead{Idea} Both lifts unfold to a bind of a \code{CoreM} action with a \code{pure} re-pairing the
incoming state; the bind equation and \code{run\_pure} give the result.
\end{proof}

\begin{lemma}[A panic succeeds]
\label{lem:Erasure-run_panic}
\lean{Erasure.run_panic, Erasure.run_panicWithPosWithDecl}
\leanok
\uses{def:run-spelling}
\lead{In short} At \code{EraseM}, \hl{\code{panic!} does not throw}.
\begin{itemize}
\item \lead{Given} the \code{Inhabited} instance \code{panic!}/\code{unreachable!} pick up at
\code{EraseM alpha}, \code{(panic msg : EraseM alpha) s ctx cctx ref w = .ok (default, s) w}, and
likewise for the elaborated \code{panicWithPosWithDecl} form the macros produce.
\end{itemize}
\end{lemma}
\begin{proof}
\leanok
\lead{Idea} Both are \code{rfl}: the stack's \code{Inhabited} instance reduces \code{panic} (and its
elaborated form) to \code{pure default} once applied to all five arguments.
\end{proof}

\begin{remark}
\lead{Caveat} Three shipping consequences are modelled, not excluded: \code{addAxiom}'s duplicate
guard still registers a second entry (Theorem~\ref{thm:Erasure-run_addAxiom_ok}); a binder helper can
return a default \lbox{} term on an unexpected shape (Lemma~\ref{lem:Erasure-run_withLocalDecl_ok});
\code{register\_inductive}'s \code{unreachable!} arms fall through successfully.
\end{remark}

\leanok
\uses{def:run-spelling}
\lead{In short} \hl{``On a successful run, \code{Q} holds'' is admissible} at the bottom transformer

\begin{itemize}
\item \lead{Given} any \code{Q : Void sigma -> alpha -> Void sigma -> Prop}, the predicate
\code{fun x => forall w a w', x w = .ok a w' -> Q w a w'} on \code{EST eps sigma alpha} is closed

\end{itemize}
\end{lemma}
\begin{proof}
\leanok
\lead{Idea} In the pointwise CCPO of flat orders over \code{EST.bot} (\code{EST}'s own instance),
\code{admissible\_pi\_apply}/\code{admissible\_pi} peel the quantifiers and
\code{admissible\_flatOrder} reduces to the motive at bottom, which holds vacuously.
\end{proof}

\begin{lemma}[The canonical run-ok motive is admissible]
\label{lem:Erasure-eraseM_admissible_ok}
\lean{Erasure.eraseM_admissible_ok}
\leanok
\uses{def:run-spelling, def:Erasure-EraseM}
\lead{In short} For any predicate \code{Q} of a run's inputs and outputs, \hl{the run-ok property of
\code{Q} is admissible} on \code{EraseM tau}, hence a legal fixpoint-induction motive.
\lead{Lean} five curried, Unicode-subscripted variants (arity one to five) cover all eighteen members.
\end{lemma}
\begin{proof}
\leanok
\uses{lem:Erasure-est_admissible_ok}
\lead{Idea} Four applications of \code{admissible\_pi\_apply}, one per transformer layer with its
predicate supplied explicitly, reduce to the paired leaf lemma; each further arity adds one
application (raising the instance-synthesis budget at arities four and five).
\end{proof}

\begin{lemma}[Below the fixpoint, a successful run is the fixpoint's run]
\label{lem:Erasure-run_ok_of_le}
\lean{Erasure.approx_rfl, Erasure.run_ok_of_le}
\leanok
\uses{def:run-spelling}
\lead{In short} \hl{A successful run of \code{x} below \code{y} is verbatim a run of \code{y}}.
\begin{itemize}
\item \lead{Given} \code{x, y : EraseM tau} with \code{x} below \code{y} in
\code{partial\_fixpoint}'s order, every successful run of \code{x} matches \code{y}'s result, state
and world.
\end{itemize}
\lead{Lean} the converse fails, so the induction conjunct must be the order itself, not run
agreement; a one-argument variant states the same for an applied family member.
\end{lemma}
\begin{proof}
\leanok
\lead{Idea} The order is pointwise down to the flat order over \code{EST.bot} (``bottom below
anything''): apply the hypothesis at the five arguments; the bottom case is absurd (\code{EST.bot}
is an \code{.error}), the reflexivity case identical runs.
\end{proof}

\leanok

\lead{Idea} \code{CCPO.csup\_le} (a chain below \code{c} has supremum below \code{c}) and

\begin{definition}[The modelled registration states]

\lead{In short} The exact post-states of the four ways a run can register something.
\begin{itemize}
\item \code{addAxiomState n s}: registry gains \code{n} at \code{toKername n}; declarations gain one

\item \code{addRealizerState n t s}: the same, with body \code{t} instead of none -- what

\item \code{mutualBlockKn ii}: the kername minted for a block, the root of its members' concatenated

\code{indBlocks}, so a later block minting the same key is caught (Definition~\ref{def:Erasure-KernameFresh})
instead of overwriting the entry.
\end{itemize}
\lead{Lean} \code{mutualBlockKn\_eq\_toKername}: at a singleton, root-named block,
\code{mutualBlockKn} agrees with \code{toKername} -- printing the sole member back as that root
component is the whole gap.
\end{definition}

\begin{proposition}[\code{visitMutual} in Hoare form over its five exits]
\label{prop:Erasure-run_visitMutual_ok}
\lean{Erasure.run_visitMutual_ok}
\leanok
\uses{def:Erasure-addAxiomState, def:Erasure-nonrecConstState, def:Erasure-visitMutual,
def:Erasure-prepare_erasure}
\lead{In short} A predicate \code{Q} closed under all five exits \hl{survives every successful
\code{visitMutual} run}.
\begin{itemize}
\item \lead{Given} \code{Q} closed under: the inline entry; the axiom delta; the realizer delta
(F-QUOT, F-EQREC); a \code{recursorRealizer} run, abstract; a preprocessing-pass run; a
shipping-\code{visitExpr} run yielding a body meeting predicates \code{Nf}, \code{Cl}; the
non-recursive registration, given \code{Nf}/\code{Cl}; and the block registration, given the block's
length and that each body is the closure of a good erasure output -- then \code{Q} at entry gives
\code{Q} at exit.
\end{itemize}
\end{proposition}
\begin{proof}
\leanok
\uses{prop:Erasure-run_nonrec_exit_ok, lem:Erasure-run_bind_ok, lem:Erasure-run_liftCoreM_ok,

\lead{Idea} Step the declaration query and environment read; split four discriminants at once
(quotient, recursor, has-value, externally-implemented, config) by nested cases, since \code{split}
cannot: axiom and realizer (Theorem~\ref{thm:Erasure-run_addAxiom_ok}), the recursor-realizer clause
handed straight to its hypothesis, the rest by Proposition~\ref{prop:Erasure-run_nonrec_exit_ok}.
\end{proof}

\begin{remark}
\lead{Caveat} The preprocessing-pass clause is a \emph{hypothesis}, not proved: its simplification
branch runs a \code{Core} transformation \emph{at} \code{EraseM}, so lifting does not give state
transparency. Downstream, \code{run\_prepare\_erasure\_state} (Chapter~\ref{chap:coldstart})
discharges it when the simplification flag is off; the flag-on case is unhandled.
\end{remark}

\begin{proposition}[The four exit rules, state form]
\label{prop:Erasure-run_nonrec_exit_ok}
\lean{Erasure.run_inline_tail_ok, Erasure.run_inline_prefix_ok, Erasure.run_nonrec_exit_ok,
Erasure.run_rec_exit_ok}
\leanok
\uses{def:Erasure-nonrecConstState, def:Erasure-addAxiomState, def:Erasure-mkLambda}
\lead{In short} Four Hoare rules, one per exit shape, boolean tests and log messages left abstract.
\begin{center}
\begin{tabular}{p{0.25\linewidth}p{0.68\linewidth}}
\toprule
Rule & What it covers \\
\midrule
inline \emph{tail} & post-registration bookkeeping; prepends at most one inline-list entry \\
inline \emph{prefix} & an optional attribute test, then the same continuation either way \\
non-recursive exit & steps the F-KERNAME freshness guard (silently, for this rule), erases the
preprocessed body under the reader update, prepends the constant, runs the inline tail \\
recursive exit & mints one fresh variable per sibling, refuses unless the members' kernames are
distinct (F-UNSAFEREC; Theorem~\ref{thm:Erasure-run_rec_exit_nodup} reads that guard), erases each
body via \code{mkDef} under the fixpoint binding, steps a freshness guard and prepends one
$\eta$-expanded registry entry per name \\
\bottomrule
\end{tabular}
\end{center}
\lead{Lean} the recursive exit's closure hypothesis takes the stored block's shape: one definition
per name, each the closure of a good erasure output over the block's names.
\end{proposition}
\begin{proof}
\leanok
\uses{lem:Erasure-run_bind_ok, lem:Erasure-run_pure, lem:Erasure-run_list_mapM_ok,
thm:Erasure-run_mkDef_ok, lem:Erasure-run_modify_forIn_ok, lem:Erasure-run_getConstInfo_state,
lem:Erasure-run_checkKernameFresh_ok}
\lead{Idea} Each rule steps its \code{do}-block with the bind inversion: \code{mapM} for the sibling
loop, the definition-shape theorem for \code{mkDef}, the fold rule for registration; a guard's
freshness reading is available but, for this Hoare form, discarded -- only \code{Q} at exit is
concluded. The recursive exit's \code{Nodup} test is stepped by \code{split}, its throwing branch
refuted by \code{run\_throwError\_ne\_ok} (part of Lemma~\ref{lem:Erasure-run_pure}).
\end{proof}

\begin{proposition}[The four exit rules, world-indexed]
\label{prop:Erasure-run_nonrec_exit_ok-prime}
\lean{Erasure.run_inline_tail_ok', Erasure.run_inline_prefix_ok', Erasure.run_nonrec_exit_ok',
Erasure.run_rec_exit_ok'}
\leanok
\uses{prop:Erasure-run_nonrec_exit_ok, def:Erasure-nonrecConstState}
\lead{In short} The same four rules for a predicate of the state \hl{and the world token}.
\begin{itemize}
\item \lead{Given} every state-transparent primitive (logging, instance test, minting, lookup) now
needs its own world-preservation clause; the erasure clause is keyed on both the preprocessing and
the erasure run, so a caller re-establishing an invariant at entry sees the run that produced it.
\end{itemize}
\end{proposition}
\begin{proof}
\leanok
\uses{prop:Erasure-run_nonrec_exit_ok, lem:Erasure-run_bind_ok, thm:Erasure-run_mkDef_ok,
lem:Erasure-run_modify_forIn_ok, lem:Erasure-run_list_mapM_ok, lem:Erasure-run_checkKernameFresh_ok}
\lead{Idea} The same walks, world token carried; the definition builder and registration loop are
world-preserving, needing no clause; the freshness guards are stepped and discarded as in the
state-only form.
\end{proof}

\begin{theorem}[The F-UNSAFEREC guard's reading: a block's keys are distinct]
\label{thm:Erasure-run_rec_exit_nodup}
\lean{Erasure.run_rec_exit_nodup}
\leanok
\uses{def:run-spelling}
\lead{In short} A successful recursive-exit run \hl{proves the members' kernames distinct} --
\code{Erasure.remove\_unsafe\_rec} is not injective (it maps both \code{u} and its
\code{u.\_unsafe\_rec} sibling to \code{u}), so the guard, not an unconditional fact about
\code{Lean.Compiler.LCNF.getDeclInfo?}, is the only place this can be read: \code{EraserAsks} states
no unconditional version of it.
\lead{Status} \stProved{}.
\end{theorem}
\begin{proof}
\leanok
\uses{lem:Erasure-run_bind_ok, lem:Erasure-run_pure}
\lead{Idea} Step the sibling \code{mapM}, then \code{split} on the \code{Nodup} test: the true
branch is the conclusion directly, the false branch throws and is refuted by
\code{run\_throwError\_ne\_ok} (part of Lemma~\ref{lem:Erasure-run_pure}).
\end{proof}

\begin{lemma}[Each registration delta is a run conclusion]
\label{lem:Erasure-runConcl_nonrecConstState}
\lean{Erasure.runConcl_inlinings, Erasure.runConcl_addAxiomState,
Erasure.runConcl_nonrecConstState, Erasure.runConcl_addRealizerState,
Erasure.runConcl_foldl_recConstStep, Erasure.runConcl_recConstState,
Erasure.nonrecConstState_get?, Erasure.addAxiomState_get?, Erasure.addRealizerState_get?,
Erasure.foldl_recConstStep_get?, Erasure.recConstState_get?}
\leanok
\uses{def:Erasure-RunConcl, def:Erasure-nonrecConstState, def:Erasure-addAxiomState}
\lead{In short} Each of the five state deltas the eraser performs is \hl{a \code{RunConcl} step}.
\begin{itemize}
\item the inline entry writes only the inline list;
\item the axiom exit registers one constant at its canonical kername, prepends one value-less
declaration; the non-recursive and the realizer exits do the same with a body, the realizer delta
reducing to the non-recursive one verbatim;
\item the block registration, a fold of the non-recursive step, composes out of the previous one.
\end{itemize}
\end{lemma}
\begin{proof}
\leanok
\uses{def:Erasure-RunConcl, def:Erasure-nonrecConstState}
\lead{Idea} Each delta inserts at most one key and prepends at most one declaration, so extension
and canonicity read off directly; the block case inducts over the fold by transitivity.
\end{proof}

\begin{theorem}[The identifier-minting loop]
\label{thm:Erasure-run_mkFreshFVarId_list}
\lean{Erasure.run_mkFreshFVarId_list}
\leanok
\uses{def:run-spelling}
\lead{In short} Minting one fresh identifier per name in a list gives \hl{pairwise distinct}
identifiers.
\begin{itemize}
\item \lead{Given} a freshness specification: each successful minting run returns an identifier
unreserved before, reserved after, the generator advanced monotonically, and reserved by a fixed
ambient generator;
\item \lead{Then} minting once per name returns as many identifiers as names, pairwise distinct, at
an unchanged state, generator advanced, each reserved by the final and the ambient generator, and
outside any given list the initial generator reserves.
\end{itemize}
\lead{Status} \stProved{}.
\end{theorem}
\begin{proof}
\leanok
\uses{lem:Erasure-run_list_mapM_ok, lem:Erasure-run_getConstInfo_state}
\lead{Idea} One \code{mapM} application chaining state and world: each new identifier is unreserved
while every earlier one is (invariant, monotonicity); disjointness from an ambient list is the same
argument.
\end{proof}

\begin{remark}
\lead{Caveat} The generator vocabulary is lean4lean's, from a 28-line \code{sorry}-free module: no
trust inherited.
\end{remark}

\begin{theorem}[The sibling loop, chained]
\label{thm:Erasure-run_rec_exit_siblings_chained}
\lean{Erasure.run_rec_exit_siblings_chained}
\leanok
\uses{def:run-spelling, def:Erasure-prepare_erasure, def:Erasure-mkLambda, def:FixDef}
\lead{In short} \code{visitMutual}'s sibling loop, run \hl{one index at a time in run order}.
\begin{itemize}
\item \lead{Given} a per-sibling package \code{R} and a step hypothesis handed the four primitive
runs in run order at the states/worlds they consume: lookup, preprocessing under the sibling's own
reader, the abstract eraser, the definition builder;
\item \lead{Then} a successful run produces as many definitions as names, carries the invariant from
entry to exit, and returns, at every index, a definition satisfying \code{R} for that name.
\end{itemize}
\lead{Status} \stProved{}.
\end{theorem}
\begin{proof}
\leanok
\uses{lem:Erasure-run_list_mapM_ok, lem:Erasure-run_bind_ok, lem:Erasure-run_pure}
\lead{Idea} One \code{mapM} application carrying output length, the caller's invariant and the
per-index package, feeding lookup, reader change and definition builder to the step hypothesis in
order.
\end{proof}

\section{How the run model is consumed}
\label{sec:run:world}

Two clients spend this library.
\begin{itemize}
\item \textbf{Refinement} (Chapter~\ref{chap:refinement}): eighteen motives in run-ok form, each
concluding an \Erases-flavoured relation with \code{RunConcl}
(Theorem~\ref{thm:visitExpr_refines_erasesLB}).
\item \textbf{Cold start} (Chapter~\ref{chap:coldstart}): the same induction, world-indexed
(\code{RunClosedW}, Definition~\ref{def:RunClosedW}), concluding for every \lbox{}-returning member
the shape \code{ShapeCW} (Definition~\ref{def:ShapeCW}): fix-free, de Bruijn closed, applied. Ten
clauses cover the ambient primitives (state-transparent, world-advancing); two read
\code{ConfigPinned} (Definition~\ref{def:ConfigPinned}), five pinned restrictions -- no
simplification-attribute replacement, no external-constant axiomatisation, Peano naturals, no
argument-mask pruning, no typeclass-dispatch inlining -- three of which the default \code{\#erase}
configuration violates.
\item \textbf{Registration invariants} (Chapter~\ref{chap:refinement}): the per-field cold-branch
deltas (Lemma~\ref{lem:Erasure-run_register_inductive_cold_constants}) and
\code{IndBlocksCover}'s mirror (Section~\ref{sec:run:indblocks}) are what an accumulated
registration invariant reads at one block registration; that invariant's own preservation is
proved there, not here.
\end{itemize}

\section{Deviations from the paper and caveats}
\label{sec:run:caveats}

\begin{itemize}
\item \lead{Deviation} \textbf{No environment pass.} [JACM, Sec.~7.3] erases a whole environment first;
this eraser builds \lbox{} declarations \emph{during} the walk, demand-driven and memoised
(Lemma~\ref{lem:Erasure-run_get_constant_kername_ok}), via \ErasesEnv's operational counterpart
(Section~\ref{sec:run:reg}): a bodyless axiom, a constant with a body, or a block member.
\item \lead{Deviation} \textbf{Program logic, not equational reasoning.} MetaCoq's proofs induct on
equations; here it is Hoare rules, admissibility and approximation -- the price of verifying an
eraser not written to be verified.
\item \lead{Caveat} \textbf{Panics as successful branches.} [JACM] has no analogue: a panic carries a
default fall-through everywhere (Lemma~\ref{lem:Erasure-run_panic}), recording the ineffective
duplicate guard, not excluding it -- unlike the F-KERNAME/F-UNSAFEREC guards
(Definition~\ref{def:Erasure-KernameFresh}, Theorem~\ref{thm:Erasure-run_rec_exit_nodup}), which
genuinely throw, so their guarantees hold only of a successful run.
\item \lead{Deviation} \textbf{No constructor-argument pruning.}
Theorem~\ref{thm:Erasure-run_register_inductive_cold_entries}: pruning off, one of
\code{ConfigPinned}'s five restrictions, every mask keeps a constructor's whole field count.
\item \lead{Deviation} \textbf{Applied-form constructors.} This frontend applies a bare constructor
through application nodes, not MetaCoq/CertiCoq's block node; Chapter~\ref{chap:coldstart}
\emph{concludes} this rather than assuming it.
\item \lead{Caveat} \textbf{Sources.} This chapter's declarations live in namespace \code{Erasure},
not \code{LeanToLambdaBox.Erasure}.
\end{itemize}
