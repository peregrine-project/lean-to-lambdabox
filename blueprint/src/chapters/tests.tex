\chapter{Tests and non-vacuity}\label{chap:tests}

\section*{At a glance}
\begin{itemize}
\item \lead{Goal} Show that the hypotheses of the main statements can be discharged on concrete
  programs, relate the restated relations to lean4lean's, and pin the design choices of the
  divergence register by tests.
\item \lead{Main results} Five instances of \code{erases\_correct} (NV-1, NV-2, NV-5, NV-6,
  NV-7), each with a unique non-$\Box$ witness; \code{TrS} and \code{ProgEnv} are instances of
  lean4lean's \code{TrExprS} and \code{TrEnv'}; the register tests of DV-11 and DV-18; two
  instances of \code{Pure.isErasable\_sound}.
\item \lead{Status} \stProved{} every node; \stOpen{} NV-3, NV-4, and the instances of
  \code{erasePure\_erases} and \code{erase\_correct} (Section~\ref{sec:tests-instances}).
\item \lead{Lean files}
  \code{proof/EraseProof/Test/\{NV1, NV2, NV5, NV6, NV7, Bridge, LBEval, Atoms, Oracle\}.lean};
  namespace \code{EraseProof.Test}.
\end{itemize}
