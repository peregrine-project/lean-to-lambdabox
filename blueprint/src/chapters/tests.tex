\chapter{Tests and non-vacuity}\label{chap:tests}

\section*{At a glance}
\begin{itemize}
\item \lead{Goal} Show that the hypotheses of the main statements can be discharged on concrete
  programs, relate the restated relations to lean4lean's, and pin the design choices of the
  divergence register by tests.
\item \lead{Main results} The model-side hypotheses of NV-1; \code{TrS} and \code{ProgEnv} are
  instances of lean4lean's \code{TrExprS} and \code{TrEnv'}.
\item \lead{Status} \stProved{} NV-1's model side and the bridge tests; \stPlanned{} the register
  tests, determinism of $\lambda_\Box$ evaluation, and the other instances.
\item \lead{Lean files} \code{proof/EraseProof/Test/\{NV1, Bridge\}.lean}; namespace
  \code{EraseProof.Test}.
\end{itemize}

\begin{lemma}[Register test: definition heads are not atoms]
  \label{lem:EraseProof-Test-Atoms-defHead_kept}
  \lean{EraseProof.Test.Atoms.defHead_kept}
  \planned
  \uses{def:Erasure-Pure-isErasable, def:EraseProof-EvalEnv-isAtom}
  \lead{In short} With \code{def Q : Prop := forall P : Prop, P -> P} and \code{axiom hq : Q}, the
  oracle \hl{keeps the spine \code{hq A a}}, so a proof headed by a definition cannot be an atom
  (DV-11).
\end{lemma}
\begin{proof}
  \uses{def:Erasure-Pure-isErasable}
  \lead{Idea} \code{decide}.
\end{proof}

\begin{lemma}[Register test: atomhood does not depend on levels]

  \code{hq.\{0\}}, so \hl{atomhood cannot depend on the occurrence's levels} (DV-11).
\end{lemma}
\begin{proof}
  \uses{def:Erasure-Pure-isErasable}

\end{proof}

\begin{lemma}[Register test: the oracle's \texorpdfstring{$\delta$}{delta}]
