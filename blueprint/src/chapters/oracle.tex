\chapter{Soundness of the pure oracle}\label{chap:oracle}

\section*{At a glance}
\begin{itemize}
\item \lead{Goal} The pure oracle's answer "erasable" is true in the model, it boxes every atom,
  and its answer depends only on the locals the term reaches.
\item \lead{Main results} \code{Pure.isErasable\_sound}, \code{Pure.isErasable\_atom},
  \code{Pure.isErasable\_agree}.
\item \lead{Status} \stProved{} the contexts, soundness of \code{Pure.whnf} and
  \code{Pure.inferType} (inheriting L1--L5), and the frame lemma; \stPlanned{} the arity and level
  tests, \code{Pure.isErasable\_sound} and \code{Pure.isErasable\_atom}.
\item \lead{Lean files} \code{proof/EraseProof/Oracle/\{Whnf, Infer, Agree\}.lean}; planned:
  \code{Oracle.lean}, \code{Oracle/\{AtomShapes, Atom\}.lean}.
\item \lead{Reference} MetaRocq \code{is\_erasableb} and \code{is\_erasableP}
  (\code{erasure/theories/ErasureFunction.v}), \code{hnf\_sound}, \code{type\_of\_typing}
  (\code{safechecker/theories/}).
\end{itemize}

  \leanok
  \uses{def:EraseProof-TrS, def:Erasure-TravCtx}
  \srcloc{proof/EraseProof/Oracle/Whnf.lean}{21}
  \lead{In short} \code{LocalsOK venv Us ls Delta}: the traversal's locals \hl{mirror a lean4lean
  context} of free-variable entries, each type and \code{let} value translated and typed;
  \code{OracleCtx}: that context under the de Bruijn binders the oracle entered, whose types the
  oracle keeps in its list \code{Gamma}.
  \lead{Reference} \code{wf\_local} for the context of MetaRocq's \code{erase}; DV-13.
\end{definition}

  \srcloc{proof/EraseProof/Oracle/Whnf.lean}{74}

  translates.
  \begin{itemize}
  \item \code{LocalsOK.wf}, \code{OracleCtx.wf}: well-formed lean4lean contexts;
  \item \code{OracleCtx.weak}, \code{TrS.weak\_nil}: closed translations lift into the oracle's
        context;
  \item \code{LocalsOK.find\_value}, \code{find\_type}, \code{OracleCtx.find\_bvar}: a local's
        value and type, and a binder's lifted type, translate to the entry's.
  \end{itemize}
  \lead{Reference} \code{wf\_local} and \code{weakening}
  (\code{pcuic/theories/Typing/PCUICWeakeningTyp.v}); \code{type\_Rel}.
\end{lemma}
\begin{proof}
  \uses{lem:EraseProof-TrS-wf, lem:EraseProof-TrS-weakBV, lem:EraseProof-TrS-closed}
  \leanok
  \lead{Idea} Inductions on \code{LocalsOK} and \code{OracleCtx}; weakening by \code{TrS.weakBV}.
\end{proof}

\section{Soundness}\label{sec:oracle-sound}

\begin{lemma}[Weak-head normalization]
  \label{lem:EraseProof-Pure-whnf_sound}
  \lean{EraseProof.Pure.whnf_sound, EraseProof.Pure.whnf_defeqU}
  \leanok
  \uses{def:Erasure-Pure-isErasable, def:EraseProof-LocalsOK, def:EraseProof-ProgEnv,
    def:EraseProof-SubEnv}
  \srcloc{proof/EraseProof/Oracle/Whnf.lean}{317}
  \inherited{L1, L2, L3, L4, L5}
  \lead{In short} A successful \code{Pure.whnf} on a translated, typed term computes \hl{a typed
  definitional equality} of the model.
  \begin{itemize}
  \item \lead{Given} \code{henv}, \code{hsub}: the oracle's declarations are a sub-environment of
        \code{P}; \code{hloc}, \code{hGamma}: the contexts translate;
  \item \code{he}, \code{hT}: \code{e} translates to \code{e'} of type \code{A};
        \code{h}: \code{whnf} returns \code{w};
  \item \lead{Then} \code{w} translates to some \code{w'} with \code{e'} $\equiv$ \code{w'} at
        \code{A};
  \item \code{whnf\_defeqU}: the untyped form, with no typing hypothesis.
  \end{itemize}
  \lead{Reference} \code{hnf\_sound} (\code{safechecker/theories/PCUICSafeReduce.v}).
\end{lemma}
\begin{proof}
  \uses{lem:EraseProof-TrS-inst, lem:EraseProof-TrS-inst_let, lem:EraseProof-TrS-instLevels,
    lem:EraseProof-TrS-wf, lem:EraseProof-ProgEnv-lookupDefn, lem:EraseProof-LocalsOK-wf,
    lem:EraseProof-ProgEnv-wf}
  \leanok
  \lead{Idea} Fuel induction for \code{whnf\_defeqU}: $\beta$ by \code{TrS.inst}, $\zeta$ by
  \code{TrS.inst\_let} and the let-bound locals, $\delta$ by \code{ProgEnv.lookupDefn} and
  \code{TrS.instLevels}; typing by lean4lean's \code{IsDefEqU.of\_l}.
\end{proof}

\begin{lemma}[Oracle frame]
  \label{lem:EraseProof-Pure-isErasable_agree}
  \lean{EraseProof.Pure.isErasable_agree, EraseProof.Pure.whnf_agree,
    EraseProof.Pure.inferType_agree, EraseProof.Pure.isArity_agree,
    EraseProof.Pure.instLevel_hasMVar', EraseProof.Pure.fvarsIn_instLevels,
    EraseProof.Except.bind_congr_ok, EraseProof.Except.ok_of_bind}
  \leanok
  \uses{def:Erasure-Pure-isErasable, def:EraseProof-LocalsAgree}
  \srcloc{proof/EraseProof/Oracle/Agree.lean}{279}
  \lead{In short} With closed declarations, the oracle answers alike on two lists of locals that
  agree on a set \code{S} closed under the locals and containing the term's free variables: its
  answer \hl{depends only on the locals reachable from the term}.
  \begin{itemize}
  \item \code{whnf\_agree}, \code{inferType\_agree}, \code{isArity\_agree}: the same for the parts,
        whose results stay within \code{S};
  \item \code{instLevel\_hasMVar'}, \code{fvarsIn\_instLevels}, \code{Except.*}: level and
        \code{Except} plumbing.
  \end{itemize}
  \lead{Reference} none: MetaRocq erases a constant body in the empty context; DV-13.
\end{lemma}
\begin{proof}
  \uses{def:EraseProof-LocalsAgree}
  \leanok
  \lead{Idea} Fuel inductions through \code{whnf}, \code{inferType} and \code{isArity}.
\end{proof}

\section*{Caveats}
\begin{itemize}
\item \lead{Caveat} The oracle is sound and not complete (DV-18): an answer "keep" needs no proof,
  since the relation's structural rules also apply to erasable terms; atoms, which have no
  structural rule, are boxed (\code{Pure.isErasable\_atom}).
\end{itemize}
