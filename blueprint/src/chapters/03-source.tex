\section{The lean4lean grounding}
\label{sec:source:lean4lean}

Everything here is stated against a kernel model from the pinned lean4lean checkout, not a node of
this blueprint.

\begin{definition}[The lean4lean grounding]
\label{def:lean4lean-grounding}
\lean{Lean4Lean.TrExprS, Lean4Lean.VExpr, Lean4Lean.VEnv, Lean4Lean.VLCtx,
Lean4Lean.VEnv.HasType, Lean4Lean.VEnv.IsDefEqU, Lean4Lean.TrProj}
\leanok
\lead{In short} A source term gets meaning only by \hl{translating into lean4lean's kernel model}.
\begin{itemize}
\item \code{TrExprS}\ $env\ Us\ \Delta\ e\ v$: $e$ translates to the kernel term $v$ (de Bruijn, no
metadata, no literals) at level scope $Us$, local context $\Delta$;
\item $env$ stores a constant table, defining equations and a pattern table, no declaration list,
so every classification below exhibits its own witnessing list and is monotone in $env$;
\item typing $env \vdash v : T$; definitional equality $env \vdash v_1 \equiv v_2$; \code{TrProj}
is the separate projection-clause relation.
\end{itemize}
\lead{Caveat} The pinned fork keeps sixteen \code{sorry} sites; statements routing through
\TrExprS{} or the two judgements inherit them (Chapter~\ref{chap:trust}).
\end{definition}

\begin{remark}
\lead{Caveat} Three capstone binders live entirely in this theory: the subject's translation
(checked at every rung), and the value's translation and typing (both open, Chapter~\ref{chap:trust}).
\end{remark}

\section{Source evaluation: flags, the step obligation, the relation}
\label{sec:source:seval}

\begin{definition}[The source application spine]
\label{def:mkApps}
\lean{LeanToLambdaBox.mkApps, LeanToLambdaBox.mkApps_const_inj}
\leanok
\lead{In short} $\mkApps$ builds the \hl{left-nested application spine} every rule of source
evaluation matches on: $\mkApps\ f\ [a_1,\dots,a_n] = (\dots((f\,a_1)\,a_2)\dots a_n)$, injective
at a constant head (equal spines on $\mathtt{const}\ c\ us$, $\mathtt{const}\ c'\ us'$ force
$c=c'$, $us=us'$, equal arguments).
\lead{Lean} A different declaration from the target-side spine builder on \lbox{} terms
(Definition~\ref{def:LBTerm-mkApps}); the capstone's observable cites both.
\end{definition}

\begin{definition}[Permitted reductions]
\label{def:SEvalFlags}
\lean{LeanToLambdaBox.SEvalFlags, LeanToLambdaBox.fullFlags, LeanToLambdaBox.deltaOnly,
LeanToLambdaBox.SEvalFlags.le_refl, LeanToLambdaBox.SEvalFlags.le_trans,
LeanToLambdaBox.deltaOnly_le_fullFlags}
\leanok
\lead{In short} Six booleans, \code{beta}/\code{delta}/\code{zeta}/\code{iota}/\code{proj}/
\code{lit}, say which reductions a source evaluation may use, \hl{preordered by "enables no more
than"}: reflexive, transitive, \code{deltaOnly} below \code{fullFlags}, the point the capstone and
every rung of Chapter~\ref{chap:capstone} use.
\end{definition}

\begin{definition}[The definitional equality a step owes]
\label{def:StepDefeq}
\lean{LeanToLambdaBox.StepDefeq, LeanToLambdaBox.StepDefeq.le}
\leanok
\uses{def:lean4lean-grounding}
\lead{In short} $\mathsf{StepDefeq}$ is what a source reduction $e_1 \longrightarrow e_2$ owes the
kernel: \hl{both sides translate and the translations are definitionally equal}.
\begin{itemize}
\item $\exists v_1, v_2.\ \TrExprS\ env\ Us\ \Delta\ e_i\ v_i \wedge env \vdash v_1 \equiv v_2$;
\item carried by the delta, iota and projection arms of $\SEval$; monotone in the environment.
\end{itemize}
\lead{Lean} Both translations are bound existentially on purpose: continuing past $e_1$'s step
needs the reduct's own translation, which a universal form cannot supply.
\end{definition}

\begin{lemma}[The existential form entails the universal one]
\label{lem:StepDefeq-uniq_form}
\lean{LeanToLambdaBox.StepDefeq.uniq_form}
\leanok
\uses{def:StepDefeq, def:lean4lean-grounding}
\lead{In short} In a well-formed environment and context, $\mathsf{StepDefeq}\ env\ Us\ \Delta\
e_1\ e_2$ implies \hl{every} pair of translations of $e_1, e_2$ is definitionally equal.
\end{lemma}
\begin{proof}
\leanok
\uses{def:StepDefeq, asm:lean4lean-trust}
\lead{Idea} Unpack the recorded pair; lean4lean's translation-uniqueness identifies any other
translation with it up to definitional equality, then transitivity chains the equalities.
\end{proof}

\begin{lemma}[\dots and the converse fails]
\label{lem:forall_form_not_stepDefeq}
\lean{LeanToLambdaBox.forall_form_not_stepDefeq, LeanToLambdaBox.trExprS_bvar_nil_elim}
\leanok
\uses{def:StepDefeq, def:lean4lean-grounding}
\lead{In short} At $e_1 = \mathtt{bvar}\ 0$ in the empty context the universal side condition
holds \hl{vacuously}, while $\mathsf{StepDefeq}$ does not.
\end{lemma}
\begin{proof}
\leanok
\uses{def:StepDefeq}
\lead{Idea} No arm of the translation relation translates a loose \code{bvar} in the empty
context, so the universal form is vacuous while the existential one needs such a translation.
\end{proof}

\begin{definition}[The eliminator's source-side segmentation]
\label{def:CasesOnShape}
\lean{LeanToLambdaBox.CasesOnShape, LeanToLambdaBox.MajorPremiseAt,
LeanToLambdaBox.CasesOnShape.mono}
\leanok
\uses{def:lean4lean-grounding, def:isCasesOnName}
\lead{In short} $\mathrm{CasesOnShape}\ env\ c\ I\ dp\ nm$ says $c$ is \hl{the \code{casesOn} of
$I$}, at the segmentation $I$'s declaration fixes.
\begin{itemize}
\item $c$ passes the \code{casesOn} name test with prefix $I$, and its type peels $dp$ binders
then takes a major premise headed by $I$ at every level instantiation;
\item a declaration list below $env$ declares a block with $I$ where $dp = np+1+(arity-np)$ and
$nm=|\mathrm{ctors}(I)|$, read off the inductive declaration itself, not the pattern table;
\item monotone (survives environment extension).
\end{itemize}
\end{definition}

\begin{remark}
\lead{Caveat} The block this predicate's existential exhibits matches the iota rule's own block
only together with the block-uniqueness fact of Definition~\ref{def:UpstreamAsks}.
\end{remark}

\begin{definition}[Source evaluation]
\label{def:SEval}
\lean{LeanToLambdaBox.SEval}
\leanok
\uses{def:StepDefeq, def:CasesOnShape, def:mkApps, def:SEvalFlags, def:CtorOf, def:IndInfo,
def:IndArity, def:ConstOrigin, def:InformativeInd, def:lean4lean-grounding}
\lead{In short} $\SEval\ env\ bo\ Us\ fl\ \Delta\ e\ v$ is \hl{weak call-by-value big-step
evaluation} of \code{Lean.Expr}, given the kernel environment $env$, compiler-body table $bo$,
level scope $Us$, permitted reductions $fl$ and local context $\Delta$.

\begin{tabular}{lp{0.38\linewidth}p{0.38\linewidth}}
\toprule
Rule & Fires on & Result / obligation \\
\midrule
\code{lam} & a $\lambda$-abstraction & itself (never enters a binder) \\
\code{sort}, \code{forallE} & a sort, a Pi type & itself \\
\code{ctorVal} & ctor spine, args evaluate, $|args|\le np+nfs[k]$ & itself, at the evaluated args \\
\code{indVal} & inductive type name applied, args evaluate & itself, at the evaluated args \\
\code{beta}$^{fl}$ & $f\,a$, $f\Downarrow\lambda x.b$, $a\Downarrow av$ & $b[av/x]$ evaluates \\
\code{zeta}$^{fl}$ & \code{let} $x{:=}v;b$, $v\Downarrow vv$ & $b[vv/x]$ evaluates \\
\code{deltaC}$^{fl}$ & $c$ tabled, args evaluate, not a \code{casesOn} shape & unfolded body
evaluates; \code{StepDefeq} at the applied instance \\
\code{iota}$^{fl}$ & \code{casesOn} spine, pre-args/discriminant/minors/over-application all
evaluate, discriminant $\Downarrow$ ctor spine, informative $I$ & selected minor on ctor fields
plus over-application evaluates; \code{StepDefeq} \\
\code{proj}$^{fl}$ & discriminee $\Downarrow$ ctor spine of a structure, $np+i$ in range &
field $np+i$ evaluates \\
\code{lit}$^{fl}$ & a literal & unfolds to constructor form; \code{StepDefeq} \\
\bottomrule
\end{tabular}

Superscript $fl$: guarded by the matching flag. Value arms key on the \emph{source theory's}
reading of \code{Expr.const}, never a name test or the compiler table. \code{deltaC}'s eliminator
guard blocks a second derivation unfolding a \code{casesOn} body instead of firing \code{iota};
\code{iota}'s informativity premise restricts to eliminations the target performs; \code{proj}
carries no relevance premise of its own (the erasure relation's does).
\end{definition}

\begin{remark}
\lead{Caveat} Spine arms index with \code{getElem!} (silently defaults out of range). Length
equations keep most reads in range, but \code{ctorVal}'s arity bound reads the field-count list at
the constructor index with no bound on that index.
\end{remark}

\begin{lemma}[Widening along the flags]
\label{lem:SEval-mono}
\lean{LeanToLambdaBox.SEval.mono}
\leanok
\uses{def:SEval, def:SEvalFlags}
\lead{In short} If $fl \le fl'$, every $\SEval$ derivation at $fl$ is one at $fl'$, \hl{same
value}.
\end{lemma}
\begin{proof}
\leanok
\uses{def:SEval}
\lead{Idea} Induction; each arm rebuilds with the same premises, flag guards transported along
$fl\le fl'$.
\end{proof}

\begin{lemma}[Stability under an environment extension]
\label{lem:SEval-le}
\lean{LeanToLambdaBox.SEval.le}
\leanok
\uses{def:SEval, def:CasesOnShape}
\lead{In short} If $env \le env'$ and $env'$ has \hl{no new \code{CasesOnShape}}, every $\SEval$
derivation in $env$ is one in $env'$.
\lead{Caveat} The extra hypothesis matters: the delta arm's eliminator guard is negative, and an
extension could make a constant a \code{casesOn} head it was not.
\end{lemma}
\begin{proof}
\leanok
\uses{def:SEval, def:StepDefeq, def:CasesOnShape}
\lead{Idea} Induction: value arms transport by monotonicity of the readings, stepping arms by
\code{StepDefeq}'s monotonicity, delta's negative guard by contraposing the hypothesis.
\end{proof}

\section{What is not a value}
\label{sec:source:values}

Four results: the value arms exclude a fact about the source theory, not a convention.

\begin{lemma}[A body-less plain constant heads no value]
\label{lem:untabled_const_not_value}
\lean{LeanToLambdaBox.untabled_const_not_value}
\leanok
\uses{def:SEval, def:mkApps, def:CasesOnShape, def:CtorOf, def:IndInfo}
\lead{In short} If $bo\ c=\mathrm{none}$ and $c$ is no constructor, type name or
\code{CasesOnShape}, \hl{no spine on $c$ evaluates at all}, at any flags.
\lead{Why} This is PCUIC's treatment of an axiom, keeping an axiom-freedom premise out of the
simulation.
\end{lemma}
\begin{proof}
\leanok
\uses{def:SEval, def:mkApps}
\lead{Idea} Strong induction on spine length: classification excludes the two value arms, spine
inversion excludes non-constant-headed subjects, beta peels an argument, delta/iota need a tabled
body/a \code{CasesOnShape}.
\end{proof}

\begin{lemma}[An over-applied constructor spine has no value]
\label{lem:overapplied_ctor_not_value}
\lean{LeanToLambdaBox.OverApplied, LeanToLambdaBox.overapplied_ctor_not_value}
\leanok
\uses{def:SEval, def:mkApps, def:CtorOf, def:IndInfo}
\lead{In short} $n$ arguments \hl{over-apply} untabled $c$ (at no \code{CasesOnShape}) when every
constructor reading gives $np+nfs[k] < n$; such a spine has no source value.
\lead{Why} This is what \code{ctorVal}'s arity bound buys: without it the arm would call a stuck
spine a value.
\end{lemma}
\begin{proof}
\leanok
\uses{def:SEval, lem:CtorOf-not_indInfo}
\lead{Idea} The constructor arm contradicts over-application directly; the type-former arm is
excluded since a constructor name is not a type name (Lemma~\ref{lem:CtorOf-not_indInfo}); the
rest as in the previous lemma.
\end{proof}

\begin{remark}
\lead{Caveat} Over-application quantifies over \emph{every} classification, since constructor and
block readings may each read a different declaration list below $env$;
Definition~\ref{def:UpstreamAsks}'s uniqueness would collapse this to one.
\end{remark}

\begin{lemma}[A constructor is not a type name]
\label{lem:CtorOf-not_indInfo}
\lean{LeanToLambdaBox.CtorOf.not_indInfo, LeanToLambdaBox.CtorOf.constant_ctorResult,
LeanToLambdaBox.IsArity.piBody_sort}
\leanok
\uses{def:CtorOf, def:IndInfo, def:IsArity}
\lead{In short} A constructor of some inductive in the model is \hl{never also} a type name of
that model.
\end{lemma}
\begin{proof}
\leanok
\uses{def:CtorOf, def:IndInfo}
\lead{Idea} A constructor's declared type is headed by its type former; a type former's declared
type is an arity whose Pi-body is a sort -- not a constructor spine.
\end{proof}

\begin{remark}
\lead{Why} Discharges, in this repository, the first conjunct of
Definition~\ref{def:UpstreamAsks}'s origin field -- redundant by that field's own account, kept so
the field grants everything the upstream ask covers.
\end{remark}

\begin{lemma}[A stuck head stays a head]
\label{lem:seval_const_head_value}
\lean{LeanToLambdaBox.seval_const_head_value, LeanToLambdaBox.SEval.forallE_inv}
\leanok
\uses{def:SEval, def:mkApps, def:CasesOnShape}
\lead{In short} Where delta cannot move ($bo\ c=\mathrm{none}$) and iota cannot fire (no
\code{CasesOnShape}), \hl{the value of a spine on $c$ is a spine on $c$} -- never a lambda; a Pi
type's only value is itself.
\end{lemma}
\begin{proof}
\leanok
\uses{def:SEval, def:mkApps}
\lead{Idea} Strong induction on spine length, generalised over the subject: value arms return a
spine on the same head, beta peels an argument, the rest excluded by the two hypotheses or shape.
\end{proof}

\begin{remark}
\lead{Why} The Pi-type inversion makes the \code{forallE} value arm non-optional: without it a
delta redex with a Pi-typed argument has no derivation.
\end{remark}

\begin{definition}[The compiler-body table's typing hypothesis]
\label{def:CompilerBodies}
\lean{LeanToLambdaBox.CompilerBodies, LeanToLambdaBox.CompilerBodies.le}
\leanok
\uses{def:lean4lean-grounding}
\lead{In short} $\mathrm{CompilerBodies}\ lenv\ env\ bo$ says every tabled body is \hl{kernel-
typeable at its constant's declared type}.
\begin{itemize}
\item for $c$ with $bo\ c=\mathrm{some}\ b$: $c$ declared in $lenv$; $b$ and $c$'s declared type
both translate; the translated body has the translated type at $c$'s level scope, empty context;
\item monotone; checkable per program by running lean4lean's checker.
\end{itemize}
\end{definition}

\begin{remark}
\lead{Caveat} A hypothesis, never proved here: capstone binder \code{hcb}, discharged only at the
first rung.
\begin{itemize}
\item blocked elsewhere: a projection node's translation routes through lean4lean's unproven
projection relation;
\item the eraser's actual input is the compiler-facing body, after \code{csimp} and
\code{@[extern]}/\code{@[implemented\_by]} substitution, not the kernel-checked one
(Chapter~\ref{chap:trust}, \code{hcb}).
\end{itemize}
\end{remark}

\section{Three environment fixtures}
\label{sec:source:witnesses}

Three fixtures hand-build kernel environments, showing the arms' side conditions are jointly
satisfiable. Only the three results below matter to a reader.

\begin{proposition}[A tabled body that is not the kernel body]
\label{prop:DeltaWitness-seval_sel_applied}
\lean{LeanToLambdaBox.DeltaWitness.seval_sel_applied,
LeanToLambdaBox.DeltaWitness.seval_add_deltaOnly,
LeanToLambdaBox.DeltaWitness.seval_add_applied,
LeanToLambdaBox.DeltaWitness.selBody_ne_addBody}
\leanok
\uses{def:SEval, def:mkApps, def:SEvalFlags, def:StepDefeq}
\lead{In short} In a two-constant fixture, one tabled body \emph{is} its kernel body and the other
is \hl{not} -- yet the second's applied instance still unfolds and beta-reduces to a sort, with
\code{StepDefeq} discharged (both bodies compute to the same value there).
\end{proposition}
\begin{proof}
\leanok
\uses{def:SEval, def:StepDefeq}
\lead{Idea} Apply the delta arm (the eliminator guard is a name test here); arguments evaluate by
the sort/Pi value arms; \code{StepDefeq} chains the defining equation with both bodies' applied
computations; two beta steps finish.
\end{proof}

\begin{remark}
\lead{Caveat} Shows the delta arm models compiler bodies differing from kernel bodies. Its
environment is never shown well-formed: read as "jointly satisfiable side conditions", not "fires
in a well-formed environment".
\end{remark}

\begin{proposition}[The branch rule fires]
\label{prop:NatWitness-seval_iota_fires}
\lean{LeanToLambdaBox.NatWitness.seval_iota_fires,
LeanToLambdaBox.NatWitness.seval_iota_overapplied_fires,
LeanToLambdaBox.NatWitness.nat_casesOnShape, LeanToLambdaBox.NatWitness.natDecl_wf,
LeanToLambdaBox.NatWitness.nat_wf'_cas, LeanToLambdaBox.NatWitness.nat_stuck_not_value,
LeanToLambdaBox.NatWitness.natEnv, LeanToLambdaBox.NatWitness.nat_indInfo}
\leanok
\uses{def:SEval, def:CasesOnShape, def:SEvalFlags, def:mkApps, def:IndInfo}
\lead{In short} A hand-built, well-formed \code{Nat} block supports a \code{CasesOnShape} ("one
argument before the major premise, one minor per constructor"), and the \hl{iota arm fires} on it
both saturated and one argument past the emitted node -- the shape the eraser applies outside it.
The same fixture's body-less constant heads no value.
\lead{Lean} At the full flag point.
\end{proposition}
\begin{proof}
\leanok
\uses{def:SEval, def:StepDefeq, lem:untabled_const_not_value}
\lead{Idea} Instantiate the iota arm with the fixture's classification facts and its
\code{CasesOnShape}; pre-argument, discriminant and minors evaluate by the value arms; the step's
equality is a hand-recorded conversion; a beta step finishes.
\end{proof}

\begin{remark}
\lead{Caveat} Staging puts the major premise last, so the environment is the staged one
\emph{extended} with two hand-written defining equations; well-formedness is proved only for the
staged environment. Reused: a capstone rung inhabits its spike-facts bundle from six of its facts,
and \code{natSpecEnv} (Proposition~\ref{prop:natEnv_elimCovered}) builds the \lbox{} side.
\end{remark}

\begin{proposition}[The projection rule fires]
\label{prop:ProjWitness-seval_proj_fires}
\lean{LeanToLambdaBox.ProjWitness.seval_proj_fires, LeanToLambdaBox.ProjWitness.p_projStep,
LeanToLambdaBox.ProjWitness.p_trProj, LeanToLambdaBox.ProjWitness.pDecl_wf,
LeanToLambdaBox.ProjWitness.p_wf'}
\leanok
\uses{def:SEval, def:SEvalFlags, def:lean4lean-grounding}
\lead{In short} A hand-built two-field, one-constructor structure, staged so its iota rule
registers as a translated environment would, supports the \hl{projection arm} at the full flag
point: the second field is selected, \code{StepDefeq} read off a recorded defining equation, the
translation's projection premise supplied at the projection's expansion.
\end{proposition}
\begin{proof}
\leanok
\uses{def:SEval, def:StepDefeq}
\lead{Idea} The discriminee evaluates to the constructor spine; the fixture's constructor/arity
facts fix head and parameter count; the field index is in range; the obligation is the recorded
projection step.
\end{proof}

\begin{remark}
\lead{Caveat} Same caveat as \code{Nat}. At a projection the exposure to lean4lean is
\emph{inhabitation}, not axioms: a real projection's translation exists only through a lemma that
is \code{sorry} at the pin, so this arm is non-vacuous only on hand-built witnesses.
\end{remark}

\begin{definition}[Manifest lambda-telescopes]
\label{def:IsLamTelescope}
\lean{LeanToLambdaBox.IsLamTelescope, LeanToLambdaBox.isLamTelescopeB,
LeanToLambdaBox.isLamTelescopeB_iff, LeanToLambdaBox.IsLamTelescope.instantiate1}
\leanok
\lead{In short} $e$ is a manifest lambda-telescope of depth $\ge n$ when it is syntactically
\hl{$n$ nested lambdas} -- needed because the eraser's intro branch eta-expands a non-lambda minor
and the relation has no eta rule.
\begin{itemize}
\item Lean's \code{match} compiler emits minors as explicit \code{fun a b => ...}, so real

\item decidable; survives opening a binder with a free variable.

\end{definition}

\section{The fragment checker and its soundness}
\label{sec:source:checker}

\begin{definition}[The fragment checker]
\label{def:supportedB}
\lean{LeanToLambdaBox.supportedB, LeanToLambdaBox.supportedHead, LeanToLambdaBox.supportedGo,
LeanToLambdaBox.supportedTerm, LeanToLambdaBox.constNames, LeanToLambdaBox.stepNames,
LeanToLambdaBox.reachNames, LeanToLambdaBox.saturatedB, LeanToLambdaBox.checkNames,
LeanToLambdaBox.kernameSepB, LeanToLambdaBox.informativeB, LeanToLambdaBox.resultSort,
LeanToLambdaBox.succSortB}
\leanok
\uses{def:SupportError, def:IsLamTelescope, def:CtorSaturated, def:Witness-SourceTable}
\lead{In short} $\mathrm{supportedB}\ tbl\ \mathit{fuel}\ e$ decides fragment membership: \hl{total
and fuel-indexed}. Decides key separation, shape-checks $e$, then unfolds the delta closure of
$e$'s constants \emph{fuel} times; once saturated it shape-checks every tabled body, else reports
exhaustion (an error, never a certificate). The head check runs the four plainness tests, then
either the eliminator branch (tabled inductive, informative, arity bound, one telescope per minor)
or the constant branch (saturation, no recursor head, a known column).
\lead{Why} The dependency closure is cyclic through mutual blocks, not structural in $e$; a
\code{partial def} would be kernel-opaque, which a kernel-level \code{decide} cannot afford.
\end{definition}

\begin{remark}
\lead{Caveat} The Boolean relevance test (result sort never \code{Prop}) twins the relation's
informativity predicate. The stronger syntactic successor-sort test, read only by the first-order
layer, would reject 9 of 21 benchmark projection heads. Two \code{decide} examples check
\code{sparseCasesOn} naming and live projection informativity.
\end{remark}

\begin{lemma}[The checker's relevance test is sound]
\label{lem:informativeInd_of_tabled}
\lean{LeanToLambdaBox.informativeInd_of_tabled, LeanToLambdaBox.vResultSort_of_trExprS,
LeanToLambdaBox.ErasureSpec.contains_of_find}
\leanok
\uses{def:TableSafe, def:supportedB, def:ErasureSpec, def:Witness-SourceTableAdequate,
def:InformativeInd}
\lead{In short} Given an adequate, safe table connected to the model by the erasure specification,
a tabled inductive the checker calls informative \hl{is informative in the model}.
\end{lemma}
\begin{proof}
\leanok
\uses{def:TableSafe, asm:ErasureSpec-decl_adequate}
\lead{Idea} Safety lets declaration-adequacy translate the pinned declared type; the level
translation is a homomorphism, carrying lean4lean's never-zero lemma across it.
\end{proof}

\begin{lemma}[A tabled inductive is the model's type former]
\label{lem:indInfo_of_tabled}
\lean{LeanToLambdaBox.indInfo_of_tabled, LeanToLambdaBox.peanoReady_of_tabled}
\leanok
\uses{def:IndInfo, def:ErasureSpec, def:Witness-SourceTableAdequate}
\lead{In short} A tabled inductive \hl{is} the model's type former for its block, at the table's
parameter and field counts; the model can build a peano tower whenever the (additionally safe)
table can.
\end{lemma}
\begin{proof}
\leanok
\uses{asm:ErasureSpec-block_adequate}
\lead{Idea} Table adequacy names the block and the kernel's constructor indexing; block-adequacy
reads that as the model's block data.
\end{proof}

\begin{theorem}[Soundness of the fragment checker]
\label{thm:supportedB_sound}
\lean{LeanToLambdaBox.supportedB_sound, LeanToLambdaBox.supportedHead_sound,
LeanToLambdaBox.supportedGo_sound, LeanToLambdaBox.checkNames_sound,
LeanToLambdaBox.mem_of_reaches, LeanToLambdaBox.kernames_of_kernameSepB}
\leanok
\uses{def:Supported, def:supportedB, def:TableSafe, def:ErasureSpec,
def:Witness-SourceTableAdequate}
\lead{In short} Given a table that is an adequate, safe copy of the elaboration environment's
slice, an \code{ok} verdict of $\mathrm{supportedB}\ tbl\ \mathit{fuel}\ e$ (any fuel) implies
$\mathrm{Supported}\ env\ tbl\ e$: \hl{the subject and every delta-reachable tabled body is inside
the fragment}.
\lead{Status} \stProved{}.
\end{theorem}
\begin{proof}
\leanok
\uses{def:Supported, def:SupportedTm, def:supportedB, def:KnownHead, def:PlainHead,
def:CtorSaturated, def:IsLamTelescope, lem:informativeInd_of_tabled, lem:indInfo_of_tabled}
\lead{Idea} Structural induction on \code{Lean.Expr} mirroring the checker's shape function.
\begin{enumerate}
\item Key separation: list combinatorics on the deduplicated key list.
\item Constant case: plainness from the four negative tests.
\item Eliminator arm: Lemma~\ref{lem:informativeInd_of_tabled} for informativity, telescope
correspondence for the minors.
\item Constant arm: the known-head column from Lemma~\ref{lem:indInfo_of_tabled} or from a safe
declaration being a model constant.
\item Bodies: a saturated closure list holding the subject's constants holds everything
delta-reachable, and every checked body passed the shape check.
\end{enumerate}
\end{proof}

\begin{remark}
\lead{Why} The hinge from a computed verdict to the capstone's coverage hypothesis: every rung of
Chapter~\ref{chap:capstone} discharges it by applying this theorem to a \code{decide}-style
computation on the committed reified table.
\end{remark}

\begin{lemma}[The fragment is closed under opening a binder]
\label{lem:SupportedTm-instantiate1}
\lean{LeanToLambdaBox.SupportedTm.instantiate1, LeanToLambdaBox.SupportedTm.instantiate1'}
\leanok
\uses{def:SupportedTm}
\lead{In short} If a term is inside the fragment under a spine, so is the term with its
\hl{outermost bound variable instantiated} by a free variable, spine instantiated too.
\end{lemma}
\begin{proof}
\leanok
\uses{def:SupportedTm, asm:lean-reflection-axioms}
\lead{Idea} Induction: every arm's side conditions concern names, tabled data or the model,
untouched by substitution; transport to Lean's own instantiation is a rewrite by the modelling
equation (source of the reflection-axiom footprint).
\end{proof}

\begin{remark}
\lead{Why} The form the refinement's binder cases recurse in: the eraser's lambda and \code{let}
steps call the continuation on the body instantiated with a fresh free variable.
\end{remark}

\begin{lemma}[Reading the fragment back]
\label{lem:Supported-projInfo}
\lean{LeanToLambdaBox.Supported.projInfo, LeanToLambdaBox.SupportedTm.proj_inv,
LeanToLambdaBox.Supported.projInto, LeanToLambdaBox.Supported.projInd,
LeanToLambdaBox.SupportedTm.head, LeanToLambdaBox.Supported.head}
\leanok
\uses{def:SupportedTm, def:Supported, def:ProjInfo, def:IndArity}
\lead{In short} Two read-backs: a fragment term's projection head carries its \hl{block data}

\end{lemma}
\begin{proof}
\leanok
\uses{def:SupportedTm}
\lead{Idea} Induction on the shape derivation: the projection arm's block-data premise yields the

\end{proof}

\lead{Caveat} \code{Supported.projInfo} takes a shape derivation, not a fragment verdict, despite

\end{remark}

\end{remark}

\section*{Deviations from the paper and caveats}

\begin{itemize}
\item \lead{Deviation} No source semantics to cite, so this chapter builds one: $\SEval$ is a
sub-relation of the kernel's own convertibility, evaluation order explicit, every non-value arm
paying a definitional-equality obligation.
\item \lead{Deviation} Delta is not kernel delta: the eraser reads compiler bodies after
\code{csimp}/\code{@[extern]}/\code{@[implemented\_by]}, paying \code{StepDefeq} at the applied
instance; \code{CompilerBodies} is discharged only at the first rung.
\item \lead{Deviation} The coverage predicate has no counterpart in [S]: MetaCoq's erasure is total
on well-typed PCUIC, but this eraser ships and partly panics; only Arith is inside the fragment.
\item \lead{Caveat} The recursor-head exclusion guards against vacuous coverage, since a recursor
spine has no source evaluation. \code{@[implemented\_by]} is closed by \code{ConfigPinned}, not
the fragment; it differs from the default \code{\#erase} configuration on three of five fields.
\item \lead{Why} Specification environments are \code{erases\_global} \emph{derived from a run},
not posited: the eraser registers constants and blocks as it discovers them.
\item \lead{Caveat} Two of three fixture environments are proved well-formed only before their
hand-added equations; the delta fixture's is never shown well-formed at all: read all three as
jointly satisfiable side conditions, not as firing in a well-formed environment.
\end{itemize}
