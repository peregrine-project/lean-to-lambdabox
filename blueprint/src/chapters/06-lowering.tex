\section{The pass relation \texorpdfstring{$\Lower$}{Lower}}
\label{sec:lowering:pass}

\begin{definition}[Environment queries of the pass]
\label{def:ElimDecl}
\lean{LeanToLambdaBox.DefnDecl, LeanToLambdaBox.IndBodyOf, LeanToLambdaBox.ElimDecl,
LeanToLambdaBox.RuntimeKey, LeanToLambdaBox.ClosedBodies, LeanToLambdaBox.FVarFreeBodies}
\leanok
\uses{def:ElimBody, def:GlobalDeclarations, def:LBTerm, def:Kername, def:InductiveId,
def:OneInductiveBody, def:LBTerm-envLookup, def:LBClosed, def:hasFVar}
\lead{In short} Six predicates on a \emph{specification} environment $\Gamma$ -- the only thing about
the environment \hl{the pass ever reads}.
\begin{itemize}
\item \code{DefnDecl}\ \Gamma\ kn\ b: $kn$ is declared in $\Gamma$ with body $b$ (a definitional
equation on \code{LBTerm.envLookup}, not an inductive characterisation);
\item \code{IndBodyOf}\ iid\ np\ nfs\ mib: block $mib$ has $np$ parameters, $iid$ selects a
non-propositional body with constructor field counts $nfs$;
\item \code{ElimDecl}\ \Gamma\ kn\ iid\ np\ dp\ nfs: $kn$ is an \emph{eliminator constant with its
block} -- body one of the two canonical shapes (Definition~\ref{def:ElimBody}), block data matching;
\item \code{RuntimeKey}\ \Gamma\ kn: the existential closure of \code{ElimDecl} -- a key the pass
\emph{consumes} into \code{case} nodes, its declaration pruned; a constructor is never a runtime key;
\item \code{ClosedBodies}, \code{FVarFreeBodies}\ \Gamma: every declared body is closed, resp.\
variable-free -- independent, since closedness says nothing at an \code{fvar} node.
\end{itemize}
\lead{Since the snapshot} \stLanded{} \code{ec84ccf}: \code{IndBodyOf} drops its
``\code{propositional} is \code{false}'' conjunct, now carried beside the eliminator as
\code{IndNotPropositional}, because the shipping eraser sets the flag (F-PROP, \code{d28459f}).
\end{definition}

\begin{lemma}[Eliminator data are determined by the key]
\label{lem:ElimDecl-uniq}
\lean{LeanToLambdaBox.ElimDecl.uniq}
\leanok
\uses{def:ElimDecl}
\lead{In short} Two eliminator declarations at the same key \hl{agree on all four data}.
\begin{itemize}
\item \lead{Given} $\code{ElimDecl}\ \Gamma\ kn\ iid\ np\ dp\ nfs$ and
$\code{ElimDecl}\ \Gamma\ kn\ iid'\ np'\ dp'\ nfs'$;
\item \lead{Then} $iid = iid'$, $np = np'$, $dp = dp'$, $nfs = nfs'$.
\end{itemize}
\lead{Lean} The conclusion is a fourfold conjunction, not an equality of the whole
datum.
\end{lemma}
\begin{proof}
\leanok
\uses{def:ElimDecl, def:ElimBody, def:mkElimBody}
\lead{Idea} Environment lookup is a function, so both readings name one body; the two eliminator shapes
are each injective in $(iid, np, dp, nfs)$ and never equal to each other.
\end{proof}

\begin{definition}[Block rewriting and block closure]
\label{def:ConstToFVar}
\lean{LeanToLambdaBox.ConstToFVar, LeanToLambdaBox.CloseConstAt}
\leanok
\uses{def:LBTerm, def:Kername, def:closeFix}
\lead{In short} \code{ConstToFVar} rewrites a block's own constants to its fix variables;
\code{CloseConstAt} \hl{closes one body} with that rewriting plus \code{closeFix}.
\begin{itemize}
\item \code{ConstToFVar}\ kns\ ids: $\code{const}\ kns[j] \mapsto \code{fvar}\ ids[j]$
(\code{hit}), else a constant to itself, else a congruence with binder names free; a \code{fix} node
maps to itself, sound since the source declares no block member as \code{fix};
\item \code{CloseConstAt}\ kns\ ids\ t\ u: $\code{closeFix}\ ids\ 0$ applied to the
\code{ConstToFVar} image of $t$ -- the fold the eraser performs when storing a block.
\end{itemize}
\end{definition}

\begin{definition}[The pass relation]
\label{def:Lower}
\lean{LeanToLambdaBox.Lower, LeanToLambdaBox.LowerAlt, LeanToLambdaBox.LowerAlts}
\leanok
\uses{def:ElimDecl, def:ConstToFVar, def:LBTerm, def:LBTerm-mkApps, def:FixDef,
def:BinderName, def:GlobalDeclarations}
\lead{In short} $\Lower\ \Gamma\ t\ t'$: $t'$ is a \lbox{} term \hl{the shipping eraser may emit}
for $t$ over $\Gamma$ -- fourteen arms, indexed by $\Gamma$ alone.
\begin{center}
\begin{tabular}{p{0.27\linewidth}p{0.46\linewidth}p{0.18\linewidth}}
\toprule
\textbf{Arm} & \textbf{Source shape / rule} & \textbf{MetaRocq counterpart} \\
\midrule
\code{box}, \code{bvar}, \code{fvar}, \code{prim} & atom to itself & \code{optimize} \\
\code{const} & non-runtime-key constant to itself & \code{optimize} \\
\code{lambda}, \code{letIn}, \code{app}, \code{proj}, \code{construct} & congruence, names free &
\code{optimize} \\
\code{case} & discriminant \& branches, arities pinned & \code{iota\_red} \\
\code{elimApp} & eliminator spine to \code{case} spine, $dp$ leading args dropped, minors peeled by
\code{LowerAlt} & \code{iota\_red} \\
\code{fixConst} & a block member's constant to the block's \code{fix} node & \code{fixSubst} \\
\code{fixBody} & a block member's specification body to the same \code{fix} node & \code{fixSubst} \\
\bottomrule
\end{tabular}
\end{center}
\lead{Lean} list premises are indexed (length plus bounded quantifier): a \code{List.Forall2} premise
nests an inductive occurrence the kernel rejects.
\lead{Since the snapshot} \stLanded{} \code{ec84ccf} adds a fifteenth arm, \code{fixEta}: a block
member's body relates to the eta-expansion of the block's node, the shape \code{visitMutual} now
registers (F-ETA, \code{42989c5}). Its target binder name must stay free, or \code{Lower.constToFix}
fails -- mechanised before the arm landed.
\end{definition}

\begin{remark}
\lead{Caveat} Four things differ from [S]'s reader expectations.
\begin{itemize}
\item \textbf{non-deterministic} at a block member (\code{const}, \code{fixConst} both apply): read
``emits $t'$'' as ``there is an emission $t'$'';
\item no constructor-introduction or eta-expansion arm (two once existed, refuted; now finding F-ETA2);
\item no \code{fix} congruence arm (Proposition~\ref{prop:Lower-source_const});
\item \code{elimApp} drops arguments before the discriminant with no premise: sound only forward, not a
bisimulation.
\end{itemize}
\end{remark}

\begin{definition}[The block hypothesis bundle]
\label{def:LowerBlock}
\lean{LeanToLambdaBox.LowerBlock}
\leanok
\uses{def:Lower, def:ElimDecl, def:ConstToFVar, def:FixDef, def:hasFVar}
\lead{In short} The premises the two recursion arms share, packaged as a \hl{twelve-field structure}.
\begin{itemize}
\item $kns$, $bs$, $bs'$, $defs$: the members' (distinct) kernames, their specification bodies (declared
in $\Gamma$), their pass images, and the emitted definitions -- each body the block closure of the
matching $bs'$ member; $ids$: one \textbf{block-shared} list of fresh fix variables (one per block, not
per member);
\item \code{hfresh}: no fix variable already occurs in a lowered body;
\item \code{hrarg}: every emitted principal-argument index is $0$ -- the only value the emitter
produces, and where one step consumes exactly one argument;
\item \code{hfl}: every emitted definition body is lambda-headed -- the \code{fixLambda} clause of
Definition~\ref{def:LBWfPeregrine}, at this one block.
\end{itemize}
\end{definition}

\begin{remark}
\lead{Status} A hypothesis bundle, not a proved fact: wherever it appears as a hypothesis, each field
is an assumed premise.
\lead{Non-vacuity} Lemma~\ref{lem:LowerFixFixture-lowerfix_app_step} inhabits all twelve fields at a
genuine two-member mutual block; \code{hrarg} shows the emitter's own comment -- calling the
principal-argument index irrelevant -- wrong under this evaluation relation.
\end{remark}

\begin{lemma}[The three arms in packaged form]
\label{lem:Lower-fixConst-prime}
\lean{LeanToLambdaBox.Lower.fixConst', LeanToLambdaBox.Lower.fixBody',
LeanToLambdaBox.Lower.elimApp'}
\leanok
\uses{def:Lower, def:LowerBlock, def:ElimDecl}
\lead{In short} \code{fixConst}, \code{fixBody} and \code{elimApp}, re-exposed with their premises
packaged through \hl{\code{LowerBlock}} and \code{LowerAlts} rather than inlined; every downstream
fixpoint or eliminator derivation goes through these.
\lead{Lean} The inlining they undo
is forced by the Lean kernel, which rejects a structure premise mentioning the inductive being defined.
\end{lemma}
\begin{proof}
\leanok
\uses{def:Lower, def:LowerBlock}
\lead{Idea} Each is the corresponding constructor applied to the bundle's projections.
\end{proof}

\section{Metatheory of the pass}
\label{sec:lowering:meta}

\begin{lemma}[A block's node is closed and variable-free]
\label{lem:lbClosed_fix_of_block}
\lean{LeanToLambdaBox.lbClosed_closeFix, LeanToLambdaBox.ConstToFVar.closed,
LeanToLambdaBox.lbClosed_fix_of_block, LeanToLambdaBox.constToFVar_not_hasFVar,
LeanToLambdaBox.closeConstAt_not_hasFVar, LeanToLambdaBox.fixNode_not_hasFVar,
LeanToLambdaBox.toBvar_fixNode}
\leanok
\uses{def:ConstToFVar, def:LBClosed, def:closeFix, def:hasFVar, def:toBvar}
\lead{In short} The \code{fix} node a block builds is \hl{inert}: closed at every bound and free of
every variable, once its lowered bodies are.
\begin{itemize}
\item closing at base $0$ raises closedness to the block's width (neither \code{const} nor \code{fvar}
binds an index);
\item block closure of a variable-free body is variable-free at \emph{every} variable, including the
block's own -- so abstraction is the identity on the \code{fix} node.
\end{itemize}
\end{lemma}
\begin{proof}
\leanok
\uses{def:ConstToFVar, def:closeFix, lem:toBvar_eq_of_not_hasFVar, lem:closeFixFold_fvar_head}
\lead{Idea} Induction over the rewriting relation; read each definition body as a \code{closeFix}
image and observe that closing at base $0$ removes exactly the scheduled identifiers.
\end{proof}

\begin{proposition}[The pass invents no de Bruijn index]
\label{prop:Lower-closed}
\lean{LeanToLambdaBox.Lower.closed}
\leanok
\uses{def:Lower, def:ElimDecl, def:LBClosed}
\lead{In short} Over an environment with closed bodies, $\Lower$ \hl{preserves closedness} at every
bound.
\begin{itemize}
\item \lead{Given} every constant declared in $\Gamma$ has a closed body, and $\Lower\ \Gamma\ s\ t$;
\item \lead{Then} $\code{LBClosed}\ s\ k$ implies $\code{LBClosed}\ t\ k$, for every bound $k$.
\end{itemize}
\lead{Lean} the bound $k$ is quantified inside the conclusion, so induction can raise it under binders.
\end{proposition}
\begin{proof}
\leanok
\uses{def:Lower, lem:lbClosed_fix_of_block, lem:LBClosed-substList}
\lead{Idea} Induction with the mutual recursor; no congruence arm invents an index, and the two
recursion arms read $\Gamma$ and discharge through Lemma~\ref{lem:lbClosed_fix_of_block}.
\end{proof}

\begin{proposition}[The pass commutes with shifting and substitution]
\label{prop:Lower-subst_comm}
\lean{LeanToLambdaBox.Lower.shift_comm, LeanToLambdaBox.Lower.subst_comm,
LeanToLambdaBox.Lower.substList_comm}
\leanok
\uses{def:Lower, def:ElimDecl, def:LBTerm-shift, def:LBTerm-subst}
\lead{In short} Over an environment with closed bodies, the pass commutes with de Bruijn shifting, with
substitution of related terms, and with a whole simultaneous substitution of pointwise related terms --
\hl{the laws a forward simulation spends} at beta, zeta and iota.
\lead{Lean} The
substitution laws require the substituends themselves to be related, not merely closed.
\end{proposition}
\begin{proof}
\leanok
\uses{def:Lower, prop:Lower-closed, lem:lbClosed_fix_of_block}
\lead{Idea} Induction on the derivation: congruence arms are immediate since the pass reads no index;
the two recursion arms are closed (Lemma~\ref{lem:lbClosed_fix_of_block}), so shift and substitution act
as the identity there; the list form iterates the single form.
\end{proof}

\begin{proposition}[The pass commutes with binder abstraction]
\label{prop:Lower-abstract}
\lean{LeanToLambdaBox.Lower.noFVar, LeanToLambdaBox.Lower.abstract,
LeanToLambdaBox.LowerAlt.abstract}
\leanok
\uses{def:Lower, def:ElimDecl, def:toBvar, def:hasFVar}
\lead{In short} Over an environment whose declared bodies have \hl{no free variable}, the pass
introduces no free variable and commutes with abstraction at any level.
\begin{itemize}
\item \lead{Given} $\Lower\ \Gamma\ s\ t$, every declared body of $\Gamma$ variable-free;
\item \lead{Then} $\Lower\ \Gamma\ (\code{toBvar}\ x\ lvl\ s)\ (\code{toBvar}\ x\ lvl\ t)$; the
alternative form shifts the level by the branch's field count.
\end{itemize}
\end{proposition}
\begin{proof}
\leanok
\uses{def:Lower, lem:lbClosed_fix_of_block}
\lead{Idea} Variable-freedom comes first, independently, via Lemma~\ref{lem:lbClosed_fix_of_block}; the
abstraction law then needs no cutoff or closedness premise, since abstraction never touches an existing
index; the alternative form follows by induction on the field count.
\end{proof}

\begin{lemma}[Closedness does not replace variable-freedom]
\label{lem:FVarFixture-noFVar_needs_fvarFree}
\lean{LeanToLambdaBox.FVarFixture.noFVar_needs_fvarFree,
LeanToLambdaBox.FVarFixture.abstract_needs_fvarFree,
LeanToLambdaBox.FVarFixture.closed_cxEnv, LeanToLambdaBox.FVarFixture.not_fvarFree,
LeanToLambdaBox.FVarFixture.blk, LeanToLambdaBox.FVarFixture.lower_const}
\leanok
\uses{def:Lower, def:ElimDecl, def:hasFVar}
\lead{In short} Proposition~\ref{prop:Lower-abstract} is \hl{false} if variable-freedom is weakened to
closedness.
\begin{itemize}
\item \lead{Given} a one-member block whose \emph{declared} body is a lambda over a stray free variable
-- \code{ClosedBodies} holds (closedness says nothing at an \code{fvar} node) but not
\code{FVarFreeBodies};
\item \lead{Then} the \code{fixConst} arm relates the variable-free \code{const} to a \code{fix} node
carrying the declared body's stray variable, refuting both weakened laws.
\end{itemize}
\end{lemma}
\begin{proof}
\leanok
\uses{def:LowerBlock, lem:Lower-fixConst-prime, prop:Lower-closed}
\lead{Idea} Exhibit the one-entry environment and one-member block; \code{fixConst} fires, refuting the
variable statement; abstracting the stray variable under the block's binders would manufacture an
out-of-range index, which closedness preservation rules out.
\end{proof}

\begin{lemma}[Target-side inversion and spine congruence]
\label{lem:Lower-target_fix}
\lean{LeanToLambdaBox.Lower.target_box, LeanToLambdaBox.Lower.target_bvar,
LeanToLambdaBox.Lower.target_fvar, LeanToLambdaBox.Lower.target_prim,
LeanToLambdaBox.Lower.target_const, LeanToLambdaBox.Lower.target_fix,
LeanToLambdaBox.Lower.target_construct, LeanToLambdaBox.Lower.mkApps}
\leanok
\uses{def:Lower, def:LowerBlock, def:ElimDecl, def:LBTerm-mkApps}
\lead{In short} What a target of a given shape can \hl{only} have come from.
\begin{itemize}
\item an atom, a constant or a constructor node comes from the same shape, argument by argument (no arm
introduces, erases, renumbers or renames one; a target constant is not a runtime key);
\item a \code{fix} node comes from one of the two recursion arms, carrying a whole \code{LowerBlock};
the relation lifts from a head and its arguments to a whole spine.
\end{itemize}
\lead{Lean} each takes the target shape as a separate hypothesis, so it applies before the target is
known.
\end{lemma}
\begin{proof}
\leanok
\uses{def:Lower, def:LowerBlock}
\lead{Idea} Case analysis on the derivation with the target shape fixed; the surviving arm is the
congruence of the same node, or (at a \code{fix} target) one of the two recursion arms with its whole
bundle; the spine congruence is an induction on the argument list through \code{app}.
\end{proof}

\begin{proposition}[A block member's specification body is a lambda]
\label{prop:LowerBlock-lambda_of_fixLambda}
\lean{LeanToLambdaBox.LowerBlock.lambda_of_fixLambda,

\leanok
\uses{def:LowerBlock, def:Lower, def:atomValue}
\lead{In short} Unconditionally, a block member's specification body is \hl{lambda-headed} -- no

\lead{Lean} stated through the Boolean head test \code{isLambda} rather than an existential.
\end{proposition}
\begin{proof}
\leanok
\uses{def:LowerBlock, def:Lower, def:ConstToFVar, def:closeFix, def:toBvar}
\lead{Idea} Abstraction, block-closing and the constant-to-fixvar rewriting are each faithful on the

\end{proof}

\begin{lemma}[Only a constant or a lambda has a block image]
\label{lem:Lower-ne_fix_of_block}
\lean{LeanToLambdaBox.Lower.notFix_of_block, LeanToLambdaBox.Lower.ne_fix_of_block}
\leanok
\uses{def:Lower, def:atomValue}
\lead{In short} If $\Lower\ \Gamma\ s\ (\code{fix}\ defs\ j)$ then $s$ is a constant or

\end{lemma}
\begin{proof}
\leanok
\uses{lem:Lower-target_fix, prop:LowerBlock-lambda_of_fixLambda}
\lead{Idea} Invert the target (Lemma~\ref{lem:Lower-target_fix}): \code{fixConst} gives a constant,
\code{fixBody} gives a declared body, pinned to a lambda by

\end{proof}

\begin{proposition}[Source-side inversion]
\label{prop:Lower-source_const}
\lean{LeanToLambdaBox.Lower.source_box, LeanToLambdaBox.Lower.source_bvar,
LeanToLambdaBox.Lower.source_fvar, LeanToLambdaBox.Lower.source_prim,
LeanToLambdaBox.Lower.source_letIn, LeanToLambdaBox.Lower.source_proj,
LeanToLambdaBox.Lower.source_construct, LeanToLambdaBox.Lower.source_case,
LeanToLambdaBox.Lower.source_fix, LeanToLambdaBox.Lower.source_lambda,
LeanToLambdaBox.Lower.source_const, LeanToLambdaBox.Lower.source_construct_nil,
LeanToLambdaBox.LowerAlt.arity}
\leanok
\uses{def:Lower, def:ElimDecl}
\lead{In short} For each source shape, the possible targets -- \hl{no environment-side guard}.
\begin{itemize}
\item an atom, a \code{let} or a projection goes to the congruent shape, component by component;
\item a constructor node goes to a constructor node, argument by argument;
\item a \code{case} goes to a \code{case} with the same inductive, the same parameter count, the same
\textbf{branch count}, and matching branch arities;
\item a \code{fix} in the source has \emph{no image at all} -- \code{fixConst}'s source is a constant,
\code{fixBody}'s is pinned to a lambda;
\item a lambda or a constant goes to itself \emph{or} to a block's \code{fix} node -- either way, not a
runtime key, which lets an eliminator declaration at that key refute both;
\item branch peeling pins the alternative's binder count to the field arity and nothing else.
\end{itemize}
\end{proposition}
\begin{proof}
\leanok
\uses{def:Lower, lem:Lower-ne_fix_of_block, prop:LowerBlock-lambda_of_fixLambda}
\lead{Idea} Case analysis with the source shape fixed; at every shape but a constant and a lambda,
Lemma~\ref{lem:Lower-ne_fix_of_block} excludes the two recursion arms, and \code{elimApp} is excluded
since its source is constant-headed; at a constant, \code{fixBody} is excluded by lambda-headedness,
leaving the runtime-key guard common to both.
\end{proof}

\begin{remark}
\lead{Caveat} The application case is not here: \code{Lower.source\_app} lives in
\code{ErasesCorrect/Steps.lean}, owned by Chapter~\ref{chap:correctness}
(Lemma~\ref{lem:Lower-source_app}).
\end{remark}

\section{The fixpoint round trip}
\label{sec:lowering:fix}

\begin{definition}[Substituting for free variables; the static fix substitution]
\label{def:substFVar}
\lean{LeanToLambdaBox.substFVar, LeanToLambdaBox.substFVarList, LeanToLambdaBox.substFix}
\leanok
\uses{def:LBTerm, def:FixDef, def:LBTerm-fixSubst}
\lead{In short} Three substitution operators with \hl{no de Bruijn bookkeeping under binders}.
\begin{itemize}
\item \code{substFVar}\ x\ s: replaces $x$ by $s$ throughout, every law carrying closedness of $s$;
\code{substFVarList}: iterates it, innermost binding first;
\item \code{substFix}\ ids\ defs: sends $\code{fvar}\ ids[j]$ to $\code{fix}\ defs\ j$ -- the
\emph{static} counterpart of the target's dynamic guarded-fixpoint rule.
\end{itemize}
\lead{Lean} \code{substFix} is built by zipping $ids$ with its own index.
\end{definition}

\begin{lemma}[Abstraction against substitution: the two primitives]
\label{lem:subst_toBvar_self}
\lean{LeanToLambdaBox.subst_toBvar_self, LeanToLambdaBox.subst_toBvar_succ,
LeanToLambdaBox.substList_toBvar, LeanToLambdaBox.substList_toBvar'}
\leanok
\uses{def:substFVar, def:toBvar, def:LBClosed, def:LBTerm-subst, def:hasFVar}

the substitution's internal shift is the identity since $s$ is closed. The simultaneous form iterates
the single step.
\end{proof}

\begin{definition}[The delta-reachability closure]
\label{def:ReachableFrom}
\lean{LeanToLambdaBox.constRefs, LeanToLambdaBox.constRefsArgs, LeanToLambdaBox.constRefsAlts,

\lead{In short} $\code{ReachableFrom}\ \Gamma\ t\ kn$: the fuelled closure, from seed
$\code{constRefs}\ t$, of \hl{everything the program can delta-step to}.
\code{constRefs}\ t collects every kername $t$ names -- constants referenced \emph{and} the
inductive block every constructor, \code{case} or projection node reads; one delta-step unfolds every
name already seen and adds what it references; iterating with fuel $|\Gamma|$ defines the closure.
\lead{Lean} decidable by construction, as a list-bounded closure.
\end{definition}

\begin{theorem}[The fixed fuel saturates]
\label{thm:reachFrom_saturated}
\lean{LeanToLambdaBox.reachFrom_saturated, LeanToLambdaBox.find_stall,
LeanToLambdaBox.expandRefs_stall, LeanToLambdaBox.reachFrom_stall, LeanToLambdaBox.HasBody,
LeanToLambdaBox.HasBody.mem_keys, LeanToLambdaBox.envLookup_mem,
LeanToLambdaBox.mem_expandRefs}
\leanok
\uses{def:ReachableFrom}
\lead{In short} $|\Gamma|$ delta-steps saturate the closure of any seed: \hl{beyond that, nothing new
is reachable}.
\lead{Lean} a list inclusion, resting on a one-step characterisation of what expansion adds.
\lead{Status} \stProved{}.
\end{theorem}
\begin{proof}
\leanok
\uses{def:ReachableFrom}
\lead{Idea} A step unfolding nothing new is a fixed point and stays one; every non-fixed step consumes
one new key with a body, and a successful lookup exhibits the entry it answered with, so a bodied
constant is one of $\Gamma$'s $|\Gamma|$ keys, bounding the search. Uses classical decidability of
kername equality locally, hence \code{Classical.choice} in the footprint.
\end{proof}

\begin{lemma}[Reachability threads through the simulation]
\label{lem:ReachableFrom-through_body}
\lean{LeanToLambdaBox.ReachableFrom.subterm, LeanToLambdaBox.ReachableFrom.app,
LeanToLambdaBox.ReachableFrom.alt, LeanToLambdaBox.ReachableFrom.through_body,
LeanToLambdaBox.ReachableFrom.substList, LeanToLambdaBox.SubTerm.constRefs_subset,
LeanToLambdaBox.reachFrom_add}
\leanok
\uses{def:ReachableFrom, def:SubTerm, def:LBTerm-subst}
\lead{In short} Three threading laws, one per shape the simulation's induction meets.
\begin{itemize}
\item monotonicity: what a \emph{part} of a term reaches, the whole term reaches, and a subterm naming
no kername adds nothing;
\item what a delta-unfolded body reaches, the program reaches -- the closure composes, by
Theorem~\ref{thm:reachFrom_saturated};
\item what an iota reduct or beta contractum reaches, the term and the substituends reach.
\end{itemize}
\end{lemma}
\begin{proof}
\leanok
\uses{def:ReachableFrom, thm:reachFrom_saturated, def:SubTerm}
\lead{Idea} The subterm law is the seed inclusion by induction plus monotonicity; the body law composes
two closures and cuts back with saturation; the substitution law is the corresponding fact about which
kernames a substitution can name, lifted to the closure.
\end{proof}

\lean{LeanToLambdaBox.NoBodylessRefs, LeanToLambdaBox.isBodylessConst,

\leanok

\lead{Lean} decided by a list traversal, its adequacy proved separately.
\end{definition}

\end{remark}

\section*{Deviations from the paper and caveats}
\begin{itemize}
\item \textbf{One-directional.} The eliminator arm drops arguments before the discriminant unchecked:
sound for a forward simulation, not a bisimulation.
\item \textbf{Non-deterministic.} At a block member, both \code{const} and \code{fixConst} apply:
``the pass emits'' means ``there is an emission such that''.
\item \textbf{A manufactured source dialect.} Eliminators are still constants, recursion still by
environment reference, so the pass builds two node kinds MetaRocq's erasure gets free from Rocq's
\code{match}/\code{fix} -- why the fixpoint node is not inert under box transport
(Lemma~\ref{lem:noBox_lower_needs_noFix}).
\item \textbf{Applied, not block, form.} The emitted output is applied-form, while the one fully
verified pass, Theorem~\ref{thm:LBOptimize_correct}, is stated at block-form flags.
\item \textbf{The weaker conclusion.} At the snapshot the emitted program satisfies
Definition~\ref{def:LBWfPeregrine}, \emph{not} \code{PeregrinePre}: the fixpoint eta clause is false
here, a named finding. \lead{Since the snapshot} the finding is fixed (\code{42989c5}) and the two
predicates are merged into one (\code{2ace82d}).
\end{itemize}
